constexpr ImGuiConfigFlags Or(ImGuiConfigFlags first, ImGuiConfigFlags second) {
  return static_cast<ImGuiConfigFlags>(static_cast<unsigned>(first) |
                                       static_cast<unsigned>(second));
}
} // namespace

void ui::ImGuiBackend::Init(GLFWwindow *ui_window, const char *glsl) {
  IMGUI_CHECKVERSION();
  ImGui::CreateContext();

  ImGuiIO &io = ImGui::GetIO();
  io.ConfigFlags = Or(io.ConfigFlags, ImGuiConfigFlags_DockingEnable);
  io.ConfigFlags = Or(io.ConfigFlags, ImGuiConfigFlags_NavEnableKeyboard);

  ImGui::StyleColorsDark();

  ImGui_ImplGlfw_InitForOpenGL(ui_window, /*install_callbacks=*/true);
  ImGui_ImplOpenGL3_Init(glsl);
}

void ui::ImGuiBackend::Shutdown() {
  ImGui_ImplOpenGL3_Shutdown();
  ImGui_ImplGlfw_Shutdown();
  ImGui::DestroyContext();
}

void ui::ImGuiBackend::BeginFrame() {
  ImGui_ImplOpenGL3_NewFrame();
  ImGui_ImplGlfw_NewFrame();
  ImGui::NewFrame();
}

void ui::ImGuiBackend::EndFrame() {
  ImGui::Render();
  ImGui_ImplOpenGL3_RenderDrawData(ImGui::GetDrawData());
}

void ui::ImGuiBackend::DrawDockspace()
{
    ImGuiIO& io = ImGui::GetIO();
    ImGuiViewport* viewport = ImGui::GetMainViewport();

    // Хост-окно на весь экран без фона
    ImGui::SetNextWindowPos(viewport->Pos);
    ImGui::SetNextWindowSize(viewport->Size);
    ImGui::SetNextWindowViewport(viewport->ID);

    ImGuiWindowFlags host_flags =
        ImGuiWindowFlags_NoDocking |
        ImGuiWindowFlags_NoTitleBar |
        ImGuiWindowFlags_NoCollapse |
        ImGuiWindowFlags_NoResize |
        ImGuiWindowFlags_NoMove |
        ImGuiWindowFlags_NoBringToFrontOnFocus |
        ImGuiWindowFlags_NoNavFocus |
        ImGuiWindowFlags_NoBackground; // ← важное: нет фона, сцена видна

    ImGui::PushStyleVar(ImGuiStyleVar_WindowRounding, 0.0f);
    ImGui::PushStyleVar(ImGuiStyleVar_WindowBorderSize, 0.0f);

    ImGui::Begin("##DockSpaceHost", nullptr, host_flags);
    ImGui::PopStyleVar(2);

    // Сам докспейс — с Passthru в центре
    ImGuiID dockspace_id = ImGui::GetID("MainDockSpace");
    ImGuiDockNodeFlags dock_flags = ImGuiDockNodeFlags_PassthruCentralNode; // ← оставляем
    ImGui::DockSpace(dockspace_id, ImVec2(0.0f, 0.0f), dock_flags);

    ImGui::End();
}

#pragma once
#include <memory>
#include <ranges>
#include <type_traits>
#include <vector>

#include "ui/EditorState.h"

namespace ui {
  struct EditorContext;

class Editor {
public:
  Editor() = default;

  // Запрет копирования
  Editor(const Editor &) = delete;
  Editor &operator=(const Editor &) = delete;

  // Перемещение: переносим владение, но сбрасываем стек (переактивация позже)
  Editor(Editor &&other) noexcept
      : m_panels(std::move(other.m_panels)), m_tools(std::move(other.m_tools)),
        m_modules(std::move(other.m_modules)) {
    m_activeStack.clear();
  }

  Editor &operator=(Editor &&other) noexcept {
    if (this != &other) {
      ShutdownModules(); // на всякий случай корректно выгружаем свои
      m_panels = std::move(other.m_panels);
      m_tools = std::move(other.m_tools);
      m_modules = std::move(other.m_modules);
      m_activeStack.clear();
    }
    return *this;
  }

  ~Editor() {
    // Гарантируем Shutdown модулей перед разрушением редактора
    for (auto &m_module : std::ranges::reverse_view(m_modules)) {
      m_module->Shutdown();
    }
  }

  // --- Регистрация панелей/инструментов (как было) ---
  void RegisterPanel(std::unique_ptr<IPanel> panel) {
    m_panels.emplace_back(std::move(panel));
  }
  void RegisterTool(std::unique_ptr<ITool> tool) {
    m_tools.emplace_back(std::move(tool));
  }

  template <class T> T *GetTool() {
    static_assert(std::is_base_of_v<ITool, T>, "T must be ITool");
    for (auto &up : m_tools) {
      if (auto *tool_pointer = dynamic_cast<T *>(up.get())) {
        return tool_pointer;
      }
    }
    return nullptr;
  }
  template <class T> const T *GetTool() const {
    static_assert(std::is_base_of_v<ITool, T>, "T must be ITool");
    for (const auto &up : m_tools) {
      if (auto *tool_pointer = dynamic_cast<const T *>(up.get())) {
        return tool_pointer;
      }
    }
    return nullptr;
  }

  // --- Модули ---
  void RegisterModule(std::unique_ptr<IModule> module) {
    m_modules.emplace_back(std::move(module));
    m_modules.back()->Startup(*this);
  }

  // При желании — явная остановка модулей до деструктора
  void ShutdownModules() {
    for (auto &m_module : std::ranges::reverse_view(m_modules)) {
      m_module->Shutdown();
    }
  }

  // --- Активные инструменты (как уже сделали) ---
  void SetActiveTool(ToolType type, EditorContext &ctx) {
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnDeactivate(ctx);
      m_activeStack.clear();
    }
    if (auto *tool = FindTool(type)) {
      m_activeStack.push_back(tool);
      tool->OnActivate(ctx);
    }
  }
  void PushTool(ToolType type, EditorContext &ctx) {
    if (auto *tool = FindTool(type)) {
      if (!m_activeStack.empty()) {
        m_activeStack.back()->OnDeactivate(ctx);
      }
      m_activeStack.push_back(tool);
      tool->OnActivate(ctx);
    }
  }
  void PopTool(EditorContext &ctx) {
    if (m_activeStack.empty()) {
      return;
    }
    m_activeStack.back()->OnDeactivate(ctx);
    m_activeStack.pop_back();
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnActivate(ctx);
    }
  }

  void DrawUI(EditorContext &ctx) {
    ctx.m_editor = this;
    for (auto &panel : m_panels) {
      panel->Draw(ctx);
    }
  }
  void TickTools(EditorContext &ctx) {
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnUpdate(ctx);
    }
  }

  [[nodiscard]] ToolType ActiveToolType() const {
    return m_activeStack.empty() ? ToolType::OTHER_TOOL
                                 : m_activeStack.back()->Type();
  }
  EditorState m_state;

private:
  ITool *FindTool(ToolType type) {
    for (auto &tool : m_tools) {
      if (tool->Type() == type) {
        return tool.get();
      }
    }
    return nullptr;
  }

  std::vector<std::unique_ptr<IPanel>> m_panels;
  std::vector<std::unique_ptr<ITool>> m_tools;
  std::vector<ITool *> m_activeStack;
  std::vector<std::unique_ptr<IModule>> m_modules; // <-- NEW
};

} // namespace ui

    inline bool DragVec3(const char* label,
                     glm::vec3& v,
                     float speed = 0.1F,
                     float v_min = -FLT_MAX,
                     float v_max =  FLT_MAX,
                     const char* fmt = "%.3f")

void PrefabBrowserPanel::Draw(EditorContext& ctx) {
  bool open = true;
  if (!ImGui::Begin(Id(), &open)) { ImGui::End(); return; }

  const auto& ids = ctx.m_scene->ListPrefabIds();
  if (ids.empty()) {
    ImGui::TextDisabled("No prefabs loaded");
    ImGui::End();
    return;
  }

  // Поддерживаем корректный выбор и m_edit
  const std::string& id = EnsureSelection(ctx, ids);

  // ===== ЛЕВО =====
  ImGui::BeginChild("##left", ImVec2(220, 0), true);
  DrawLeftColumn(ctx, ids);
  ImGui::EndChild();
  ImGui::SameLine();

  // ===== ПРАВО =====
  ImGui::BeginChild("##right", ImVec2(0, 0), true);
  AgentPrefab& P = *m_edit;
  DrawHeader(id);

  DrawType(P);
  DrawModelSection(P);
  DrawBounds(P);
  DrawPhysics(P);
  DrawFirePoints(P);
  DrawVisuals(P);

  ImGui::Separator();
  DrawSaveButtons(ctx, id, P);
  DrawSavePopup(ctx, id, P);
  ImGui::EndChild();

  ImGui::End();
}

const std::string& PrefabBrowserPanel::EnsureSelection(EditorContext& ctx, const std::vector<std::string>& ids) {
  if (m_selectedIndex < 0 || m_selectedIndex >= (int)ids.size()) {
    m_selectedIndex = 0;
    m_edit.reset();
  }
  const std::string& id = ids[m_selectedIndex];
  if (!m_edit.has_value()) {
    m_edit = ctx.m_scene->GetPrefabSystem().GetPrefab(id);
    // Синхронизируем текстовые буферы один раз при загрузке
    std::snprintf(m_modelBuf, sizeof(m_modelBuf), "%s", m_edit->m_modelPath.c_str());
    std::snprintf(m_matNormalBuf,  sizeof(m_matNormalBuf),  "%s", m_edit->normal_material.normal_path.c_str());
    std::snprintf(m_matBurningBuf, sizeof(m_matBurningBuf), "%s", m_edit->normal_material.specular_path.c_str());
    std::snprintf(m_matBurnedBuf,  sizeof(m_matBurnedBuf),  "%s", m_edit->normal_material.diffuse_path.c_str());
  }
  return id;
}

void PrefabBrowserPanel::DrawLeftColumn(EditorContext& ctx, const std::vector<std::string>& ids) {
  for (int i = 0; i < (int)ids.size(); ++i) {
    const bool sel = (i == m_selectedIndex);
    if (ImGui::Selectable(ids[i].c_str(), sel)) {
      m_selectedIndex = i;
      // Перезагружаем редактируемую копию и буферы
      m_edit = ctx.m_scene->GetPrefabSystem().GetPrefab(ids[i]);
      std::snprintf(m_modelBuf, sizeof(m_modelBuf), "%s", m_edit->m_modelPath.c_str());
      std::snprintf(m_matNormalBuf,  sizeof(m_matNormalBuf),  "%s", m_edit->normal_material.normal_path.c_str());
      std::snprintf(m_matBurningBuf, sizeof(m_matBurningBuf), "%s", m_edit->normal_material.specular_path.c_str());
      std::snprintf(m_matBurnedBuf,  sizeof(m_matBurnedBuf),  "%s", m_edit->normal_material.diffuse_path.c_str());
    }
    if (sel) ImGui::SetItemDefaultFocus();
  }
}

  ImGui::Separator();

void PrefabBrowserPanel::DrawType(AgentPrefab& P) {
  int t = (int)P.m_type;
  const char* types[] = { "Tree", "Burned", "Fire", "House" };
  if (ImGui::Combo("Type", &t, types, IM_ARRAYSIZE(types))) {
    P.m_type = static_cast<AgentType>(t);
  }
}

void PrefabBrowserPanel::DrawModelSection(AgentPrefab& P) {
  if (ImGui::InputText("Model path", m_modelBuf, IM_ARRAYSIZE(m_modelBuf))) {
    P.m_modelPath = m_modelBuf;
  }
  DragVec3("Model scale", P.m_modelScale, 0.01F);
}

void PrefabBrowserPanel::DrawBounds(AgentPrefab& P) {
  if (ImGui::CollapsingHeader("Bounds", ImGuiTreeNodeFlags_DefaultOpen)) {
    DragVec3("Min", P.m_bounds_min, 0.01f);
    DragVec3("Max", P.m_bounds_max, 0.01f);
  }
}

void PrefabBrowserPanel::DrawPhysics(AgentPrefab& P) {
  if (ImGui::CollapsingHeader("Physics", ImGuiTreeNodeFlags_DefaultOpen)) {
    ImGui::DragFloat("Intensity", &P.m_physics.m_intencity, 0.05f, 0.0f, 1000.0f);
  }
}

void PrefabBrowserPanel::DrawFirePoints(AgentPrefab& P) {
  if (!ImGui::CollapsingHeader("Fire points", ImGuiTreeNodeFlags_DefaultOpen)) return;

  if (ImGui::Button("Add point")) {
    P.m_firePoints.push_back({glm::vec3(0), 100.0f});
  }
  for (int i = 0; i < (int)P.m_firePoints.size(); ++i) {
    ImGui::PushID(i);
    ImGui::Separator();
    DragVec3("Pos", P.m_firePoints[i].m_position, 0.01f);
    ImGui::DragFloat("Fuel", &P.m_firePoints[i].m_initialFuel, 0.5f, 0.0f, 10000.0f);
    ImGui::SameLine();
    if (ImGui::SmallButton("Remove")) {
      P.m_firePoints.erase(P.m_firePoints.begin() + i);
      ImGui::PopID();
      break; // корректный выход из цикла после erase
    }
    ImGui::PopID();
  }
}

void PrefabBrowserPanel::DrawVisuals(AgentPrefab& P) {
    if (!ImGui::CollapsingHeader("Visuals", ImGuiTreeNodeFlags_DefaultOpen)) return;

    bool changed = false;

    // Пути к текстурам
    changed |= ImGui::InputText("Normal",  m_matNormalBuf,  IM_ARRAYSIZE(m_matNormalBuf));
    changed |= ImGui::InputText("Burning", m_matBurningBuf, IM_ARRAYSIZE(m_matBurningBuf));
    changed |= ImGui::InputText("Burned",  m_matBurnedBuf,  IM_ARRAYSIZE(m_matBurnedBuf));

    // Цвета
    changed |= ImGui::ColorEdit3("Normal diffuse",  &P.normal_material.diffuse_color[0]);
    changed |= ImGui::ColorEdit3("Normal specular", &P.normal_material.specular_color[0]);
    changed |= ImGui::ColorEdit3("Burning diffuse",  &P.burning_material.diffuse_color[0]);
    changed |= ImGui::ColorEdit3("Burning specular", &P.burning_material.specular_color[0]);
    changed |= ImGui::ColorEdit3("Burned diffuse",   &P.burned_material.diffuse_color[0]);
    changed |= ImGui::ColorEdit3("Burned specular",  &P.burned_material.specular_color[0]);

    // Шайнинесс
    changed |= ImGui::DragFloat("Normal shininess",  &P.normal_material.shininess, 1.0f, 0.0f, 256.0f);
    changed |= ImGui::DragFloat("Burning shininess", &P.burning_material.shininess, 1.0f, 0.0f, 256.0f);
    changed |= ImGui::DragFloat("Burned shininess",  &P.burned_material.shininess, 1.0f, 0.0f, 256.0f);

    // Обновляем пути текстур, если изменились
    if (changed) {
        P.normal_material.diffuse_path  = m_matNormalBuf;
        P.normal_material.specular_path = m_matNormalBuf;
        P.normal_material.normal_path   = m_matNormalBuf;

        P.burning_material.diffuse_path  = m_matBurningBuf;
        P.burning_material.specular_path = m_matBurningBuf;
        P.burning_material.normal_path   = m_matBurningBuf;

        P.burned_material.diffuse_path  = m_matBurnedBuf;
        P.burned_material.specular_path = m_matBurnedBuf;
        P.burned_material.normal_path   = m_matBurnedBuf;
    }
}

  static float tConfirm = -1000.f;

if (ImGui::Button("Apply (runtime)")) {
    ctx.m_scene->GetPrefabSystem().SavePrefab(id, P); 
    P.LoadMaterials();    
    ctx.m_scene->ReloadModelForPrefab(id, P); // <-- передаем P напрямую
    tConfirm = (float)ImGui::GetTime();
}
  if ((float)ImGui::GetTime() - tConfirm < 1.0f) {
    ImGui::SameLine(); ImGui::TextColored(ImVec4(0,1,0,1), "Saved");
  }

  ImGui::SameLine();

  if (ImGui::Button("Save to file")) {
    if (m_savePath.empty()) m_savePath = "res/prefabs/" + id + ".yaml";
    std::snprintf(m_saveBuf, sizeof(m_saveBuf), "%s", m_savePath.c_str());
    m_savePopupOpen = true;
    ImGui::OpenPopup("Save Prefab");
        P.LoadMaterials();    
    ctx.m_scene->ReloadModelForPrefab(id, P); // <-- передаем P напрямую
  }
}

void PrefabBrowserPanel::DrawSavePopup(EditorContext& ctx, const std::string& id, AgentPrefab& P) {
  if (!ImGui::BeginPopupModal("Save Prefab", &m_savePopupOpen, ImGuiWindowFlags_AlwaysAutoResize))
    return;

  ImGui::InputText("Path", m_saveBuf, IM_ARRAYSIZE(m_saveBuf));

  if (ImGui::Button("Save")) {
    m_savePath = m_saveBuf;

    auto& sys = ctx.m_scene->GetPrefabSystem();

    // --- новый id берём из имени файла (res/prefabs/<newId>.yaml) ---
    // если хочешь иначе — поменяй стратегию (читать из YAML, хранить поле id и т.д.)
    std::string newId;
    {
      // <filesystem> должен быть подключён где-то один раз в TU
      // #include <filesystem>
      std::filesystem::path out(m_savePath);
      newId = out.stem().string();
      if (newId.empty()) newId = id; // fallback
    }

    // 1) гарантируем актуальность рантайма под НОВЫМ id
    // В большинстве реализаций SavePrefab(id, P) добавляет/заменяет запись в реестре.
    // Если у тебя отдельный метод AddOrReplacePrefab, зови его здесь вместо SavePrefab.
    sys.SavePrefab(newId, P);

    // 2) пишем на диск под НОВЫМ id
    sys.SavePrefabToFile(newId, m_savePath);

    // 3) обновляем список префабов в системе (перескан каталога)
    // Если в твоём PrefabSystem нет ReloadAll(), замени это место на ImportFromFile(m_savePath)
    // или просто оставь шаг 1) — он уже внёс запись в реестр.
    // sys.ReloadAll();

    // 4) перевыбор элемента в левом списке и сброс локального кэша
    m_selectedIndex = 0;
    const auto& ids = ctx.m_scene->ListPrefabIds();
    for (int i = 0; i < (int)ids.size(); ++i) {
      if (ids[i] == newId) { m_selectedIndex = i; break; }
    }
    m_edit.reset(); // EnsureSelection() подхватит свежую копию и синхронизирует буферы

    ImGui::CloseCurrentPopup();
    m_savePopupOpen = false;
  }

  ImGui::SameLine();
  if (ImGui::Button("Cancel")) {
    ImGui::CloseCurrentPopup();
    m_savePopupOpen = false;
  }
  ImGui::EndPopup();
}

#include "ui/tools/PlaceOnGroundTool.h"

#include <GLFW/glfw3.h>
#include <cmath>  // + нужно для std::abs
#include <glm/glm.hpp>
#include <glm/gtc/matrix_inverse.hpp>

#include "math/ScreenRay.h"
#include "ui/EditorContext.h"

namespace ui {

ToolType PlaceOnGroundTool::Type() const noexcept {
  return ToolType::PLACE_ON_GROUND;
}
const char* PlaceOnGroundTool::Name() const noexcept {
  return "Place on ground";
}
const char* PlaceOnGroundTool::Id() const noexcept {
  return "Tool.PlaceOnGround";
}

void PlaceOnGroundTool::OnUpdate(EditorContext& ctx) {
  if (engine::InputSystem::Instance().UIMouseCaptured()) {
    return;
  }
  if (!engine::InputSystem::Instance().GetMouseButton(engine::MouseButton::Left,
                                                      engine::InputState::Pressed)) {
    return;
  }

  // 1) Курсор в логических координатах окна
  double mx = 0.0;
  double my = 0.0;
  glfwGetCursorPos(ctx.m_window, &mx, &my);

  // 2) Окно -> панель вьюпорта (логические пиксели панели)
  const glm::vec2 VPL = ctx.WindowToViewportLogical(static_cast<float>(mx), static_cast<float>(my));
  if (VPL.x < 0.F || VPL.y < 0.F || VPL.x >= static_cast<float>(ctx.vpW) ||
      VPL.y >= static_cast<float>(ctx.vpH)) {
    return;
  }

  // 3) Логические -> framebuffer пиксели (HiDPI масштаб панели)
  const float PX = VPL.x * ctx.m_vpScaleX;
  const float PY = VPL.y * ctx.m_vpScaleY;

  // 4) Размеры framebuffer рендер-области (под них построена проекция)
  const auto FB_W = static_cast<float>(ctx.m_viewportW);
  const auto FB_H = static_cast<float>(ctx.m_viewportH);

  // 5) Луч из пикселя экрана -> мир
  glm::vec3 ro{};
  glm::vec3 rd{};
  math::ScreenPointToRay(PX, PY, FB_W, FB_H, ctx.m_proj, ctx.m_view, ro, rd);

  // 6) Пересечение с плоскостью Y=0
  glm::vec3 hit{};
  if (!math::IntersectRayPlaneY0(ro, rd, hit)) {
    return;
  }

  // 7) Действие
  if (ctx.State().m_placeKind == EditorState::PlaceKind::FIRE) {
    ctx.m_scene->AddFire(hit.x, hit.z, ctx.State().m_fireIntensity);
  } else {
    ctx.m_scene->AddPrefab(ctx.State().m_selectedPrefab, hit.x, hit.z);
  }
}
