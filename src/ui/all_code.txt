constexpr ImGuiConfigFlags Or(ImGuiConfigFlags first, ImGuiConfigFlags second) {
  return static_cast<ImGuiConfigFlags>(static_cast<unsigned>(first) |
                                       static_cast<unsigned>(second));
}
} // namespace

void ui::ImGuiBackend::Init(GLFWwindow *ui_window, const char *glsl) {
  IMGUI_CHECKVERSION();
  ImGui::CreateContext();

  ImGuiIO &io = ImGui::GetIO();
  io.ConfigFlags = Or(io.ConfigFlags, ImGuiConfigFlags_DockingEnable);
  io.ConfigFlags = Or(io.ConfigFlags, ImGuiConfigFlags_NavEnableKeyboard);

  ImGui::StyleColorsDark();

  ImGui_ImplGlfw_InitForOpenGL(ui_window, /*install_callbacks=*/true);
  ImGui_ImplOpenGL3_Init(glsl);
}

void ui::ImGuiBackend::Shutdown() {
  ImGui_ImplOpenGL3_Shutdown();
  ImGui_ImplGlfw_Shutdown();
  ImGui::DestroyContext();
}

void ui::ImGuiBackend::BeginFrame() {
  ImGui_ImplOpenGL3_NewFrame();
  ImGui_ImplGlfw_NewFrame();
  ImGui::NewFrame();
}

void ui::ImGuiBackend::EndFrame() {
  ImGui::Render();
  ImGui_ImplOpenGL3_RenderDrawData(ImGui::GetDrawData());
}

void ui::ImGuiBackend::DrawDockspace()
{
    ImGuiIO& io = ImGui::GetIO();
    ImGuiViewport* viewport = ImGui::GetMainViewport();

    // Хост-окно на весь экран без фона
    ImGui::SetNextWindowPos(viewport->Pos);
    ImGui::SetNextWindowSize(viewport->Size);
    ImGui::SetNextWindowViewport(viewport->ID);

    ImGuiWindowFlags host_flags =
        ImGuiWindowFlags_NoDocking |
        ImGuiWindowFlags_NoTitleBar |
        ImGuiWindowFlags_NoCollapse |
        ImGuiWindowFlags_NoResize |
        ImGuiWindowFlags_NoMove |
        ImGuiWindowFlags_NoBringToFrontOnFocus |
        ImGuiWindowFlags_NoNavFocus |
        ImGuiWindowFlags_NoBackground; // ← важное: нет фона, сцена видна

    ImGui::PushStyleVar(ImGuiStyleVar_WindowRounding, 0.0f);
    ImGui::PushStyleVar(ImGuiStyleVar_WindowBorderSize, 0.0f);

    ImGui::Begin("##DockSpaceHost", nullptr, host_flags);
    ImGui::PopStyleVar(2);

    // Сам докспейс — с Passthru в центре
    ImGuiID dockspace_id = ImGui::GetID("MainDockSpace");
    ImGuiDockNodeFlags dock_flags = ImGuiDockNodeFlags_PassthruCentralNode; // ← оставляем
    ImGui::DockSpace(dockspace_id, ImVec2(0.0f, 0.0f), dock_flags);

    ImGui::End();
}

  engine::Scene* m_scene = nullptr;          // не владеем
  const engine::Camera* m_camera = nullptr;  // не владеем

  // Размеры вьюпорта в framebuffer-пикселях (для Ray/Projection)

  int m_viewportH = 1080;

  // --- NEW: прямоугольник рабочей области (логические пиксели ImGui)
  int vpX{0}, vpY{0}, vpW{1920}, vpH{1080};

  // --- NEW: HiDPI scale (логические → framebuffer)
  float vpScaleX{1.f}, vpScaleY{1.f};

  Editor* m_editor = nullptr;  // не владеем
  enum class PlaceKind { PREFAB, FIRE };
  PlaceKind m_placeKind = PlaceKind::PREFAB;

  std::string selectedPrefab = "default_tree";
  float m_fireIntensity = 1.0f;

  [[nodiscard]] ToolType GetActiveToolType() const noexcept;

  // --- NEW: перевод координаты окна → координаты вьюпорта в framebuffer-пикселях
  [[nodiscard]] glm::vec2 WindowToViewportLogical(float mx, float my) const {
    return {mx - static_cast<float>(vpX), my - static_cast<float>(vpY)};
  }

  // окно -> внутрь вьюпорта (framebuffer-пиксели)
  [[nodiscard]] glm::vec2 WindowToViewportFB(float mx, float my) const {
    return {(mx - static_cast<float>(vpX)) * vpScaleX, (my - static_cast<float>(vpY)) * vpScaleY};
  }
};

#pragma once
#include <memory>

namespace ui {

class Editor {
public:
  Editor() = default;

  // Запрет копирования
  Editor(const Editor &) = delete;
  Editor &operator=(const Editor &) = delete;

  // Перемещение: переносим владение, но сбрасываем стек (переактивация позже)
  Editor(Editor &&other) noexcept
      : m_panels(std::move(other.m_panels)), m_tools(std::move(other.m_tools)),
        m_modules(std::move(other.m_modules)) {
    m_activeStack.clear();
  }

  Editor &operator=(Editor &&other) noexcept {
    if (this != &other) {
      ShutdownModules(); // на всякий случай корректно выгружаем свои
      m_panels = std::move(other.m_panels);
      m_tools = std::move(other.m_tools);
      m_modules = std::move(other.m_modules);
      m_activeStack.clear();
    }
    return *this;
  }

  ~Editor() {
    // Гарантируем Shutdown модулей перед разрушением редактора
    for (auto &m_module : std::ranges::reverse_view(m_modules)) {
      m_module->Shutdown();
    }
  }

  // --- Регистрация панелей/инструментов (как было) ---
  void RegisterPanel(std::unique_ptr<IPanel> panel) {
    m_panels.emplace_back(std::move(panel));
  }
  void RegisterTool(std::unique_ptr<ITool> tool) {
    m_tools.emplace_back(std::move(tool));
  }

  template <class T> T *GetTool() {
    static_assert(std::is_base_of_v<ITool, T>, "T must be ITool");
    for (auto &up : m_tools) {
      if (auto *tool_pointer = dynamic_cast<T *>(up.get())) {
        return tool_pointer;
      }
    }
    return nullptr;
  }
  template <class T> const T *GetTool() const {
    static_assert(std::is_base_of_v<ITool, T>, "T must be ITool");
    for (const auto &up : m_tools) {
      if (auto *tool_pointer = dynamic_cast<const T *>(up.get())) {
        return tool_pointer;
      }
    }
    return nullptr;
  }

  // --- Модули ---
  void RegisterModule(std::unique_ptr<IModule> module) {
    m_modules.emplace_back(std::move(module));
    m_modules.back()->Startup(*this);
  }

  // При желании — явная остановка модулей до деструктора
  void ShutdownModules() {
    for (auto &m_module : std::ranges::reverse_view(m_modules)) {
      m_module->Shutdown();
    }
  }

  // --- Активные инструменты (как уже сделали) ---
  void SetActiveTool(ToolType type, EditorContext &ctx) {
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnDeactivate(ctx);
      m_activeStack.clear();
    }
    if (auto *tool = FindTool(type)) {
      m_activeStack.push_back(tool);
      tool->OnActivate(ctx);
    }
  }
  void PushTool(ToolType type, EditorContext &ctx) {
    if (auto *tool = FindTool(type)) {
      if (!m_activeStack.empty()) {
        m_activeStack.back()->OnDeactivate(ctx);
      }
      m_activeStack.push_back(tool);
      tool->OnActivate(ctx);
    }
  }
  void PopTool(EditorContext &ctx) {
    if (m_activeStack.empty()) {
      return;
    }
    m_activeStack.back()->OnDeactivate(ctx);
    m_activeStack.pop_back();
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnActivate(ctx);
    }
  }

  void DrawUI(EditorContext &ctx) {
    ctx.m_editor = this;
    for (auto &panel : m_panels) {
      panel->Draw(ctx);
    }
  }
  void TickTools(EditorContext &ctx) {
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnUpdate(ctx);
    }
  }

  [[nodiscard]] ToolType ActiveToolType() const {
    return m_activeStack.empty() ? ToolType::OTHER_TOOL
                                 : m_activeStack.back()->Type();
  }

private:
  ITool *FindTool(ToolType type) {
    for (auto &tool : m_tools) {
      if (tool->Type() == type) {
        return tool.get();
      }
    }
    return nullptr;
  }

  std::vector<std::unique_ptr<IPanel>> m_panels;
  std::vector<std::unique_ptr<ITool>> m_tools;
  std::vector<ITool *> m_activeStack;
  std::vector<std::unique_ptr<IModule>> m_modules; // <-- NEW
};

} // namespace ui

#include "ui/panels/PrefabBrowserPanel.h"
#include "ui/EditorContext.h"
#include "core/scene/Scene.h"
#include "data/PrefabSystem.h"
#include <imgui.h>

using engine::data::AgentPrefab;

namespace ui {

void PrefabBrowserPanel::Draw(EditorContext& ctx) {
  bool open = true;
  if (!ImGui::Begin(Id(), &open)) { ImGui::End(); return; }

  const auto& ids = ctx.m_scene->ListPrefabIds();
  if (ids.empty()) {
    ImGui::TextDisabled("No prefabs loaded");
    ImGui::End();
    return;
  }

  // Левый сайдбар: список префабов
  ImGui::BeginChild("##left", ImVec2(200, 0), true);
  for (int i = 0; i < (int)ids.size(); ++i) {
    bool sel = (i == m_selectedIndex);
    if (ImGui::Selectable(ids[i].c_str(), sel)) {
      m_selectedIndex = i;
      // загружаем копию в редактор
      m_edit = ctx.m_scene->GetPrefabSystem().GetPrefab(ids[i]); // const ref -> копия
    }
    if (sel) ImGui::SetItemDefaultFocus();
  }
  ImGui::EndChild();
  ImGui::SameLine();

  // Правая часть: свойства
  ImGui::BeginChild("##right", ImVec2(0, 0), true);
  const std::string& id = ids[m_selectedIndex];
  if (!m_edit.has_value()) {
    m_edit = ctx.m_scene->GetPrefabSystem().GetPrefab(id);
  }
  AgentPrefab& P = *m_edit;

  ImGui::Text("Prefab: %s", id.c_str());
  ImGui::Separator();

  // Тип
  {
    int t = (int)P.type;
    const char* types[] = { "Tree", "Burned", "Fire", "House" /* подставь свои */ };
    if (ImGui::Combo("Type", &t, types, IM_ARRAYSIZE(types))) {
      P.type = static_cast<AgentType>(t);
    }
  }

  // Модель
  {
    char buf[512]; std::snprintf(buf, sizeof(buf), "%s", P.model_path.c_str());
    if (ImGui::InputText("Model path", buf, IM_ARRAYSIZE(buf))) {
      P.model_path = buf;
    }
    DragVec3("Model scale", P.model_scale, 0.01f);
  }

  // Bounds
  if (ImGui::CollapsingHeader("Bounds", ImGuiTreeNodeFlags_DefaultOpen)) {
    DragVec3("Min", P.bounds_min, 0.01f);
    DragVec3("Max", P.bounds_max, 0.01f);
  }

  // Physics
  if (ImGui::CollapsingHeader("Physics", ImGuiTreeNodeFlags_DefaultOpen)) {
    ImGui::DragFloat("Intensity", &P.physics.intencity, 0.05f, 0.0f, 1000.0f);
  }

  // Fire points
  if (ImGui::CollapsingHeader("Fire points", ImGuiTreeNodeFlags_DefaultOpen)) {
    if (ImGui::Button("Add point")) {
      P.fire_points.push_back({glm::vec3(0), 100.0f});
    }
    for (int i = 0; i < (int)P.fire_points.size(); ++i) {
      ImGui::PushID(i);
      ImGui::Separator();
      DragVec3("Pos", P.fire_points[i].position, 0.01f);
      ImGui::DragFloat("Fuel", &P.fire_points[i].initial_fuel, 0.5f, 0.0f, 10000.0f);
      ImGui::SameLine();
      if (ImGui::SmallButton("Remove")) {
        P.fire_points.erase(P.fire_points.begin() + i);
        ImGui::PopID();
        break;
      }
      ImGui::PopID();
    }
  }

  // Visuals (плейсхолдеры под материалы)
  if (ImGui::CollapsingHeader("Visuals", ImGuiTreeNodeFlags_DefaultOpen)) {
    char nbuf[256], bbuf[256], dbuf[256];
    std::snprintf(nbuf, sizeof(nbuf), "%s", P.visuals.normal_material.c_str());
    std::snprintf(bbuf, sizeof(bbuf), "%s", P.visuals.burning_material.c_str());
    std::snprintf(dbuf, sizeof(dbuf), "%s", P.visuals.burned_material.c_str());
    if (ImGui::InputText("Normal", nbuf, IM_ARRAYSIZE(nbuf)))   P.visuals.normal_material  = nbuf;
    if (ImGui::InputText("Burning", bbuf, IM_ARRAYSIZE(bbuf)))  P.visuals.burning_material = bbuf;
    if (ImGui::InputText("Burned", dbuf, IM_ARRAYSIZE(dbuf)))   P.visuals.burned_material  = dbuf;
  }

  ImGui::Separator();
  // Применить в рантайм (в память) и сохранить на диск
  if (ImGui::Button("Apply (runtime)")) {
    ctx.m_scene->GetPrefabSystem().SavePrefab(id, P); // перезаписали в систему
    // если этот префаб сейчас выбран — ничего больше не нужно
  }
  ImGui::SameLine();
  if (ImGui::Button("Save to file")) {
    if (m_savePath.empty()) m_savePath = "res/prefabs/" + id + ".yaml";
    // простейший input под путь
    ImGui::OpenPopup("Save Prefab");
  }

  if (ImGui::BeginPopupModal("Save Prefab", nullptr, ImGuiWindowFlags_AlwaysAutoResize)) {
    static char pathBuf[512];
    std::snprintf(pathBuf, sizeof(pathBuf), "%s", m_savePath.c_str());
    ImGui::InputText("Path", pathBuf, IM_ARRAYSIZE(pathBuf));
    if (ImGui::Button("Save")) {
      m_savePath = pathBuf;
      // перед сохранением убеждаемся, что рантайм обновлён
      ctx.m_scene->GetPrefabSystem().SavePrefab(id, P);
      ctx.m_scene->GetPrefabSystem().SavePrefabToFile(id, m_savePath);
      ImGui::CloseCurrentPopup();
    }
    ImGui::SameLine();
    if (ImGui::Button("Cancel")) {
      ImGui::CloseCurrentPopup();
    }
    ImGui::EndPopup();
  }

  ImGui::EndChild();
  ImGui::End();
}

} // namespace ui

void SceneIOPanel::Draw(EditorContext& ctx) {
  bool open = true;
  if (!ImGui::Begin(Id(), &open)) { ImGui::End(); return; }

  ImGui::TextUnformatted("Scene file");
  ImGui::InputText("##path", m_path.data(), m_path.size());

  ImGui::Separator();

  // Кнопки Load / Save
  if (ImGui::Button("Load")) {
    // Предполагаем, что в Scene добавлены методы: bool LoadScene(const char* path);
    try {
      ctx.m_app->LoadScene(std::string(m_path.data()));
      m_lastOk = true;
    } catch (...) {
      m_lastOk = false;
    }
    m_lastOp = "Load";
  }
  ImGui::SameLine();
  if (ImGui::Button("Save")) {
    try {
      ctx.m_app->SaveScene(std::string(m_path.data()));
      m_lastOk = true;
    } catch (...) {
      m_lastOk = false;
    }
    m_lastOp = "Save";
  }

  // Статус
  ImGui::SameLine();
  if ((m_lastOp != nullptr) && (*m_lastOp != 0)) {
    ImGui::Text("%s: %s", m_lastOp, (m_lastOk ? "OK" : "FAILED"));
  }

  // Быстрые пресеты (опционально)
  if (ImGui::CollapsingHeader("Quick slots", ImGuiTreeNodeFlags_DefaultOpen)) {
    if (ImGui::Button("data/scene1.yml")) { strncpy(m_path.data(), "data/scene1.yml", m_path.size());
}
    ImGui::SameLine();
    if (ImGui::Button("data/scene2.yml")) { strncpy(m_path.data(), "data/scene2.yml", m_path.size());
}
  }

  ImGui::End();
}

// ui/panels/ToolbarPanel.cpp
#include "ui/panels/ToolbarPanel.h"

#include <algorithm>
#include <iostream>
#include "core/Input.h"  // ← у тебя именно так называется инпут
#include "core/scene/Scene.h"
#include "imgui.h"

#include "ui/Editor.h"

namespace ui {

void ToolbarPanel::Draw(EditorContext& ctx) {
  bool visible = true;
  if (!ImGui::Begin(Id(), &visible)) { ImGui::End(); return; }

  const bool isPlace = (ctx.GetActiveToolType() == ToolType::PLACE_ON_GROUND);
  if (ImGui::Selectable("Place on ground", isPlace)) {
    ctx.m_editor->SetActiveTool(ToolType::PLACE_ON_GROUND, ctx);
  }

  ImGui::Separator();

  // --- Чёткая привязка к состоянию без локальных mode ---
bool isPrefab = (ctx.m_placeKind == EditorContext::PlaceKind::PREFAB);
if (ImGui::RadioButton("Prefab##mode", isPrefab)) {
  ctx.m_placeKind = EditorContext::PlaceKind::PREFAB;
  std::cout << "Prefab mode" << std::endl;
}
ImGui::SameLine();
bool isFire = (ctx.m_placeKind == EditorContext::PlaceKind::FIRE);
if (ImGui::RadioButton("Fire##mode", isFire)) {
  ctx.m_placeKind = EditorContext::PlaceKind::FIRE;
  std::cout << "Fire mode" << std::endl;
}

  // --- Огонь ---
  if (ctx.m_placeKind == EditorContext::PlaceKind::FIRE) {
    ImGui::SliderFloat("Fire intensity", &ctx.m_fireIntensity, 0.0f, 10.0f);
    ctx.m_scene->SetFireIntensity(ctx.m_fireIntensity);

    if (ImGui::Button("Ignite at cursor")) {
      double mx=0, my=0; glfwGetCursorPos(ctx.m_window, &mx, &my);
      const glm::vec2 vp = ctx.WindowToViewportFB((float)mx, (float)my);
      if (vp.x >= 0 && vp.y >= 0 && vp.x < (float)ctx.m_viewportW && vp.y < (float)ctx.m_viewportH) {
        glm::vec3 ro, rd;
        math::ScreenPointToRay(vp.x, vp.y,
                               (float)ctx.m_viewportW, (float)ctx.m_viewportH,
                               ctx.m_proj, ctx.m_view, ro, rd);
        glm::vec3 hit;
        if (math::IntersectRayPlaneY0(ro, rd, hit))
          ctx.m_scene->AddFire(hit.x, hit.z, ctx.m_fireIntensity);
      }
    }
    static float fireX = 510.0f, fireZ = 501.0f;
    ImGui::InputFloat("X", &fireX);
    ImGui::InputFloat("Z", &fireZ);
    if (ImGui::Button("Ignite at (X,Z)")) {
      ctx.m_scene->AddFire(fireX, fireZ, ctx.m_fireIntensity);
    }
  }

  // --- Префабы ---
  if (ctx.m_placeKind == EditorContext::PlaceKind::PREFAB) {
    const auto& ids = ctx.m_scene->ListPrefabIds();
    if (ids.empty()) {
      ImGui::TextDisabled("no prefabs loaded");
    } else {
      // гарантируем валидный выбор
      if (ctx.selectedPrefab.empty()
          || std::ranges::find(ids, ctx.selectedPrefab) == ids.end()) {
        ctx.selectedPrefab = ids.front();
      }
      int current = 0;
      for (int i = 0; i < (int)ids.size(); ++i) if (ids[i] == ctx.selectedPrefab) { current = i; break; }

      if (ImGui::BeginCombo("Prefab##combo", ids[current].c_str())) {
        for (int i = 0; i < (int)ids.size(); ++i) {
          const bool sel = (i == current);
          if (ImGui::Selectable(ids[i].c_str(), sel)) {
            ctx.selectedPrefab = ids[i];
          }
          if (sel) ImGui::SetItemDefaultFocus();
        }
        ImGui::EndCombo();
      }
    }
  }

  ImGui::End();
}



}  // namespace ui

#include "ui/tools/PlaceOnGroundTool.h"

#include <GLFW/glfw3.h>
#include <glm/glm.hpp>
#include <glm/gtc/matrix_inverse.hpp>
#include <cmath>                    // + нужно для std::abs

#include "core/Input.h"
#include "core/scene/Scene.h"
#include "ui/EditorContext.h"
#include "math/ScreenRay.h"  
#include <iostream>
using engine::MouseButton;
using engine::InputState;

namespace ui {

ToolType PlaceOnGroundTool::Type() const noexcept { return ToolType::PLACE_ON_GROUND; }
const char* PlaceOnGroundTool::Name() const noexcept { return "Place on ground"; }
const char* PlaceOnGroundTool::Id()   const noexcept { return "Tool.PlaceOnGround"; }

void PlaceOnGroundTool::OnUpdate(EditorContext& ctx) {
  if (engine::InputSystem::Instance().UIMouseCaptured()) return;
  if (!engine::InputSystem::Instance().GetMouseButton(engine::MouseButton::Left,
                                                      engine::InputState::Pressed)) return;

  // 1) Курсор в логических координатах окна
  double mx = 0.0, my = 0.0;
  glfwGetCursorPos(ctx.m_window, &mx, &my);

  // 2) Окно -> панель вьюпорта (логические пиксели панели)
  const glm::vec2 vpl = ctx.WindowToViewportLogical((float)mx, (float)my);
  if (vpl.x < 0.f || vpl.y < 0.f || vpl.x >= (float)ctx.vpW || vpl.y >= (float)ctx.vpH) return;

  // 3) Логические -> framebuffer пиксели (HiDPI масштаб панели)
  const float px = vpl.x * ctx.vpScaleX;   // 0 слева
  const float py = vpl.y * ctx.vpScaleY;   // 0 СВЕРХУ (ровно как ждёт ScreenPointToRay)

  // 4) Размеры framebuffer рендер-области (под них построена проекция)
  const float fbW = (float)ctx.m_viewportW;  // = vpW * vpScaleX
  const float fbH = (float)ctx.m_viewportH;  // = vpH * vpScaleY

  // 5) Луч из пикселя экрана -> мир
  glm::vec3 ro{}, rd{};
  math::ScreenPointToRay(px, py, fbW, fbH, ctx.m_proj, ctx.m_view, ro, rd);

  // 6) Пересечение с плоскостью Y=0
  glm::vec3 hit{};
  if (!math::IntersectRayPlaneY0(ro, rd, hit)) return;

  // (необязательно) отладка — убедиться, что всё консистентно
  // std::cout << "[DBG] win=("<<mx<<","<<my<<") vpLog=("<<vpl.x<<","<<vpl.y
  //           <<") fb=("<<px<<","<<py<<") fbSize=("<<fbW<<","<<fbH<<") hit=("
  //           <<hit.x<<","<<hit.y<<","<<hit.z<<")\n";

  // 7) Действие
if (ctx.m_placeKind == ui::EditorContext::PlaceKind::FIRE) {
  ctx.m_scene->AddFire(hit.x, hit.z, ctx.m_fireIntensity);
} else {
  ctx.m_scene->AddPrefab(ctx.selectedPrefab, hit.x, hit.z);
}

}
