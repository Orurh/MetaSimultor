#pragma once
#include <cstdint>
#include "cuda_include.h"

enum AgentType {
    TREE = 0,
    FIRE_SOURCE = 1,    // Точка горения (бывшая ветка)
    BURNED = 2
};

struct FireCollider {
    float bounds_min[3], bounds_max[3];
    float heat_points[64][3];  // УВЕЛИЧИЛИ до 64 точек!
    float fuels[64];
    float intensities[64];
    int point_count;
    
    __host__ __device__
    FireCollider() : point_count(0) {
        for (int i = 0; i < 3; i++) {
            bounds_min[i] = 0.0f;
            bounds_max[i] = 0.0f;
        }
    }
};

struct Agent {
    float pos_x, pos_y, pos_z;  // Позиция объекта
    AgentType type;
    float intencity;
    int object_id;              // ID объекта
    
    // Универсальный коллайдер для огня
    FireCollider fire_collider;
    
    __host__ __device__
    Agent() : pos_x(0), pos_y(0), pos_z(0), type(TREE), object_id(-1) {}
    
    __host__ __device__
    Agent(float x, float y, float z, AgentType type, int id) 
        : pos_x(x), pos_y(y), pos_z(z), type(type), object_id(id) {}
    
    __host__ __device__
    bool isBurning() const { 
        for (int i = 0; i < fire_collider.point_count; i++) {
            if (fire_collider.intensities[i] > 0.1f) {
                return true;  
            }
        }
        return false;
    }
    
    __host__ __device__
    float getOverallIntensity() const {
        float max = 0.0f;
        for (int i = 0; i < fire_collider.point_count; i++) {
            if (fire_collider.intensities[i] > max) {
                max = fire_collider.intensities[i];
            }
        }
        return max;
    }
};

void AgentManager::setupTreeFireCollider(Agent& tree, float base_x, float base_z) {
    tree.fire_collider.bounds_min[0] = base_x - 2.0f;
    tree.fire_collider.bounds_min[1] = 0.0f;         
    tree.fire_collider.bounds_min[2] = base_z - 2.0f;
    
    tree.fire_collider.bounds_max[0] = base_x + 2.0f;
    tree.fire_collider.bounds_max[1] = 8.0f;         
    tree.fire_collider.bounds_max[2] = base_z + 2.0f;
    
    generateTreeHeatPoints(tree.fire_collider, base_x, base_z);
}
void AgentManager::IgniteRandomPoint(int agent_index)
 {
    if (agent_index < 0 || agent_index >= m_count) return;
    
    Agent agent;
    cudaMemcpy(&agent, m_agents + agent_index, sizeof(Agent), cudaMemcpyDeviceToHost);
    
    if (agent.fire_collider.point_count > 0) {
        int random_point = rand() % agent.fire_collider.point_count;
        agent.fire_collider.intensities[random_point] = 0.5f;
        cudaMemcpy(m_agents + agent_index, &agent, sizeof(Agent), cudaMemcpyHostToDevice);
    }
}
void AgentManager::generateTreeHeatPoints(FireCollider &collider, float base_x, float base_z)
{
    collider.point_count = 0;
    
    for (float height = 0.5f; height <= 8.0f; height += 0.8f) {
        collider.heat_points[collider.point_count][0] = base_x;
        collider.heat_points[collider.point_count][1] = height;
        collider.heat_points[collider.point_count][2] = base_z;
        collider.fuels[collider.point_count] = 30.0f;
        collider.intensities[collider.point_count] = 0.0f;
        collider.point_count++;
    }
    
    float levels[] = {2.0f, 4.0f, 6.0f, 7.5f};
    float level_radii[] = {1.0f, 1.8f, 2.2f, 1.5f};
    int points_per_level[] = {8, 12, 16, 8};
    
    for (int level = 0; level < 4; level++) {
        float height = levels[level];
        float radius = level_radii[level];
        int points = points_per_level[level];
        
        for (int i = 0; i < points; i++) {
            float angle = (float)i / points * 360.0f;
            float rad = glm::radians(angle);
            
            collider.heat_points[collider.point_count][0] = base_x + cos(rad) * radius;
            collider.heat_points[collider.point_count][1] = height;
            collider.heat_points[collider.point_count][2] = base_z + sin(rad) * radius;
            collider.fuels[collider.point_count] = 20.0f;
            collider.intensities[collider.point_count] = 0.0f;
            collider.point_count++;
            
            if (collider.point_count >= 64) break;
        }
        if (collider.point_count >= 64) break;
    }
    
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_real_distribution<float> dist_x(base_x - 2.0f, base_x + 2.0f);
    std::uniform_real_distribution<float> dist_z(base_z - 2.0f, base_z + 2.0f);
    std::uniform_real_distribution<float> dist_y(1.0f, 7.0f);
    
    while (collider.point_count < 50 && collider.point_count < 64) {
        collider.heat_points[collider.point_count][0] = dist_x(gen);
        collider.heat_points[collider.point_count][1] = dist_y(gen);
        collider.heat_points[collider.point_count][2] = dist_z(gen);
        collider.fuels[collider.point_count] = 15.0f;
        collider.intensities[collider.point_count] = 0.0f;
        collider.point_count++;
    }
    
    std::cout << "Generated " << collider.point_count << " points for tree" << std::endl;
}

void AgentManager::AddFire(float x, float y, float intensity) {
    if(m_count >= m_capacity) return;
    
    Agent fire;
    fire.type = FIRE_SOURCE;
    fire.object_id = -1;
    fire.pos_x = x;
    fire.pos_z = y;
    fire.pos_y = 0.0f;
    
    // Простой FireCollider для источника огня
    setupFireSourceCollider(fire, x, y);
    
    // Зажигаем точку
    fire.fire_collider.intensities[0] = intensity;
    
    AddAgent(fire);
}

void AgentManager::setupFireSourceCollider(Agent& fire, float x, float y) {
    fire.fire_collider.bounds_min[0] = x - 0.5f;
    fire.fire_collider.bounds_min[1] = 0.0f;  
    fire.fire_collider.bounds_min[2] = y - 0.5f;
    
    fire.fire_collider.bounds_max[0] = x + 0.5f;
    fire.fire_collider.bounds_max[1] = 2.0f;
    fire.fire_collider.bounds_max[2] = y + 0.5f;
    
    // Одна точка в центре
    fire.fire_collider.point_count = 1;
    fire.fire_collider.heat_points[0][0] = x;
    fire.fire_collider.heat_points[0][1] = 1.0f;
    fire.fire_collider.heat_points[0][2] = y;
    fire.fire_collider.fuels[0] = 50.0f;
    fire.fire_collider.intensities[0] = 0.0f;
}

void AgentManager::AddAgent(const Agent &agent) {
    if (m_count >= m_capacity) {
        std::cerr << "AgentManager: capacity exceeded!" << std::endl;
        return;
    }
    
    cudaMemcpy(m_agents + m_count, &agent, sizeof(Agent), cudaMemcpyHostToDevice);
    m_count++;
}

bool engine::InputSystem::GetKey(KeyCode key, InputState state) const {
  auto currentIt = m_currentKeyState.find(key);
  auto previousIt = m_previousKeyState.find(key);

  bool current =
      currentIt != m_currentKeyState.end() ? currentIt->second : false;
  bool previous =
      previousIt != m_previousKeyState.end() ? previousIt->second : false;

  switch (state) {
  case InputState::Held:
    return current;
  case InputState::Pressed:
    return current && !previous;
  case InputState::Released:
    return !current && previous;
  }
  return false;
}
void engine::InputSystem::SetMousePosition(float x, float y) {
    if (m_mousePosition != glm::vec2(0.0f)) {
        m_mouseDelta = glm::vec2(x, y) - m_mousePosition;
    }
    m_mousePosition = glm::vec2(x, y);
}

void engine::InputSystem::SetMouseButtonState(int button, bool pressed) {
    m_mouseButtonState[button] = pressed;
}

    switch (state) {
    case InputState::Held:
        return current;
    case InputState::Pressed:
        return current && !previous;
    case InputState::Released:
        return !current && previous;
    }
    return false;
}

#pragma once
#include <glm/glm.hpp>
#include <glm/gtc/matrix_transform.hpp>
#define YAW -90.0f
#define PITCH -40.0f
#define SPEED 50.0f
#define SENSITIVITY 0.1f
#define ZOOM 45.0f
namespace engine {

enum Camera_Movement { FORWARD, BACKWARD, LEFT, RIGHT };

class Camera {
   public:
    Camera(glm::vec3 position = glm::vec3(0.0f, 0.0f, -3.0f), glm::vec3 up = glm::vec3(0.0f, 1.0f, 0.0f),
           float yaw = YAW, float pitch = PITCH)
        : m_front(glm::vec3(0.0f, 0.0f, 1.0f)),
          m_movementSpeed(SPEED),
          m_mouseSensitivity(SENSITIVITY),
          m_zoom(ZOOM) {
        m_position = position;
        m_worldUp = up;
        m_yaw = yaw;
        m_pitch = pitch;
        updateCameraVectors();
    }

    glm::mat4 GetViewMatrix() { return glm::lookAt(m_position, m_position + m_front, m_up); }

    void ProcessKeyboard(Camera_Movement direction, float deltaTime) {
        float velocity = m_movementSpeed * deltaTime;
        if (direction == FORWARD) m_position += m_front * velocity;
        if (direction == BACKWARD) m_position -= m_front * velocity;
        if (direction == LEFT) m_position -= m_right * velocity;
        if (direction == RIGHT) m_position += m_right * velocity;
    }

    void ProcessMouseMovement(float xoffset, float yoffset, bool constrainPitch = true) {
        xoffset *= m_mouseSensitivity;
        yoffset *= m_mouseSensitivity;

        m_yaw += xoffset;
        m_pitch -= yoffset;

        if (constrainPitch) {
            if (m_pitch > 89.0f) m_pitch = 89.0f;
            if (m_pitch < -89.0f) m_pitch = -89.0f;
        }

        updateCameraVectors();
    }

    void updateCameraVectors() {
        // calculate the new Front vector
        glm::vec3 front;
        front.x = cos(m_yaw * (M_PI / 180.0)) * cos(m_pitch * (M_PI / 180.0));
        front.y = sin(m_pitch * (M_PI / 180.0));
        front.z = sin(m_yaw * (M_PI / 180.0)) * cos(m_pitch * (M_PI / 180.0));
           if (front.length() > 0) {
        m_front = glm::normalize(front);
    }

    // also re-calculate the Right and Up vector
    m_right = glm::normalize(glm::cross(m_front, m_worldUp));
    m_up = glm::normalize(glm::cross(m_right, m_front));
    }

    void ProcessMouseScroll(float yoffset) {
        m_zoom += (float)yoffset;
        if (m_zoom < 1.0f) m_zoom = 1.0f;
        if (m_zoom > 45.0f) m_zoom = 45.0f;
    }

    float GetZoom() const { return m_zoom; }
    glm::vec3 GetPosition() const {return m_position;}
   private:
    glm::vec3 m_position;
    glm::vec3 m_front;
    glm::vec3 m_up;
    glm::vec3 m_right;
    glm::vec3 m_worldUp;
    // euler Angles
    float m_yaw;
    float m_pitch;
    // camera options
    float m_movementSpeed;
    float m_mouseSensitivity;
    float m_zoom;
};
}

#include <glm/glm.hpp>
#include <glm/gtc/matrix_transform.hpp>

namespace engine {

struct TransformComponent {
  glm::vec3 position;
  glm::vec3 rotation = {0.0, 0.0, 0.0};
  glm::vec3 scale = {1.0, 1.0, 1.0};

  TransformComponent() = default;
  TransformComponent(const glm::vec3 &pos) : position(pos) {}
  TransformComponent(const TransformComponent &other) = default;

  glm::mat4 GetTransform() const {
    glm::vec3 rotationRad = glm::radians(rotation);
    auto rot = glm::toMat4(glm::quat(rotationRad));
    auto trans = glm::translate(glm::mat4(1.0f), position);
    auto scl = glm::scale(glm::mat4(1.0f), scale);
    return trans * rot * scl;
  }
};
}

void Scene::Update(float dt) {
    SimulationStep(m_agentManager.getDeviceAgents(), m_agentManager.getAgentCount(), dt);
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::W)) {
        m_camera.ProcessKeyboard(FORWARD, dt);
    }
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::S)) {
        m_camera.ProcessKeyboard(BACKWARD, dt);
    }
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::D)) {
        m_camera.ProcessKeyboard(RIGHT, dt);
    }
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::A)) {
        m_camera.ProcessKeyboard(LEFT, dt);
    }
    
    static glm::vec2 lastMousePos = engine::InputSystem::Instance().GetMousePosition();
    
    auto currentMousePos = engine::InputSystem::Instance().GetMousePosition();
    glm::vec2 mouseDelta = currentMousePos - lastMousePos;
    lastMousePos = currentMousePos;
    
    if (engine::InputSystem::Instance().GetMouseButton(GLFW_MOUSE_BUTTON_LEFT, InputState::Held)) {
        if (mouseDelta != glm::vec2(0.0f)) {
            m_camera.ProcessMouseMovement(mouseDelta.x, mouseDelta.y);
        }
    }
    float scrollY = engine::InputSystem::Instance().GetMouseScrollY();
    if (scrollY != 0.0f) {
        m_camera.ProcessMouseScroll(scrollY);
        engine::InputSystem::Instance().ResetMouseScroll();
    }
}

    ModelData AssimpLoader::LoadModel(const std::string& path) {
        Assimp::Importer importer;
        const aiScene* scene = importer.ReadFile(path, ModelData::IMPORT_FLAGS);
        
        if (!scene || scene->mFlags & AI_SCENE_FLAGS_INCOMPLETE || !scene->mRootNode) {
            throw std::runtime_error("Assimp error: " + std::string(importer.GetErrorString()));
        }
        
        ModelData modelData;
        modelData.directory = path.substr(0, path.find_last_of('/'));
        
        for (unsigned int i = 0; i < scene->mNumMaterials; i++) {
            modelData.materials.push_back(
                ProcessMaterial(scene->mMaterials[i], modelData.directory)
            );
        }
        
        ProcessNode(scene->mRootNode, scene, modelData);
        
        return modelData;
    }

    void AssimpLoader::ProcessNode(aiNode *node, const aiScene *scene, ModelData &modelData)
    {
        for(unsigned int i = 0; i < node->mNumMeshes; i++)
        {
            aiMesh* mesh = scene->mMeshes[node->mMeshes[i]];
            modelData.meshes.push_back(ProcessMesh(mesh, scene));
        }
        for(unsigned int i = 0; i < node->mNumChildren; i++)
        {
            ProcessNode(node->mChildren[i], scene, modelData);
        }
    }

    MeshData AssimpLoader::ProcessMesh(aiMesh *mesh, const aiScene *scene)
    {
        MeshData meshData;
        meshData.materialIndex = mesh->mMaterialIndex;
        
        for(unsigned int i = 0; i < mesh->mNumVertices; i++) {
            Vertex vertex;

            // Позиции
            vertex.position = glm::vec3(
                mesh->mVertices[i].x,
                mesh->mVertices[i].y, 
                mesh->mVertices[i].z
            );

            // Нормали
            if(mesh->HasNormals()) {
                vertex.normal = glm::vec3(
                    mesh->mNormals[i].x,
                    mesh->mNormals[i].y,
                    mesh->mNormals[i].z
                );
            }

            // Текстурные координаты (берем первый набор)
            if(mesh->mTextureCoords[0]) {
                vertex.texCoords = glm::vec2(
                    mesh->mTextureCoords[0][i].x,
                    mesh->mTextureCoords[0][i].y
                );
            } else {
                vertex.texCoords = glm::vec2(0.0f);
            }

            // Тангенты и битангенты (если есть)
            if(mesh->HasTangentsAndBitangents()) {
                vertex.tangent = glm::vec3(
                    mesh->mTangents[i].x,
                    mesh->mTangents[i].y,
                    mesh->mTangents[i].z
                );
                vertex.bitangent = glm::vec3(
                    mesh->mBitangents[i].x,
                    mesh->mBitangents[i].y, 
                    mesh->mBitangents[i].z
                );
            }

            meshData.vertices.push_back(vertex);
        }

        // Обрабатываем индексы
        for(unsigned int i = 0; i < mesh->mNumFaces; i++) {
            aiFace face = mesh->mFaces[i];
            for(unsigned int j = 0; j < face.mNumIndices; j++) {
                meshData.indices.push_back(face.mIndices[j]);
            }
        }

        return meshData;
    }
    MaterialData AssimpLoader::ProcessMaterial(aiMaterial* material, const std::string& directory) {
        MaterialData materialData;

        aiString name;
        if (material->Get(AI_MATKEY_NAME, name) == AI_SUCCESS) {
            materialData.name = name.C_Str();
        }

        // Диффузный цвет
        aiColor3D diffuse(0.f, 0.f, 0.f);
        if (material->Get(AI_MATKEY_COLOR_DIFFUSE, diffuse) == AI_SUCCESS) {
            materialData.diffuseColor = glm::vec3(diffuse.r, diffuse.g, diffuse.b);
        }

        // Спекулярный цвет
        aiColor3D specular(0.f, 0.f, 0.f);
        if (material->Get(AI_MATKEY_COLOR_SPECULAR, specular) == AI_SUCCESS) {
            materialData.specularColor = glm::vec3(specular.r, specular.g, specular.b);
        }

        // Блеск
        float shininess = 32.0f;
        if (material->Get(AI_MATKEY_SHININESS, shininess) == AI_SUCCESS) {
            materialData.shininess = shininess;
        }

        auto diffuseMaps = LoadMaterialTextures(material, aiTextureType_DIFFUSE, "texture_diffuse", directory);
        materialData.textures.insert(materialData.textures.end(), diffuseMaps.begin(), diffuseMaps.end());

        auto specularMaps = LoadMaterialTextures(material, aiTextureType_SPECULAR, "texture_specular", directory);
        materialData.textures.insert(materialData.textures.end(), specularMaps.begin(), specularMaps.end());

        auto normalMaps = LoadMaterialTextures(material, aiTextureType_NORMALS, "texture_normal", directory);
        materialData.textures.insert(materialData.textures.end(), normalMaps.begin(), normalMaps.end());
        std::cout << "Material: " << materialData.name 
              << " textures: "
              << materialData.textures.size()
              << " diffuse: (" << materialData.diffuseColor.r 
              << ", "
              << materialData.diffuseColor.g
              << ", " 
              << materialData.diffuseColor.b << ")"
              << std::endl;
        return materialData;
    }

    std::vector<TextureData> AssimpLoader::LoadMaterialTextures(
        aiMaterial* mat, aiTextureType type, const std::string& typeName, const std::string& directory) {
        
        std::vector<TextureData> textures;
        
        for (unsigned int i = 0; i < mat->GetTextureCount(type); i++) {
            aiString str;
            mat->GetTexture(type, i, &str);

            TextureData textureData;
            textureData.type = typeName;
            textureData.path = directory + "/" + str.C_Str();

            textures.push_back(textureData);
        }

        return textures;
    }

            texInfo.texture->Bind(i);
            
            std::string number;
            std::string name = "texture_";
            
            if (texInfo.type == "diffuse")
                number = std::to_string(diffuseNr++);
            else if (texInfo.type == "specular")
                number = std::to_string(specularNr++);
            else if (texInfo.type == "normal")
                number = std::to_string(normalNr++);
                
            shader.SetInt(("material." + name + number).c_str(), i);
        }
        
        glActiveTexture(GL_TEXTURE0);
        
        SetupMaterial(shader, m_material);
        m_vao->Bind();
        glDrawElements(GL_TRIANGLES, m_indexCount, GL_UNSIGNED_INT, 0);
        glBindVertexArray(0);

    void Mesh::SetupMesh(const data::MeshData& meshData) {

        m_vbo = VertexBuffer::Create(
            meshData.vertices.size() * sizeof(data::Vertex)
        );
        m_vbo->SetData(meshData.vertices.data(), meshData.vertices.size() * sizeof(data::Vertex));

        m_vbo->SetLayout({
            {ShaderDataType::Float3, "a_position"},
            {ShaderDataType::Float3, "a_normal"}, 
            {ShaderDataType::Float2, "a_texCoords"},
            {ShaderDataType::Float3, "a_tangent"},

    void Model::SetupModel()
    {
        for (size_t i = 0; i < m_modelData.meshes.size(); i++) {
            const auto& meshData = m_modelData.meshes[i];
            
            const data::MaterialData* material = nullptr;
            if (meshData.materialIndex < m_modelData.materials.size()) {
                material = &m_modelData.materials[meshData.materialIndex];
            } else {
                static data::MaterialData defaultMaterial;
                material = &defaultMaterial;
            }
            
            data::MaterialData materialWithTextures = *material;
            for (auto& texInfo : materialWithTextures.textures) {
                bool found = false;
                for (const auto& cachedTex : m_texturesLoaded) {
                    if (cachedTex.path == texInfo.path) {
                        texInfo.texture = cachedTex.texture;
                        found = true;
                        break;
                    }
                }
                
                if (!found) {
                    texInfo.texture = Texture::Create(texInfo.path);
                    m_texturesLoaded.push_back(texInfo);
                }
            }
            
            m_meshes.emplace_back(meshData, materialWithTextures);
        }
    }

OpenGLShader::OpenGLShader(const std::string &filename) {
  std::ifstream stream(filename);
  std::string line;
  enum ShaderType { NONE = -1, VERTEX = 0, GEOMETRY = 1, FRAGMENT = 2 }; 
  std::stringstream ss[3];
  ShaderType type = ShaderType::NONE;
  
  while (getline(stream, line)) {
    if (line.find("#shader") != std::string::npos) {
      if (line.find("vertex") != std::string::npos) {
        type = ShaderType::VERTEX;
      } else if (line.find("geometry") != std::string::npos) { // ДОБАВИЛ
        type = ShaderType::GEOMETRY;
      } else if (line.find("fragment") != std::string::npos) {
        type = ShaderType::FRAGMENT;
      }
    } else {
      if (type != ShaderType::NONE) {
        ss[(int)type] << line << '\n';
      }
    }
  }

  m_rendererID = glCreateProgram();
  
  GLuint vertexShader = CompileShader(GL_VERTEX_SHADER, ss[0].str());
  glAttachShader(m_rendererID, vertexShader);
  
  if (!ss[1].str().empty()) {
    GLuint geometryShader = CompileShader(GL_GEOMETRY_SHADER, ss[1].str());
    glAttachShader(m_rendererID, geometryShader);
  }
  
  GLuint fragmentShader = CompileShader(GL_FRAGMENT_SHADER, ss[2].str());
  glAttachShader(m_rendererID, fragmentShader);
  
  glLinkProgram(m_rendererID);
  
  GLint isLinked;
  glGetProgramiv(m_rendererID, GL_LINK_STATUS, &isLinked);
  if (!isLinked) {
    GLint maxLength;
    glGetProgramiv(m_rendererID, GL_INFO_LOG_LENGTH, &maxLength);
    std::string infoLog(maxLength, ' ');
    glGetProgramInfoLog(m_rendererID, maxLength, &maxLength, &infoLog[0]);
    std::cout << "Shader linking failed: " + infoLog << '\n';
  }
}

OpenGLShader::OpenGLShader(const std::string &vertexSrc,
                           const std::string &geometrySrc,
                           const std::string &fragmentSrc) {
  m_rendererID = glCreateProgram();
  
  GLuint vertexShader = CompileShader(GL_VERTEX_SHADER, vertexSrc);
  GLuint geometryShader = CompileShader(GL_GEOMETRY_SHADER, geometrySrc);
  GLuint fragmentShader = CompileShader(GL_FRAGMENT_SHADER, fragmentSrc);
  
  glAttachShader(m_rendererID, vertexShader);
  glAttachShader(m_rendererID, geometryShader);
  glAttachShader(m_rendererID, fragmentShader);
  
  glLinkProgram(m_rendererID);
  
  // ПРОВЕРКА ОШИБОК
  GLint isLinked;

    std::cout << "Shader linking failed: " + infoLog << '\n';
  }
}

GLuint OpenGLShader::CompileShader(GLenum type, const std::string &source) {
  GLuint shader = glCreateShader(type);
  const char *src = source.c_str();
  glShaderSource(shader, 1, &src, nullptr);
  glCompileShader(shader);
  GLint isCompiled;
  glGetShaderiv(shader, GL_COMPILE_STATUS, &isCompiled);
  if (!isCompiled) {
    GLint maxLength;
    glGetShaderiv(shader, GL_INFO_LOG_LENGTH, &maxLength);
    std::string infoLog(maxLength, ' ');
    glGetShaderInfoLog(shader, maxLength, &maxLength, &infoLog[0]);
    glDeleteShader(shader);
    std::cout << "Shader compilation failed: " + infoLog << '\n';
  }
  return shader;
}

  stbi_uc *data = stbi_load(path.c_str(), &width, &height, &channels, 0);
  if (data) {

    m_width = width;
    m_height = height;

    GLenum internalFormat = 0, dataFormat = 0;

    if (channels == 4) {
      internalFormat = GL_RGBA8;
      dataFormat = GL_RGBA;
    } else if (channels == 3) {
      internalFormat = GL_RGB8;
      dataFormat = GL_RGB;
    } else {
      stbi_image_free(data);
      throw std::runtime_error("Unsupported image format: " + path);
    }

    glCreateTextures(GL_TEXTURE_2D, 1, &m_rendererID);
    glTextureStorage2D(m_rendererID, 1, internalFormat, m_width, m_height);

    glTextureParameteri(m_rendererID, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTextureParameteri(m_rendererID, GL_TEXTURE_MAG_FILTER, GL_LINEAR);

    glTextureParameteri(m_rendererID, GL_TEXTURE_WRAP_S, GL_REPEAT);
    glTextureParameteri(m_rendererID, GL_TEXTURE_WRAP_T, GL_REPEAT);

    glTextureSubImage2D(m_rendererID, 0, 0, 0, m_width, m_height, dataFormat,
                        GL_UNSIGNED_BYTE, data);

    stbi_image_free(data);
  } else {
    throw std::runtime_error("Failed to load texture: " + path);
  }

static GLenum ShaderDataTypeToOpenGLBaseType(ShaderDataType type) {
  switch (type) {
  case ShaderDataType::Float:
    return GL_FLOAT;
  case ShaderDataType::Float2:
    return GL_FLOAT;
  case ShaderDataType::Float3:
    return GL_FLOAT;
  case ShaderDataType::Float4:
    return GL_FLOAT;
  case ShaderDataType::Mat3:
    return GL_FLOAT;
  case ShaderDataType::Mat4:
    return GL_FLOAT;
  case ShaderDataType::Int:
    return GL_INT;
  case ShaderDataType::Int2:
    return GL_INT;
  case ShaderDataType::Int3:
    return GL_INT;
  case ShaderDataType::Int4:
    return GL_INT;
  case ShaderDataType::Bool:
    return GL_BOOL;
  }
  return 0;
}

void OpenGLVertexArrayObject::AddVertexBuffer(
    const std::shared_ptr<VertexBuffer> &vertexBuffer) {
  glBindVertexArray(m_id);
  vertexBuffer->Bind();
  const auto &layout = vertexBuffer->GetLayout();

  for (const auto &elem : layout) {
    switch (elem.type) {
    case ShaderDataType::Float:
    case ShaderDataType::Float2:
    case ShaderDataType::Float3:
    case ShaderDataType::Float4: {
      glEnableVertexAttribArray(m_VertexBufferIndex);
      glVertexAttribPointer(m_VertexBufferIndex, elem.GetComponentCount(),
                            ShaderDataTypeToOpenGLBaseType(elem.type),
                            elem.normalized ? GL_TRUE : GL_FALSE,
                            layout.GetStride(),
                            reinterpret_cast<void *>(elem.offset));
      m_VertexBufferIndex++;
      break;
    }

    case ShaderDataType::Int:
    case ShaderDataType::Int2:
    case ShaderDataType::Int3:
    case ShaderDataType::Int4:
    case ShaderDataType::Bool: {
      glEnableVertexAttribArray(m_VertexBufferIndex);

      glVertexAttribIPointer(m_VertexBufferIndex, elem.GetComponentCount(),
                             ShaderDataTypeToOpenGLBaseType(elem.type),
                             layout.GetStride(),
                             reinterpret_cast<void *>(elem.offset));
      m_VertexBufferIndex++;
      break;
    }

    case ShaderDataType::Mat3:
    case ShaderDataType::Mat4: {
      uint8_t count = elem.GetComponentCount();
      for (uint8_t i = 0; i < count; ++i) {
        glEnableVertexAttribArray(m_VertexBufferIndex);
        glVertexAttribPointer(
            m_VertexBufferIndex, count,
            ShaderDataTypeToOpenGLBaseType(elem.type),
            elem.normalized ? GL_TRUE : GL_FALSE, layout.GetStride(),
            reinterpret_cast<void *>(elem.offset + sizeof(float) * count * i));
        glVertexAttribDivisor(m_VertexBufferIndex, 1);
        m_VertexBufferIndex++;
      }
    }
    default:
      std::cout << "Unknown ShaderDataType!" << std::endl;
    }
  }
  m_vertexBuffers.push_back(vertexBuffer);
}

void OpenGLVertexArrayObject::SetIndexBuffer(
    const std::shared_ptr<IndexBuffer> &indexBuffer) {
  glBindVertexArray(m_id);
  indexBuffer->Bind();
  m_indexBuffer = indexBuffer;
}

void Renderer::UpdateFireParticles(float dt, const std::vector<Agent> &burningAgents)
{
     std::cout << "UpdateFireParticles: " << burningAgents.size() << " burning agents, " 
              << m_fireParticles.size() << " existing particles" << std::endl;
    
    for (auto it = m_fireParticles.begin(); it != m_fireParticles.end();) {
        it->lifetime -= dt;
        it->velocity.y += 2.0f * dt; // Гравитация
        it->position += it->velocity * dt;
        it->size *= 0.98f; // Постепенно уменьшаем
        
        if (it->lifetime <= 0.0f) {
            it = m_fireParticles.erase(it);
        } else {
            ++it;
        }
    }
    
    // Спауним новые частицы для горящих агентов
    m_particleSpawnTimer += dt;
    if (m_particleSpawnTimer >= 0.016f) { // ~60 FPS
        m_particleSpawnTimer = 0.0f;
        
        for (const auto& agent : burningAgents) {
            if (!agent.isBurning()) continue;
            
            // Находим горящие точки для этого агента
            for (int i = 0; i < agent.fire_collider.point_count; i++) {
                float intensity = agent.fire_collider.intensities[i];
                if (intensity <= 0.1f) continue;
                
                glm::vec3 point_pos = glm::vec3(
                    agent.fire_collider.heat_points[i][0],  
                    agent.fire_collider.heat_points[i][1],  
                    agent.fire_collider.heat_points[i][2]   
                );
                
                // Количество частиц зависит от интенсивности
                int particles_to_spawn = 1 + static_cast<int>(intensity * 3);
                
                for (int p = 0; p < particles_to_spawn && m_fireParticles.size() < 1000; p++) {
                    FireParticle particle;
                    particle.position = point_pos;
                    
                    // Случайное направление
                    float angle = static_cast<float>(rand()) / RAND_MAX * 6.2832f;
                    float speed = 1.0f + static_cast<float>(rand()) / RAND_MAX * 2.0f;
                    
                    particle.velocity = glm::vec3(
                        cos(angle) * 0.3f,
                        speed * 1.5f, 
                        sin(angle) * 0.3f
                    );
                    
                    particle.lifetime = 0.5f + static_cast<float>(rand()) / RAND_MAX * 0.3f;
                    particle.max_lifetime = particle.lifetime;
                    particle.size = 0.2f + intensity * 0.4f;
                    
                    // Цвет от желтого к красному
                    float color_mix = static_cast<float>(rand()) / RAND_MAX;
                    particle.color = glm::mix(
                        glm::vec3(1.0f, 0.8f, 0.1f), // Желтый
                        glm::vec3(1.0f, 0.3f, 0.0f), // Красный
                        color_mix
                    );
                    
                    m_fireParticles.push_back(particle);
                }
            }
        }
    }
}
void Renderer::RenderFireParticles()
{
    if (m_fireParticles.empty()) return;
    
    m_modelMatrices.clear();
    m_colors.clear();
    
    for (const auto& particle : m_fireParticles) {
        float life_ratio = particle.lifetime / particle.max_lifetime;
        
        glm::mat4 model = glm::mat4(1.0f);
        model = glm::translate(model, particle.position);
        model = glm::scale(model, glm::vec3(particle.size * life_ratio));
        
        m_modelMatrices.push_back(model);
        m_colors.push_back(particle.color * life_ratio); // Блекнут со временем
    }
    
    // Инстансинг частиц
    std::vector<float> instanceData;
    for (size_t i = 0; i < m_fireParticles.size(); i++) {
        const glm::mat4& model = m_modelMatrices[i];
        for (int col = 0; col < 4; col++) {
            for (int row = 0; row < 4; row++) {
                instanceData.push_back(model[col][row]);
            }
        }
        instanceData.push_back(m_colors[i].x);
        instanceData.push_back(m_colors[i].y);
        instanceData.push_back(m_colors[i].z);
    }
    
    m_instanceVBO->SetData(instanceData.data(), instanceData.size() * sizeof(float));
    
    m_shader->Bind();
    m_shader->SetMat4("u_view", m_viewMat);
    m_shader->SetMat4("u_projection", m_projectionMat);
    
    // Включить blending для прозрачности
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE);
    
    m_vao->Bind();
    glDrawElementsInstanced(GL_TRIANGLES, 36, GL_UNSIGNED_INT, 0, m_fireParticles.size());
    m_vao->Unbind();
    
    glDisable(GL_BLEND);
    m_shader->Unbind();
}
void Renderer::RenderFire(const std::vector<Agent> &agents, float dt)
{
    std::vector<Agent> burningAgents;
    for (const auto& agent : agents) {
        if (agent.isBurning()) {
            burningAgents.push_back(agent);
        }
    }
    
    UpdateFireParticles(dt, burningAgents);
    RenderFireParticles();
}

void Renderer::RenderWithGeometryShader()
 {
    // Настройка рендеринга
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE); // Аддитивное смешение для огня
    glDepthMask(GL_FALSE); // Огонь не пишет в глубину
    
    m_fireGeometryShader->Bind();
    m_fireGeometryShader->SetMat4("u_view", m_viewMat);
    m_fireGeometryShader->SetMat4("u_projection", m_projectionMat);
    m_fireGeometryShader->SetFloat("u_time", glfwGetTime());
    
    // Передаем векторы камеры для billboard
    glm::mat4 view = m_viewMat;
    glm::vec3 camera_right = glm::vec3(view[0][0], view[1][0], view[2][0]);
    glm::vec3 camera_up = glm::vec3(view[0][1], view[1][1], view[2][1]);
    
    m_fireGeometryShader->SetVec3("u_camera_right", camera_right);
    m_fireGeometryShader->SetVec3("u_camera_up", camera_up);
    
    // Рендерим точки - геометрический шейдер сделает остальное
    m_agentVAO->Bind();
    glDrawArrays(GL_POINTS, 0, m_agentRenderData.size());
    m_agentVAO->Unbind();
    
    m_fireGeometryShader->Unbind();
    
    glDepthMask(GL_TRUE);
    glDisable(GL_BLEND);
}

void Renderer::DrawAgents(const std::vector<Agent> &agents)
{
    if (agents.empty()) return;
    m_modelMatrices.clear();
    m_colors.clear();
    
    for (const auto& agent : agents) {

        for (int i = 0; i < agent.fire_collider.point_count; i++) {

             glm::vec3 world_pos = glm::vec3(
                agent.fire_collider.heat_points[i][0],  
                agent.fire_collider.heat_points[i][1],  
                agent.fire_collider.heat_points[i][2]   
            );
         
            glm::mat4 model = glm::mat4(1.0f);
            model = glm::translate(model, world_pos);
            
            float intensity = agent.fire_collider.intensities[i];
            if(intensity <= 0.1) {
                continue;
            }
            float scale = 1 + intensity * 0.3f; 
            model = glm::scale(model, glm::vec3(scale));
            
            m_modelMatrices.push_back(model);
            
            glm::vec3 color;
            if (intensity > 0.1f) {
                color = glm::vec3(1.0f, 1.0f - intensity * 0.7f, 0.0f);
            } else {
                color = glm::vec3(0.0f, 0.0f, 1.0f);
            }
            m_colors.push_back(color);
        }

    if (m_modelMatrices.empty()) return;

    std::vector<float> instanceData;
    for (size_t i = 0; i < m_modelMatrices.size(); i++) {
        const glm::mat4& model = m_modelMatrices[i];
        for (int col = 0; col < 4; col++) {
            for (int row = 0; row < 4; row++) {
                instanceData.push_back(model[col][row]);
            }
        }
        instanceData.push_back(m_colors[i].x);
        instanceData.push_back(m_colors[i].y);
        instanceData.push_back(m_colors[i].z);
    }

    m_instanceVBO->SetData(instanceData.data(), instanceData.size() * sizeof(float));

    m_shader->Bind();
    m_shader->SetMat4("u_view", m_viewMat);
    m_shader->SetMat4("u_projection", m_projectionMat);
    
    m_vao->Bind();
    glDrawElementsInstanced(GL_TRIANGLES, 36, GL_UNSIGNED_INT, 0, m_modelMatrices.size());
    m_vao->Unbind();
    m_shader->Unbind();
}
}

enum class ShaderDataType {
  Float,
  Float2,
  Float3,
  Float4,
  Int,
  Int2,
  Int3,
  Int4,
  Mat3,
  Mat4,
  Bool

static uint32_t ShaderDataTypeSize(ShaderDataType type) {
  switch (type) {
  case ShaderDataType::Float:
    return 4;
  case ShaderDataType::Float2:
    return 4 * 2;
  case ShaderDataType::Float3:
    return 4 * 3;
  case ShaderDataType::Float4:
    return 4 * 4;
  case ShaderDataType::Mat3:
    return 4 * 3 * 3;
  case ShaderDataType::Mat4:
    return 4 * 4 * 4;
  case ShaderDataType::Int:
    return 4;
  case ShaderDataType::Int2:
    return 4 * 2;
  case ShaderDataType::Int3:
    return 4 * 3;
  case ShaderDataType::Int4:
    return 4 * 4;
  case ShaderDataType::Bool:
    return 1;
  }
  return 0;
}
struct BufferElement {
  ShaderDataType type;
  std::string name;
  uint32_t size;
  uint32_t offset;
  bool normalized;
  BufferElement(ShaderDataType type, const std::string &name,
                bool normalized = false)
      : name(name), type(type), size(ShaderDataTypeSize(type)), offset(0),
        normalized(normalized) {}
  uint32_t GetComponentCount() const {
    switch (type) {
    case ShaderDataType::Float:
      return 1;
    case ShaderDataType::Float2:
      return 2;
    case ShaderDataType::Float3:
      return 3;
    case ShaderDataType::Float4:
      return 4;
    case ShaderDataType::Int:
      return 1;
    case ShaderDataType::Int2:
      return 2;
    case ShaderDataType::Int3:
      return 3;
    case ShaderDataType::Int4:
      return 4;
    case ShaderDataType::Mat3:
      return 3;
    case ShaderDataType::Mat4:
      return 4;
    case ShaderDataType::Bool:
      return 1;
    }
    return 0;
  }
};

class BufferLayout {
private:
  std::vector<BufferElement> m_elements;
  uint32_t m_Stride = 0;

public:
  BufferLayout() = default;
  BufferLayout(const std::initializer_list<BufferElement> &elements)
      : m_elements(elements) {
    CalculateOffsetsAndStride();
  }
  const std::vector<BufferElement> &GetElements() const { return m_elements; }
  uint32_t GetStride() const { return m_Stride; }
  // Итераторы для for-each
  std::vector<BufferElement>::iterator begin() { return m_elements.begin(); }
  std::vector<BufferElement>::iterator end() { return m_elements.end(); }
  std::vector<BufferElement>::const_iterator begin() const {
    return m_elements.begin();
  }
  std::vector<BufferElement>::const_iterator end() const {
    return m_elements.end();
  }

private:
  void CalculateOffsetsAndStride() {
    size_t offset = 0;
    m_Stride = 0;
    for (auto &element : m_elements) {
      element.offset = offset;
      offset += element.size;
      m_Stride += element.size;
    }
  }
};

#pragma once
#include <cmath>
#include <glm/glm.hpp>
#include <glm/gtc/matrix_inverse.hpp>

namespace math {
inline void ScreenPointToRay(float px, float py, float viewportW, float viewportH,
                             const glm::mat4& proj, const glm::mat4& view,
                             glm::vec3& outOrigin, glm::vec3& outDir) {
    float x = (2.0F * px / viewportW) - 1.0F;
    float y = 1.0F - (2.0F * py / viewportH); // flip Y
    glm::vec4 ray_clip(x, y, -1.0F, 1.0F);

    glm::mat4 inv_proj = glm::inverse(proj);
    glm::vec4 ray_eye = inv_proj * ray_clip; // camera space
    ray_eye = glm::vec4(ray_eye.x, ray_eye.y, -1.0f, 0.0f);

    glm::mat4 inv_view = glm::inverse(view);
    glm::vec4 ray_world4 = inv_view * ray_eye;
    glm::vec3 ray_world = glm::normalize(glm::vec3(ray_world4));

    auto cam_pos = glm::vec3(inv_view[3]);
    outOrigin = cam_pos;
    outDir = ray_world;
}


inline bool IntersectRayPlaneY0(const glm::vec3& ro, const glm::vec3& rd, glm::vec3& hit) {
    if (std::fabs(rd.y) < 1e-6f) return false;
    float t = -ro.y / rd.y;
    if (t < 0.0f) return false;
    hit = ro + rd * t;
    return true;
}

}  // namespace math
