void EditorExtension::OnInit(engine::Application &app) {
  m_imgui = std::make_unique<ui::ImGuiBackend>();
  m_imgui->Init(static_cast<GLFWwindow *>(app.GetNativeWindow()),
                "#version 330");

  m_editor = std::make_unique<ui::Editor>();
  m_editor->RegisterModule(std::make_unique<ui::BasicToolsModule>());
}

void EditorExtension::OnBeginUI(engine::Application & /*app*/,
                                ui::EditorContext &ctx) {
  // 1) начинаем кадр ImGui
  m_imgui->BeginFrame();

  // 2) рендерим докспейс и панели (построение графа окон — до Scene::Render)
  m_imgui->DrawDockspace();
  m_editor->DrawUI(ctx);
  m_editor->TickTools(ctx);
}

void EditorExtension::OnEndUI(engine::Application &, ui::EditorContext &) {
  // 3) завершаем кадр ImGui (Render + RenderDrawData + платформенные окна)
  m_imgui->EndFrame();
}

void EditorExtension::OnShutdown(engine::Application &) {
  m_editor.reset(); // сначала редактор
  if (m_imgui) {
    m_imgui->Shutdown();
  }
  m_imgui.reset();
}

#pragma once
#include <cstdint>

#include "cuda_include.h"
enum AgentType {
    TREE = 0,
    FIRE_SOURCE = 1,   
    BURNED = 2
};

struct Float3 {
    float x, y, z;
    
    __host__ __device__ Float3() : x(0), y(0), z(0) {}
    __host__ __device__ Float3(float x, float y, float z) : x(x), y(y), z(z) {}
    __host__ __device__ static Float3 fromGlm(const glm::vec3& v) {
        return Float3(v.x, v.y, v.z);
    }
    
    __host__ __device__ glm::vec3 toGlm() const {
        return glm::vec3(x, y, z);
    }
    __host__ __device__ float& operator[](int i) { 
        return i == 0 ? x : (i == 1 ? y : z); 
    }
    __host__ __device__ float operator[](int i) const { 
        return i == 0 ? x : (i == 1 ? y : z); 
    }
};

struct FireCollider {
    Float3 bounds_min, bounds_max;
    Float3 heat_points[64];
    float fuels[64];
    float intensities[64];
    int point_count;
    
    __host__ __device__
    FireCollider() : point_count(0) {
        for (int i = 0; i < 3; i++) {
            bounds_min[i] = 0.0f;
            bounds_max[i] = 0.0f;
        }
    }
};

struct Agent {
    Float3 position;
    AgentType type;
    float intencity;
    int object_id; 
    
    // Универсальный коллайдер для огня
    FireCollider fire_collider;
    
    __host__ __device__
    Agent() : position(), type(TREE), object_id(-1) {}
    
    __host__ __device__
    Agent(float x, float y, float z, AgentType type, int id) 
        : position(x, y, z), type(type), object_id(id) {}
    
    __host__ __device__
    bool isBurning() const { 
        for (int i = 0; i < fire_collider.point_count; i++) {
            if (fire_collider.intensities[i] > 0.1f) {
                return true;  
            }
        }
        return false;
    }
    
    __host__ __device__
    float getOverallIntensity() const {
        float max = 0.0f;
        for (int i = 0; i < fire_collider.point_count; i++) {
            if (fire_collider.intensities[i] > max) {
                max = fire_collider.intensities[i];
            }
        }
        return max;
    }
};

Agent AgentManager::AddTree(int tree_id, float base_x, float base_z) {
    Agent tree;
    tree.type = TREE;
    tree.object_id = tree_id;
    tree.position = {base_x, 0.0f, base_z};
    
    setupTreeFireCollider(tree, base_x, base_z);
    
    AddAgent(tree);
    return tree;
}

void AgentManager::setupTreeFireCollider(Agent& tree, float base_x, float base_z) {
    tree.fire_collider.bounds_min.x = base_x - 2.0f;
    tree.fire_collider.bounds_min.y = 0.0f;         
    tree.fire_collider.bounds_min.z = base_z - 2.0f;
    
    tree.fire_collider.bounds_max.x = base_x + 2.0f;
    tree.fire_collider.bounds_max.y = 8.0f;         
    tree.fire_collider.bounds_max.z = base_z + 2.0f;
    
    generateTreeHeatPoints(tree.fire_collider, base_x, base_z);
}

void AgentManager::IgniteRandomPoint(int agent_index)
 {
    if (agent_index < 0 || agent_index >= m_count) return;
    Agent agent;
    cudaMemcpy(&agent, m_agents + agent_index, sizeof(Agent), cudaMemcpyDeviceToHost);
    
    if (agent.fire_collider.point_count > 0) {
        int random_point = rand() % agent.fire_collider.point_count;
        agent.fire_collider.intensities[random_point] = 1.0f;
        cudaMemcpy(m_agents + agent_index, &agent, sizeof(Agent), cudaMemcpyHostToDevice);
    }
}

void AgentManager::generateTreeHeatPoints(FireCollider &collider, float base_x, float base_z)
{
    collider.point_count = 0;
    
    for (float height = 0.5f; height <= 8.0f; height += 0.8f) {
        collider.heat_points[collider.point_count].x = base_x;
        collider.heat_points[collider.point_count].y = height;
        collider.heat_points[collider.point_count].z = base_z;
        collider.fuels[collider.point_count] = 30.0f;
        collider.intensities[collider.point_count] = 0.0f;
        collider.point_count++;
    }
    
    float levels[] = {2.0f, 4.0f, 6.0f, 7.5f};
    float level_radii[] = {1.0f, 1.8f, 2.2f, 1.5f};
    int points_per_level[] = {8, 12, 16, 8};
    
    for (int level = 0; level < 4; level++) {
        float height = levels[level];
        float radius = level_radii[level];
        int points = points_per_level[level];
        
        for (int i = 0; i < points; i++) {
            float angle = (float)i / points * 360.0f;
            float rad = glm::radians(angle);
            
            collider.heat_points[collider.point_count].x = base_x + cos(rad) * radius;
            collider.heat_points[collider.point_count].y = height;
            collider.heat_points[collider.point_count].z = base_z + sin(rad) * radius;
            collider.fuels[collider.point_count] = 20.0f;
            collider.intensities[collider.point_count] = 0.0f;
            collider.point_count++;
            
            if (collider.point_count >= 64) break;
        }
        if (collider.point_count >= 64) break;
    }
    
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_real_distribution<float> dist_x(base_x - 2.0f, base_x + 2.0f);
    std::uniform_real_distribution<float> dist_z(base_z - 2.0f, base_z + 2.0f);
    std::uniform_real_distribution<float> dist_y(1.0f, 7.0f);
    
    while (collider.point_count < 50 && collider.point_count < 64) {
        collider.heat_points[collider.point_count].x = dist_x(gen);
        collider.heat_points[collider.point_count].y = dist_y(gen);
        collider.heat_points[collider.point_count].z = dist_z(gen);
        collider.fuels[collider.point_count] = 15.0f;
        collider.intensities[collider.point_count] = 0.0f;
        collider.point_count++;
    }
}

void AgentManager::AddFire(float x, float y, float intensity) {
    if(m_count >= m_capacity) return;
    
    Agent fire;
    fire.type = FIRE_SOURCE;
    fire.object_id = -1;
    // fire.position = {x, y, 0.0f};
    fire.position = {x, 0.0f, y};  
    
    setupFireSourceCollider(fire, x, y);
    
    fire.fire_collider.intensities[0] = intensity;
    
    AddAgent(fire);
}

void AgentManager::setupFireSourceCollider(Agent& fire, float x, float y) {
    fire.fire_collider.bounds_min.x = x - 0.5f;
    fire.fire_collider.bounds_min.y = 0.0f;  
    fire.fire_collider.bounds_min.z = y - 0.5f;
    
    fire.fire_collider.bounds_max.x = x + 0.5f;
    fire.fire_collider.bounds_max.y = 2.0f;
    fire.fire_collider.bounds_max.z = y + 0.5f;
    
    fire.fire_collider.point_count = 1;
    fire.fire_collider.heat_points[0].x = x;
    fire.fire_collider.heat_points[0].y = 1.0f;
    fire.fire_collider.heat_points[0].z = y;
    fire.fire_collider.fuels[0] = 50.0f;
    fire.fire_collider.intensities[0] = 0.0f;
}

void AgentManager::AddAgent(const Agent &agent) {
    if (m_count >= m_capacity) {
        std::cerr << "AgentManager: capacity exceeded!" << std::endl;
        return;
    }
    
    cudaMemcpy(m_agents + m_count, &agent, sizeof(Agent), cudaMemcpyHostToDevice);
    m_count++;
}

Application::Application(int width, int height, const char *title)
    : m_last_frame_time(static_cast<float>(glfwGetTime())), m_width(width),
      m_height(height) {
  // Создание окна/GL
  m_window = std::make_unique<Window>(width, height, std::string(title));
  InitGL();

  // Сцена
  m_scene = std::make_unique<Scene>();
  m_scene->Init();

engine::Application::~Application() = default; 

void Application::AddExtension(std::unique_ptr<IAppExtension> ext) {
  m_extensions.push_back(std::move(ext));
}

void Application::Run() {
  for (auto& e : m_extensions) e->OnInit(*this);

  while (!ShouldClose()) {
    TickCore();

    // Собираем контекст на кадр (размеры, матрицы и т.п.)
    ui::EditorContext ctx{};
    ctx.m_window    = GetNativeWindow();
    ctx.m_scene     = m_scene.get();
    ctx.m_viewportW = GetViewportWidth();
    ctx.m_viewportH = GetViewportHeight();
    ctx.m_view      = m_scene->GetViewMatrix();
    ctx.m_proj      = m_scene->GetProjMatrix(ctx.m_viewportW, ctx.m_viewportH);

    // === ВАЖНО: сначала начинаем UI-кадр (NewFrame + построение окон) ===
    for (auto& e : m_extensions) e->OnBeginUI(*this, ctx);

    // Теперь ImGui::GetIO().WantCaptureMouse/Keyboard — АКТУАЛЬНЫЕ
    UpdateScene();
    RenderScene();

    // Завершаем UI (Render + RenderDrawData)
    for (auto& e : m_extensions) e->OnEndUI(*this, ctx);

    Swap();
  }

  for (auto& e : m_extensions) e->OnShutdown(*this);
}


void Application::InitGL() {
  // активировать GL-контекст окна (если нужно), инициализировать glad
  auto *win = static_cast<GLFWwindow *>(GetNativeWindow());
  glfwMakeContextCurrent(win);

  if (gladLoadGLLoader((GLADloadproc)glfwGetProcAddress) == 0) {
    throw std::runtime_error("Failed to initialize GLAD");
  }
  glViewport(0, 0, m_width, m_height);
}

void Application::TickCore() {
  // dt
  auto now = static_cast<float>(glfwGetTime());
  m_dt = now - m_last_frame_time;
  m_last_frame_time = now;

  // события окна/инпут
  m_window->PollEvents();
}

void Application::UpdateScene() { m_scene->Update(m_dt); }

void Application::RenderScene() { m_scene->Render(); }

void Application::Swap() { m_window->SwapBuffers(); }

bool Application::ShouldClose() const { return m_window->ShouldClose(); }

GLFWwindow* engine::Application::GetNativeWindow() const {
  return m_window->GetWindow();
}

int Application::GetViewportWidth() const { return m_width; }
int Application::GetViewportHeight() const { return m_height; }
float Application::GetDeltaTime() const { return m_dt; }
Scene &Application::GetScene() { return *m_scene; }

bool engine::InputSystem::GetKey(KeyCode key, InputState state) const {
  auto currentIt = m_currentKeyState.find(key);
  auto previousIt = m_previousKeyState.find(key);

  bool current =
      currentIt != m_currentKeyState.end() ? currentIt->second : false;
  bool previous =
      previousIt != m_previousKeyState.end() ? previousIt->second : false;

  switch (state) {
  case InputState::Held:
    return current;
  case InputState::Pressed:
    return current && !previous;
  case InputState::Released:
    return !current && previous;
  }
  return false;
}
void engine::InputSystem::SetMousePosition(float x, float y) {
    if (m_mousePosition != glm::vec2(0.0f)) {
        m_mouseDelta = glm::vec2(x, y) - m_mousePosition;
    }
    m_mousePosition = glm::vec2(x, y);
}

void engine::InputSystem::SetMouseButtonState(int button, bool pressed) {
    m_mouseButtonState[button] = pressed;
}


bool engine::InputSystem::GetMouseButton(int button, InputState state) const {
    auto currentIt = m_mouseButtonState.find(button);
    auto previousIt = m_prevMouseButtonState.find(button);

    bool current = currentIt != m_mouseButtonState.end() ? currentIt->second : false;
    bool previous = previousIt != m_prevMouseButtonState.end() ? previousIt->second : false;

    switch (state) {
    case InputState::Held:
        return current;
    case InputState::Pressed:
        return current && !previous;
    case InputState::Released:
        return !current && previous;
    }
    return false;
}

    m_currentKeyState[key] = pressed;

#pragma once
#include <glm/glm.hpp>
#include <glm/gtc/matrix_transform.hpp>
#define YAW -90.0f
#define PITCH -40.0f
#define SPEED 50.0f
#define SENSITIVITY 0.1f
#define ZOOM 45.0f
namespace engine {

enum Camera_Movement { FORWARD, BACKWARD, LEFT, RIGHT };

class Camera {
   public:
    Camera(glm::vec3 position = glm::vec3(0.0f, 0.0f, -3.0f), glm::vec3 up = glm::vec3(0.0f, 1.0f, 0.0f),
           float yaw = YAW, float pitch = PITCH)
        : m_front(glm::vec3(0.0f, 0.0f, 1.0f)),
          m_movementSpeed(SPEED),
          m_mouseSensitivity(SENSITIVITY),
          m_zoom(ZOOM) {
        m_position = position;
        m_worldUp = up;
        m_yaw = yaw;
        m_pitch = pitch;
        updateCameraVectors();
    }

    glm::mat4 GetViewMatrix() { return glm::lookAt(m_position, m_position + m_front, m_up); }

    void ProcessKeyboard(Camera_Movement direction, float deltaTime) {
        float velocity = m_movementSpeed * deltaTime;
        if (direction == FORWARD) m_position += m_front * velocity;
        if (direction == BACKWARD) m_position -= m_front * velocity;
        if (direction == LEFT) m_position -= m_right * velocity;
        if (direction == RIGHT) m_position += m_right * velocity;
    }

    void ProcessMouseMovement(float xoffset, float yoffset, bool constrainPitch = true) {
        xoffset *= m_mouseSensitivity;
        yoffset *= m_mouseSensitivity;

        m_yaw += xoffset;
        m_pitch -= yoffset;

        if (constrainPitch) {
            if (m_pitch > 89.0f) m_pitch = 89.0f;
            if (m_pitch < -89.0f) m_pitch = -89.0f;
        }

        updateCameraVectors();
    }

    void updateCameraVectors() {
        // calculate the new Front vector
        glm::vec3 front;
        front.x = cos(m_yaw * (M_PI / 180.0)) * cos(m_pitch * (M_PI / 180.0));
        front.y = sin(m_pitch * (M_PI / 180.0));
        front.z = sin(m_yaw * (M_PI / 180.0)) * cos(m_pitch * (M_PI / 180.0));
           if (front.length() > 0) {
        m_front = glm::normalize(front);
    }

    // also re-calculate the Right and Up vector
    m_right = glm::normalize(glm::cross(m_front, m_worldUp));
    m_up = glm::normalize(glm::cross(m_right, m_front));
    }

    void ProcessMouseScroll(float yoffset) {
        m_zoom += (float)yoffset;
        if (m_zoom < 1.0f) m_zoom = 1.0f;
        if (m_zoom > 45.0f) m_zoom = 45.0f;
    }

    float GetZoom() const { return m_zoom; }
    glm::vec3 GetPosition() const {return m_position;}
   private:
    glm::vec3 m_position;
    glm::vec3 m_front;
    glm::vec3 m_up;
    glm::vec3 m_right;

struct TransformComponent {
  glm::vec3 position;
  glm::vec3 rotation = {0.0, 0.0, 0.0};
  glm::vec3 scale = {1.0, 1.0, 1.0};

  TransformComponent() = default;
  TransformComponent(const glm::vec3 &pos) : position(pos) {}
  TransformComponent(const TransformComponent &other) = default;

  glm::mat4 GetTransform() const {
    glm::vec3 rotationRad = glm::radians(rotation);
    auto rot = glm::toMat4(glm::quat(rotationRad));
    auto trans = glm::translate(glm::mat4(1.0f), position);
    auto scl = glm::scale(glm::mat4(1.0f), scale);
    return trans * rot * scl;
  }
};

void Scene::Update(float dt) {
  // 0) Симуляция — всегда идёт
  SimulationStep(m_agentManager.getDeviceAgents(),
                 m_agentManager.getAgentCount(), dt);

  // 1) Флаги захвата UI
  const ImGuiIO& io = ImGui::GetIO();
  const bool ui_mouse = io.WantCaptureMouse;
  const bool ui_kbd   = io.WantCaptureKeyboard;

  // 2) WASD — только если UI не держит клавиатуру
  if (!ui_kbd) {
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::W)) m_camera.ProcessKeyboard(FORWARD,  dt);
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::S)) m_camera.ProcessKeyboard(BACKWARD, dt);
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::D)) m_camera.ProcessKeyboard(RIGHT,    dt);
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::A)) m_camera.ProcessKeyboard(LEFT,     dt);
  }

  // 3) Поворот камеры — только если UI не держит мышь И зажат RMB
  static glm::vec2 lastMousePos = engine::InputSystem::Instance().GetMousePosition();
  const auto curMousePos = engine::InputSystem::Instance().GetMousePosition();

  if (!ui_mouse && engine::InputSystem::Instance()
                       .GetMouseButton(GLFW_MOUSE_BUTTON_RIGHT, InputState::Held)) {
    const glm::vec2 delta = curMousePos - lastMousePos;
    if (delta != glm::vec2(0.0f)) {
      m_camera.ProcessMouseMovement(delta.x, delta.y);
    }
  }
  // обновляем last всегда в конце кадра, чтобы дельта считалась корректно
  lastMousePos = curMousePos;

  // 4) Постановка объекта — ОДНОРАЗОВО на клик ЛКМ и только если UI не держит мышь
  if (!ui_mouse && engine::InputSystem::Instance()
                      .GetMouseButton(GLFW_MOUSE_BUTTON_LEFT, InputState::Pressed)) {
    const glm::vec2 mp = curMousePos;
    const glm::vec3 hit = ScreenToGround(mp, m_vpW, m_vpH);
    if (std::isfinite(hit.x) && std::isfinite(hit.z)) {
      if (m_placeMode == PlaceMode::Tree) {
        AddTree(hit.x, hit.z);
      } else {
        m_agentManager.AddFire(hit.x, hit.z, m_fireIntensity);
      }
    }
  }

  // 5) Зум — только если UI не держит мышь/скролл
  if (!ui_mouse) {
    const float scrollY = engine::InputSystem::Instance().GetMouseScrollY();
    if (scrollY != 0.0f) {
      m_camera.ProcessMouseScroll(scrollY);
      engine::InputSystem::Instance().ResetMouseScroll();
    }
  }
}


glm::vec3 Scene::ScreenToGround(const glm::vec2& mousePx, int w, int h) {
    float x =  2.0f * mousePx.x / float(w) - 1.0f;
    float y =  1.0f - 2.0f * mousePx.y / float(h);
    glm::mat4 proj = glm::perspective(m_camera.GetZoom(), float(w)/float(h), 0.1f, 2000.0f);
    glm::mat4 view = m_camera.GetViewMatrix();
    glm::mat4 invVP = glm::inverse(proj * view);
    glm::vec4 pNear = invVP * glm::vec4(x, y, -1.0f, 1.0f);
    glm::vec4 pFar  = invVP * glm::vec4(x, y,  1.0f, 1.0f);
    pNear /= pNear.w; pFar /= pFar.w;
    glm::vec3 origin = glm::vec3(pNear);
    glm::vec3 dir    = glm::normalize(glm::vec3(pFar - pNear));
    if (std::abs(dir.y) < 1e-6f) return glm::vec3(std::numeric_limits<float>::quiet_NaN());
    float t = -origin.y / dir.y;
    return origin + t * dir;
}

void Scene::Render() {
    auto projection = glm::perspective(m_camera.GetZoom(), (float)1920 / (float)1080, 0.1f, 2000.0f);
    auto view = m_camera.GetViewMatrix();
    m_renderer.BeginScene(view, projection);
    
    auto agents = LoadAgentsFromGPU(); 
    
     for (const auto& obj : m_sceneObjects) {
        if (!obj.is_visible) continue;
        
        if (obj.agent_id < agents.size()) {
            const Agent& agent = agents[obj.agent_id];
            RenderAgent(agent);
        }
    }
    
    std::vector<Agent> burningBranches;
    for (const auto& agent : agents) {
        if ((agent.type == AgentType::TREE || agent.type == AgentType::BURNED)) {
            burningBranches.push_back(agent);
        }
    }
    m_renderer.UploadAgentsData(agents); 
    m_renderer.RenderWithGeometryShader();
    
    m_renderer.EndScene();
}

void Scene::AddTree(float x, float y)
{
    static int tree_counter = 0;
    
    const auto& prefab = m_prefabSystem.GetPrefab("default_tree");
    Agent agent = m_prefabSystem.CreateAgentFromPrefab(prefab, {x, 0, y});
    m_agentManager.AddAgent(agent);
    int agent_index = m_agentManager.getAgentCount() - 1;
    SceneObject obj;
    obj.agent_id = agent_index;
    obj.model_path = prefab.model_path; 
    obj.position = glm::vec3(x, 0, y);
    obj.is_visible = true;
    m_sceneObjects.push_back(obj);
    tree_counter++;
}

    }
}

        case AgentType::BURNED:

            break;
    }   
}

    ModelData AssimpLoader::LoadModel(const std::string& path) {
        Assimp::Importer importer;
        const aiScene* scene = importer.ReadFile(path, ModelData::IMPORT_FLAGS);
        
        if (!scene || scene->mFlags & AI_SCENE_FLAGS_INCOMPLETE || !scene->mRootNode) {
            throw std::runtime_error("Assimp error: " + std::string(importer.GetErrorString()));
        }
        
        ModelData modelData;
        modelData.directory = path.substr(0, path.find_last_of('/'));
        
        for (unsigned int i = 0; i < scene->mNumMaterials; i++) {
            modelData.materials.push_back(
                ProcessMaterial(scene->mMaterials[i], modelData.directory)
            );
        }
        
        ProcessNode(scene->mRootNode, scene, modelData);
        
        return modelData;
    }

    void AssimpLoader::ProcessNode(aiNode *node, const aiScene *scene, ModelData &modelData)
    {
        for(unsigned int i = 0; i < node->mNumMeshes; i++)
        {
            aiMesh* mesh = scene->mMeshes[node->mMeshes[i]];
            modelData.meshes.push_back(ProcessMesh(mesh, scene));
        }
        for(unsigned int i = 0; i < node->mNumChildren; i++)
        {
            ProcessNode(node->mChildren[i], scene, modelData);
        }
    }

    MeshData AssimpLoader::ProcessMesh(aiMesh *mesh, const aiScene *scene)
    {
        MeshData meshData;
        meshData.materialIndex = mesh->mMaterialIndex;
        
        for(unsigned int i = 0; i < mesh->mNumVertices; i++) {
            Vertex vertex;

            // Позиции
            vertex.position = glm::vec3(
                mesh->mVertices[i].x,
                mesh->mVertices[i].y, 
                mesh->mVertices[i].z
            );

            // Нормали
            if(mesh->HasNormals()) {
                vertex.normal = glm::vec3(
                    mesh->mNormals[i].x,
                    mesh->mNormals[i].y,
                    mesh->mNormals[i].z
                );
            }

            // Текстурные координаты (берем первый набор)
            if(mesh->mTextureCoords[0]) {
                vertex.texCoords = glm::vec2(
                    mesh->mTextureCoords[0][i].x,
                    mesh->mTextureCoords[0][i].y
                );
            } else {
                vertex.texCoords = glm::vec2(0.0f);
            }

            // Тангенты и битангенты (если есть)
            if(mesh->HasTangentsAndBitangents()) {
                vertex.tangent = glm::vec3(
                    mesh->mTangents[i].x,
                    mesh->mTangents[i].y,
                    mesh->mTangents[i].z
                );
                vertex.bitangent = glm::vec3(
                    mesh->mBitangents[i].x,
                    mesh->mBitangents[i].y, 
                    mesh->mBitangents[i].z
                );
            }

            meshData.vertices.push_back(vertex);
        }

        // Обрабатываем индексы
        for(unsigned int i = 0; i < mesh->mNumFaces; i++) {
            aiFace face = mesh->mFaces[i];
            for(unsigned int j = 0; j < face.mNumIndices; j++) {
                meshData.indices.push_back(face.mIndices[j]);
            }
        }

        return meshData;
    }
    MaterialData AssimpLoader::ProcessMaterial(aiMaterial* material, const std::string& directory) {
        MaterialData materialData;

        aiString name;
        if (material->Get(AI_MATKEY_NAME, name) == AI_SUCCESS) {
            materialData.name = name.C_Str();
        }

        // Диффузный цвет
        aiColor3D diffuse(0.f, 0.f, 0.f);
        if (material->Get(AI_MATKEY_COLOR_DIFFUSE, diffuse) == AI_SUCCESS) {
            materialData.diffuseColor = glm::vec3(diffuse.r, diffuse.g, diffuse.b);
        }

        // Спекулярный цвет
        aiColor3D specular(0.f, 0.f, 0.f);
        if (material->Get(AI_MATKEY_COLOR_SPECULAR, specular) == AI_SUCCESS) {
            materialData.specularColor = glm::vec3(specular.r, specular.g, specular.b);
        }

        // Блеск
        float shininess = 32.0f;
        if (material->Get(AI_MATKEY_SHININESS, shininess) == AI_SUCCESS) {
            materialData.shininess = shininess;
        }

        auto diffuseMaps = LoadMaterialTextures(material, aiTextureType_DIFFUSE, "texture_diffuse", directory);
        materialData.textures.insert(materialData.textures.end(), diffuseMaps.begin(), diffuseMaps.end());

        auto specularMaps = LoadMaterialTextures(material, aiTextureType_SPECULAR, "texture_specular", directory);
        materialData.textures.insert(materialData.textures.end(), specularMaps.begin(), specularMaps.end());

        auto normalMaps = LoadMaterialTextures(material, aiTextureType_NORMALS, "texture_normal", directory);
        materialData.textures.insert(materialData.textures.end(), normalMaps.begin(), normalMaps.end());
        std::cout << "Material: " << materialData.name 
              << " textures: "
              << materialData.textures.size()
              << " diffuse: (" << materialData.diffuseColor.r 
              << ", "
              << materialData.diffuseColor.g
              << ", " 
              << materialData.diffuseColor.b << ")"
              << std::endl;
        return materialData;
    }

    std::vector<TextureData> AssimpLoader::LoadMaterialTextures(
        aiMaterial* mat, aiTextureType type, const std::string& typeName, const std::string& directory) {
        
        std::vector<TextureData> textures;
        
        for (unsigned int i = 0; i < mat->GetTextureCount(type); i++) {
            aiString str;
            mat->GetTexture(type, i, &str);

            TextureData textureData;
            textureData.type = typeName;
            textureData.path = directory + "/" + str.C_Str();

            textures.push_back(textureData);
        }

        return textures;
    }

    Agent PrefabSystem::CreateAgentFromPrefab(const AgentPrefab &prefab, const glm::vec3 &position)
    {
        Agent agent;
        agent.position = Float3::fromGlm(position);
            agent.fire_collider.bounds_min = Float3::fromGlm(prefab.bounds_min + position); 
    agent.fire_collider.bounds_max = Float3::fromGlm(prefab.bounds_max + position);
        
        agent.fire_collider.point_count = std::min(static_cast<int>(prefab.fire_points.size()), 64);

        for(int i = 0; i < agent.fire_collider.point_count; ++i) {
            Float3 local_pos = Float3::fromGlm(prefab.fire_points[i].position);
            agent.fire_collider.heat_points[i] = Float3(
                position.x + local_pos.x,
                position.y + local_pos.y, 
                position.z + local_pos.z
            );
            agent.fire_collider.fuels[i] = prefab.fire_points[i].initial_fuel;
            agent.fire_collider.intensities[i] = 0.0f;
        }

        agent.intencity = prefab.physics.intencity;
        agent.type = prefab.type;

        return agent;
    }

    void PrefabSystem::CreatePrefabFromAgent(const std::string &name, const Agent &agent, const std::string &model_path)
    {
        AgentPrefab prefab;
        prefab.name = name;
        prefab.type = agent.type;
        prefab.model_path = model_path;
        
        prefab.bounds_min = agent.fire_collider.bounds_min.toGlm();
        prefab.bounds_max = agent.fire_collider.bounds_max.toGlm();
        
        for (int i = 0; i < agent.fire_collider.point_count; i++) {
            AgentPrefab::FirePoint fp;
            fp.position = glm::vec3(
                agent.fire_collider.heat_points[i].x - agent.position.x,
                agent.fire_collider.heat_points[i].y - agent.position.y,
                agent.fire_collider.heat_points[i].z - agent.position.z
            );
            fp.initial_fuel = agent.fire_collider.fuels[i];
            prefab.fire_points.push_back(fp);
        }
        
        prefab.physics.intencity = agent.intencity;
        
        m_prefabs[name] = prefab;
    }

    void PrefabSystem::SerializePrefab(const AgentPrefab &prefab, YAML::Node &node) const
    {
        node["type"] = static_cast<int>(prefab.type);
        node["model_path"] = prefab.model_path;
        
        node["bounds_min"] = std::vector<float>{prefab.bounds_min.x, prefab.bounds_min.y, prefab.bounds_min.z};
        node["bounds_max"] = std::vector<float>{prefab.bounds_max.x, prefab.bounds_max.y, prefab.bounds_max.z};
        
        node["physics"]["intensity"] = prefab.physics.intencity;
        
        for (const auto& fp : prefab.fire_points) {
            YAML::Node fp_node;
            fp_node["position"] = std::vector<float>{fp.position.x, fp.position.y, fp.position.z};
            fp_node["initial_fuel"] = fp.initial_fuel;
            node["fire_points"].push_back(fp_node);
        }

        // Visuals
        node["visuals"]["normal_material"] = prefab.visuals.normal_material;
        node["visuals"]["burning_material"] = prefab.visuals.burning_material;
        node["visuals"]["burned_material"] = prefab.visuals.burned_material;
    }
    AgentPrefab PrefabSystem::DeserializePrefab(const YAML::Node &node) const
    {
        AgentPrefab prefab;
        
        // Основные поля
        prefab.type = static_cast<AgentType>(node["type"].as<int>());
        prefab.model_path = node["model_path"].as<std::string>();
        
        // Bounds
        auto bounds_min = node["bounds_min"].as<std::vector<float>>();
        auto bounds_max = node["bounds_max"].as<std::vector<float>>();
        prefab.bounds_min = glm::vec3(bounds_min[0], bounds_min[1], bounds_min[2]);
        prefab.bounds_max = glm::vec3(bounds_max[0], bounds_max[1], bounds_max[2]);
        
        // Physics
        prefab.physics.intencity = node["physics"]["intensity"].as<float>();
        
        // Fire points
        if (node["fire_points"]) {
            for (const auto& fp_node : node["fire_points"]) {
                AgentPrefab::FirePoint fp;
                auto pos = fp_node["position"].as<std::vector<float>>();
                fp.position = glm::vec3(pos[0], pos[1], pos[2]);
                fp.initial_fuel = fp_node["initial_fuel"].as<float>();
                prefab.fire_points.push_back(fp);
            }
        }

        // Visuals
        prefab.visuals.normal_material = node["visuals"]["normal_material"].as<std::string>();
        prefab.visuals.burning_material = node["visuals"]["burning_material"].as<std::string>();
        prefab.visuals.burned_material = node["visuals"]["burned_material"].as<std::string>();

        return prefab;
}

    void Model::SetupModel()
    {
        for (size_t i = 0; i < m_modelData.meshes.size(); i++) {
            const auto& meshData = m_modelData.meshes[i];
            
            const data::MaterialData* material = nullptr;
            if (meshData.materialIndex < m_modelData.materials.size()) {
                material = &m_modelData.materials[meshData.materialIndex];
            } else {
                static data::MaterialData defaultMaterial;
                material = &defaultMaterial;
            }
            
            data::MaterialData materialWithTextures = *material;
            for (auto& texInfo : materialWithTextures.textures) {
                bool found = false;
                for (const auto& cachedTex : m_texturesLoaded) {
                    if (cachedTex.path == texInfo.path) {
                        texInfo.texture = cachedTex.texture;
                        found = true;
                        break;
                    }
                }
                
                if (!found) {
                    texInfo.texture = Texture::Create(texInfo.path);
                    m_texturesLoaded.push_back(texInfo);
                }
            }
            
            m_meshes.emplace_back(meshData, materialWithTextures);
        }
    }

OpenGLShader::OpenGLShader(const std::string &filename) {
  std::ifstream stream(filename);
  std::string line;
  enum ShaderType { NONE = -1, VERTEX = 0, GEOMETRY = 1, FRAGMENT = 2 }; 
  std::stringstream ss[3];
  ShaderType type = ShaderType::NONE;
  
  while (getline(stream, line)) {
    if (line.find("#shader") != std::string::npos) {
      if (line.find("vertex") != std::string::npos) {
        type = ShaderType::VERTEX;
      } else if (line.find("geometry") != std::string::npos) { // ДОБАВИЛ
        type = ShaderType::GEOMETRY;
      } else if (line.find("fragment") != std::string::npos) {
        type = ShaderType::FRAGMENT;
      }
    } else {
      if (type != ShaderType::NONE) {
        ss[(int)type] << line << '\n';
      }
    }
  }

  m_rendererID = glCreateProgram();
  
  GLuint vertexShader = CompileShader(GL_VERTEX_SHADER, ss[0].str());
  glAttachShader(m_rendererID, vertexShader);
  
  if (!ss[1].str().empty()) {
    GLuint geometryShader = CompileShader(GL_GEOMETRY_SHADER, ss[1].str());
    glAttachShader(m_rendererID, geometryShader);
  }
  
  GLuint fragmentShader = CompileShader(GL_FRAGMENT_SHADER, ss[2].str());
  glAttachShader(m_rendererID, fragmentShader);
  
  glLinkProgram(m_rendererID);
  
  GLint isLinked;
  glGetProgramiv(m_rendererID, GL_LINK_STATUS, &isLinked);
  if (!isLinked) {
    GLint maxLength;
    glGetProgramiv(m_rendererID, GL_INFO_LOG_LENGTH, &maxLength);
    std::string infoLog(maxLength, ' ');
    glGetProgramInfoLog(m_rendererID, maxLength, &maxLength, &infoLog[0]);
    std::cout << "Shader linking failed: " + infoLog << '\n';
  }
}

GLuint OpenGLShader::CompileShader(GLenum type, const std::string &source) {
  GLuint shader = glCreateShader(type);
  const char *src = source.c_str();
  glShaderSource(shader, 1, &src, nullptr);
  glCompileShader(shader);
  GLint isCompiled;
  glGetShaderiv(shader, GL_COMPILE_STATUS, &isCompiled);
  if (!isCompiled) {
    GLint maxLength;
    glGetShaderiv(shader, GL_INFO_LOG_LENGTH, &maxLength);
    std::string infoLog(maxLength, ' ');
    glGetShaderInfoLog(shader, maxLength, &maxLength, &infoLog[0]);
    glDeleteShader(shader);
    std::cout << "Shader compilation failed: " + infoLog << '\n';
  }
  return shader;

OpenGLTexture::OpenGLTexture(const std::string &path) {
  int height, width, channels;
  stbi_set_flip_vertically_on_load(1);
  stbi_uc *data = stbi_load(path.c_str(), &width, &height, &channels, 0);
  if (data) {

    m_width = width;
    m_height = height;

    GLenum internalFormat = 0, dataFormat = 0;

    if (channels == 4) {
      internalFormat = GL_RGBA8;
      dataFormat = GL_RGBA;
    } else if (channels == 3) {
      internalFormat = GL_RGB8;
      dataFormat = GL_RGB;
    } else {
      stbi_image_free(data);
      throw std::runtime_error("Unsupported image format: " + path);
    }
    glGenTextures(1, &m_rendererID);
    glBindTexture(GL_TEXTURE_2D, m_rendererID);
    glTexImage2D(GL_TEXTURE_2D, 0, internalFormat, m_width, m_height, 0, dataFormat, GL_UNSIGNED_BYTE, data);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);

    stbi_image_free(data);
  } else {
    throw std::runtime_error("Failed to load texture: " + path);
  }
}

static GLenum ShaderDataTypeToOpenGLBaseType(ShaderDataType type) {
  switch (type) {
  case ShaderDataType::Float:
    return GL_FLOAT;
  case ShaderDataType::Float2:
    return GL_FLOAT;
  case ShaderDataType::Float3:
    return GL_FLOAT;
  case ShaderDataType::Float4:
    return GL_FLOAT;
  case ShaderDataType::Mat3:
    return GL_FLOAT;
  case ShaderDataType::Mat4:
    return GL_FLOAT;
  case ShaderDataType::Int:
    return GL_INT;
  case ShaderDataType::Int2:
    return GL_INT;
  case ShaderDataType::Int3:
    return GL_INT;
  case ShaderDataType::Int4:
    return GL_INT;
  case ShaderDataType::Bool:
    return GL_BOOL;
  }
  return 0;
}
OpenGLVertexArrayObject::OpenGLVertexArrayObject() {
  glGenVertexArrays(1, &m_id);
  glBindVertexArray(m_id);
}

OpenGLVertexArrayObject::~OpenGLVertexArrayObject() {
  glDeleteVertexArrays(1, &m_id);
}

void OpenGLVertexArrayObject::Bind() const { glBindVertexArray(m_id); }

void OpenGLVertexArrayObject::Unbind() const { glBindVertexArray(0); }

void OpenGLVertexArrayObject::AddVertexBuffer(
    const std::shared_ptr<VertexBuffer> &vertexBuffer) {
  glBindVertexArray(m_id);
  vertexBuffer->Bind();
  const auto &layout = vertexBuffer->GetLayout();

  for (const auto &elem : layout) {
    switch (elem.type) {
    case ShaderDataType::Float:
    case ShaderDataType::Float2:
    case ShaderDataType::Float3:
    case ShaderDataType::Float4: {
      glEnableVertexAttribArray(m_VertexBufferIndex);
      glVertexAttribPointer(m_VertexBufferIndex, elem.GetComponentCount(),
                            ShaderDataTypeToOpenGLBaseType(elem.type),
                            elem.normalized ? GL_TRUE : GL_FALSE,
                            layout.GetStride(),
                            reinterpret_cast<void *>(elem.offset));
      m_VertexBufferIndex++;
      break;
    }

    case ShaderDataType::Int:
    case ShaderDataType::Int2:
    case ShaderDataType::Int3:
    case ShaderDataType::Int4:
    case ShaderDataType::Bool: {
      glEnableVertexAttribArray(m_VertexBufferIndex);

      glVertexAttribIPointer(m_VertexBufferIndex, elem.GetComponentCount(),
                             ShaderDataTypeToOpenGLBaseType(elem.type),
                             layout.GetStride(),
                             reinterpret_cast<void *>(elem.offset));
      m_VertexBufferIndex++;
      break;
    }

    case ShaderDataType::Mat3:
    case ShaderDataType::Mat4: {
      uint8_t count = elem.GetComponentCount();
      for (uint8_t i = 0; i < count; ++i) {
        glEnableVertexAttribArray(m_VertexBufferIndex);
        glVertexAttribPointer(
            m_VertexBufferIndex, count,
            ShaderDataTypeToOpenGLBaseType(elem.type),
            elem.normalized ? GL_TRUE : GL_FALSE, layout.GetStride(),
            reinterpret_cast<void *>(elem.offset + sizeof(float) * count * i));
        glVertexAttribDivisor(m_VertexBufferIndex, 1);
        m_VertexBufferIndex++;
      }
    }
    default:
      std::cout << "Unknown ShaderDataType!" << std::endl;
    }
  }
  m_vertexBuffers.push_back(vertexBuffer);
}

void OpenGLVertexArrayObject::SetIndexBuffer(
    const std::shared_ptr<IndexBuffer> &indexBuffer) {
  glBindVertexArray(m_id);
  indexBuffer->Bind();
  m_indexBuffer = indexBuffer;
}

void Renderer::UploadAgentsData(const std::vector<Agent> &agents)
{
    m_agentRenderData.clear();
    
    for (const auto& agent : agents) {
        if (!agent.isBurning()) continue; 
        
        AgentRenderData data;
        data.position = glm::vec3(agent.position.x, agent.position.y, agent.position.z);
        data.intensity = agent.getOverallIntensity(); 
        
        data.bounds_min = glm::vec3(
            agent.fire_collider.bounds_min.x,
            agent.fire_collider.bounds_min.y,
            agent.fire_collider.bounds_min.z
        );
        data.bounds_max = glm::vec3(
            agent.fire_collider.bounds_max.x, 
            agent.fire_collider.bounds_max.y,
            agent.fire_collider.bounds_max.z
        );
        
        m_agentRenderData.push_back(data);
    }
    if(!m_agentRenderData.empty()) {
        m_agentVBO->Bind();
        m_agentVBO->SetData(m_agentRenderData.data(), m_agentRenderData.size() * sizeof(AgentRenderData));
    }
}

    m_fireGeometryShader->Bind();
    m_fireGeometryShader->SetMat4("u_view", m_viewMat);
    m_fireGeometryShader->SetMat4("u_projection", m_projectionMat);
    m_fireGeometryShader->SetFloat("u_time", glfwGetTime());
    
    glm::mat4 view = m_viewMat;
    glm::vec3 camera_right = glm::vec3(view[0][0], view[1][0], view[2][0]);
    glm::vec3 camera_up = glm::vec3(view[0][1], view[1][1], view[2][1]);
    
    m_fireGeometryShader->SetVec3("u_camera_right", camera_right);
    m_fireGeometryShader->SetVec3("u_camera_up", camera_up);
    m_fireTexture->Bind(0);
    m_fireGeometryShader->SetInt("u_fire_texture", 0);
    m_agentVAO->Bind();
    glDrawArrays(GL_POINTS, 0, m_agentRenderData.size());
    m_agentVAO->Unbind();
    
    m_fireGeometryShader->Unbind();
    
    glDepthMask(GL_TRUE);
    glDisable(GL_BLEND);
}

#pragma once

namespace engine {
enum class ShaderDataType {
  Float,
  Float2,
  Float3,
  Float4,
  Int,
  Int2,
  Int3,
  Int4,
  Mat3,
  Mat4,
  Bool
};
static uint32_t ShaderDataTypeSize(ShaderDataType type) {
  switch (type) {
  case ShaderDataType::Float:
    return 4;
  case ShaderDataType::Float2:
    return 4 * 2;
  case ShaderDataType::Float3:
    return 4 * 3;
  case ShaderDataType::Float4:
    return 4 * 4;
  case ShaderDataType::Mat3:
    return 4 * 3 * 3;
  case ShaderDataType::Mat4:
    return 4 * 4 * 4;
  case ShaderDataType::Int:
    return 4;
  case ShaderDataType::Int2:
    return 4 * 2;
  case ShaderDataType::Int3:
    return 4 * 3;
  case ShaderDataType::Int4:
    return 4 * 4;
  case ShaderDataType::Bool:
    return 1;
  }
  return 0;
}
struct BufferElement {
  ShaderDataType type;
  std::string name;
  uint32_t size;
  uint32_t offset;
  bool normalized;
  BufferElement(ShaderDataType type, const std::string &name,
                bool normalized = false)
      : name(name), type(type), size(ShaderDataTypeSize(type)), offset(0),
        normalized(normalized) {}
  uint32_t GetComponentCount() const {
    switch (type) {
    case ShaderDataType::Float:
      return 1;
    case ShaderDataType::Float2:
      return 2;
    case ShaderDataType::Float3:
      return 3;
    case ShaderDataType::Float4:
      return 4;
    case ShaderDataType::Int:
      return 1;
    case ShaderDataType::Int2:
      return 2;
    case ShaderDataType::Int3:
      return 3;
    case ShaderDataType::Int4:
      return 4;
    case ShaderDataType::Mat3:
      return 3;
    case ShaderDataType::Mat4:
      return 4;
    case ShaderDataType::Bool:
      return 1;
    }
    return 0;
  }
};

class BufferLayout {
private:
  std::vector<BufferElement> m_elements;
  uint32_t m_Stride = 0;

public:
  BufferLayout() = default;
  BufferLayout(const std::initializer_list<BufferElement> &elements)
      : m_elements(elements) {
    CalculateOffsetsAndStride();
  }
  const std::vector<BufferElement> &GetElements() const { return m_elements; }
  uint32_t GetStride() const { return m_Stride; }
  // Итераторы для for-each
  std::vector<BufferElement>::iterator begin() { return m_elements.begin(); }
  std::vector<BufferElement>::iterator end() { return m_elements.end(); }
  std::vector<BufferElement>::const_iterator begin() const {
    return m_elements.begin();
  }
  std::vector<BufferElement>::const_iterator end() const {
    return m_elements.end();
  }

private:
  void CalculateOffsetsAndStride() {
    size_t offset = 0;
    m_Stride = 0;
    for (auto &element : m_elements) {
      element.offset = offset;
      offset += element.size;
      m_Stride += element.size;
    }
  }
};

===== ./math/ScreenRay.h =====
#pragma once
#include <cmath>
#include <glm/glm.hpp>
#include <glm/gtc/matrix_inverse.hpp>

namespace math {
inline void ScreenPointToRay(float px, float py, float viewportW,
                             float viewportH, const glm::mat4 &proj,
                             const glm::mat4 &view, glm::vec3 &outOrigin,
                             glm::vec3 &outDir) {
  float x = (2.0F * px / viewportW) - 1.0F;
  float y = 1.0F - (2.0F * py / viewportH);
  glm::vec4 ray_clip(x, y, -1.0F, 1.0F);
  glm::mat4 inv_proj = glm::inverse(proj);
  glm::vec4 ray_eye = inv_proj * ray_clip;
  ray_eye = glm::vec4(ray_eye.x, ray_eye.y, -1.0f, 0.0f);
  glm::mat4 inv_view = glm::inverse(view);
  glm::vec4 ray_world4 = inv_view * ray_eye;
  glm::vec3 ray_world = glm::normalize(glm::vec3(ray_world4));
  outOrigin = glm::vec3(inv_view[3]);
  outDir = ray_world;
}

inline bool IntersectRayPlaneY0(const glm::vec3 &ro, const glm::vec3 &rd,
                                glm::vec3 &hit) {
  if (std::fabs(rd.y) < 1e-6F) {
    return false;
}
  float t = -ro.y / rd.y;
  if (t < 0.0F) {
    return false;
}
  hit = ro + rd * t;
  return true;
}
} // namespace math

// ui/Editor.h
#pragma once
#include <memory>
#include <ranges>
#include <type_traits>
#include <vector>

#include "ui/IModule.h"
#include "ui/IPanel.h"
#include "ui/ITool.h"
// не инклюдим IModule тут — достаточно fwd-decl
namespace ui {
struct IModule;
} // namespace ui

namespace ui {

class Editor {
public:
  Editor() = default;

  // Запрет копирования
  Editor(const Editor &) = delete;
  Editor &operator=(const Editor &) = delete;

  // Перемещение: переносим владение, но сбрасываем стек (переактивация позже)
  Editor(Editor &&other) noexcept
      : m_panels(std::move(other.m_panels)), m_tools(std::move(other.m_tools)),
        m_modules(std::move(other.m_modules)) {
    m_activeStack.clear();
  }

  Editor &operator=(Editor &&other) noexcept {
    if (this != &other) {
      ShutdownModules(); // на всякий случай корректно выгружаем свои
      m_panels = std::move(other.m_panels);
      m_tools = std::move(other.m_tools);
      m_modules = std::move(other.m_modules);
      m_activeStack.clear();
    }
    return *this;
  }

  ~Editor() {
    // Гарантируем Shutdown модулей перед разрушением редактора
    for (auto &m_module : std::ranges::reverse_view(m_modules)) {
      m_module->Shutdown();
    }
  }

  // --- Регистрация панелей/инструментов (как было) ---
  void RegisterPanel(std::unique_ptr<IPanel> panel) {
    m_panels.emplace_back(std::move(panel));
  }
  void RegisterTool(std::unique_ptr<ITool> tool) {
    m_tools.emplace_back(std::move(tool));
  }

  template <class T> T *GetTool() {
    static_assert(std::is_base_of_v<ITool, T>, "T must be ITool");
    for (auto &up : m_tools) {
      if (auto *tool_pointer = dynamic_cast<T *>(up.get())) {
        return tool_pointer;
      }
    }
    return nullptr;
  }
  template <class T> const T *GetTool() const {
    static_assert(std::is_base_of_v<ITool, T>, "T must be ITool");
    for (const auto &up : m_tools) {
      if (auto *tool_pointer = dynamic_cast<const T *>(up.get())) {
        return tool_pointer;
      }
    }
    return nullptr;
  }

  // --- Модули ---
  void RegisterModule(std::unique_ptr<IModule> module) {
    m_modules.emplace_back(std::move(module));
    m_modules.back()->Startup(*this);
  }

  // При желании — явная остановка модулей до деструктора
  void ShutdownModules() {
    for (auto &m_module : std::ranges::reverse_view(m_modules)) {
      m_module->Shutdown();
    }
  }

  // --- Активные инструменты (как уже сделали) ---
  void SetActiveTool(ToolType type, EditorContext &ctx) {
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnDeactivate(ctx);
      m_activeStack.clear();
    }
    if (auto *tool = FindTool(type)) {
      m_activeStack.push_back(tool);
      tool->OnActivate(ctx);
    }
  }
  void PushTool(ToolType type, EditorContext &ctx) {
    if (auto *tool = FindTool(type)) {
      if (!m_activeStack.empty()) {
        m_activeStack.back()->OnDeactivate(ctx);
      }
      m_activeStack.push_back(tool);
      tool->OnActivate(ctx);
    }
  }
  void PopTool(EditorContext &ctx) {
    if (m_activeStack.empty()) {
      return;
    }
    m_activeStack.back()->OnDeactivate(ctx);
    m_activeStack.pop_back();
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnActivate(ctx);
    }
  }

  void DrawUI(EditorContext &ctx) {
    ctx.m_editor = this;
    for (auto &panel : m_panels) {
      panel->Draw(ctx);
    }
  }
  void TickTools(EditorContext &ctx) {
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnUpdate(ctx);
    }
  }

  [[nodiscard]] ToolType ActiveToolType() const {
    return m_activeStack.empty() ? ToolType::OTHER_TOOL
                                 : m_activeStack.back()->Type();
  }

private:
  ITool *FindTool(ToolType type) {
    for (auto &tool : m_tools) {
      if (tool->Type() == type) {
        return tool.get();
      }
    }
    return nullptr;
  }

  std::vector<std::unique_ptr<IPanel>> m_panels;
  std::vector<std::unique_ptr<ITool>> m_tools;
  std::vector<ITool *> m_activeStack;
  std::vector<std::unique_ptr<IModule>> m_modules; // <-- NEW
};

} // namespace ui

// ui/panels/ToolbarPanel.cpp
#include "ui/panels/ToolbarPanel.h"
#include "core/scene/Scene.h"
#include "ui/Editor.h"
#include "core/Input.h"          // ← у тебя именно так называется инпут
#include "math/ScreenRay.h"      // луч из экрана

namespace ui {

void ToolbarPanel::Draw(EditorContext& ctx) {
  bool visible = true;
  if (!ImGui::Begin(Id(), &visible)) {
    ImGui::End();
    if (m_wasVisible) { OnHide(ctx); }
    return;
  }
  if (!m_wasVisible) { OnShow(ctx); }

  // Текущий активный инструмент
  const auto ACTIVE = ctx.GetActiveToolType();

  // Переключение инструментов
  const bool IS_PLACE = (ACTIVE == ToolType::PLACE_ON_GROUND);
  if (ImGui::Selectable("Place on ground", IS_PLACE)) {
    ctx.m_editor->SetActiveTool(ToolType::PLACE_ON_GROUND, ctx);
  }

  ImGui::Separator();
  ImGui::SliderFloat("Fire intensity", &m_fireIntensity, 0.0F, 10.0F);
  // // действие — связать с контекстом/сценой
  ctx.m_scene->SetFireIntensity(m_fireIntensity);
  ImGui::Separator();
  if (ImGui::Button("Ignite at cursor")) {
    // курсор в пикселях из GLFW
    double mx = NAN, my = NAN;
    glfwGetCursorPos(ctx.m_window, &mx, &my);

    // луч из экрана в мир (твоя утилита)
    glm::vec3 ro, rd;
    math::ScreenPointToRay(
      static_cast<float>(mx), static_cast<float>(my),
      static_cast<float>(ctx.m_viewportW), static_cast<float>(ctx.m_viewportH),
      ctx.m_proj, ctx.m_view, ro, rd);

    // пересечение с плоскостью Y=0 (твоя утилита)
    glm::vec3 hit;
    if (math::IntersectRayPlaneY0(ro, rd, hit)) {
      ctx.m_scene->AddFire(hit.x, hit.z, m_fireIntensity);
    }
  }

  ImGui::End();
}

} // namespace ui
