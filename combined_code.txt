#pragma once
#include <glm/glm.hpp>
struct WindConfig {
    glm::vec3 direction = glm::vec3(1.0f, 0.0f, 0.0f); // Направление ветра
    float strength = 10.0f;                              // Сила ветра (0-1)
    float gustiness = 0.3f;                             // Порывистость
    float change_speed = 0.1f;                          // Скорость изменения
    
    // Метод для получения текущего ветра с шумом
    glm::vec3 getCurrentWind(float time) const {
        glm::vec3 base_wind = glm::normalize(direction) * strength;
        
        float noise_x = sin(time * change_speed * 2.0f) * gustiness;
        float noise_z = cos(time * change_speed * 1.7f) * gustiness;
        
        return base_wind + glm::vec3(noise_x, 0.0f, noise_z);
    }
};

#pragma once
#include <cstdint>
#include <glad/glad.h>
#include <memory>
#include <string>
#include <vector>
namespace engine {
enum class ShaderDataType {
  Float,
  Float2,
  Float3,
  Float4,
  Int,
  Int2,
  Int3,
  Int4,
  Mat3,
  Mat4,
  Bool
};
static uint32_t ShaderDataTypeSize(ShaderDataType type) {
  switch (type) {
  case ShaderDataType::Float:
    return 4;
  case ShaderDataType::Float2:
    return 4 * 2;
  case ShaderDataType::Float3:
    return 4 * 3;
  case ShaderDataType::Float4:
    return 4 * 4;
  case ShaderDataType::Mat3:
    return 4 * 3 * 3;
  case ShaderDataType::Mat4:
    return 4 * 4 * 4;
  case ShaderDataType::Int:
    return 4;
  case ShaderDataType::Int2:
    return 4 * 2;
  case ShaderDataType::Int3:
    return 4 * 3;
  case ShaderDataType::Int4:
    return 4 * 4;
  case ShaderDataType::Bool:
    return 1;
  }
  return 0;
}
struct BufferElement {
  ShaderDataType type;
  std::string name;
  uint32_t size;
  uint32_t offset;
  bool normalized;
  BufferElement(ShaderDataType type, const std::string &name,
                bool normalized = false)
      : name(name), type(type), size(ShaderDataTypeSize(type)), offset(0),
        normalized(normalized) {}
  uint32_t GetComponentCount() const {
    switch (type) {
    case ShaderDataType::Float:
      return 1;
    case ShaderDataType::Float2:
      return 2;
    case ShaderDataType::Float3:
      return 3;
    case ShaderDataType::Float4:
      return 4;
    case ShaderDataType::Int:
      return 1;
    case ShaderDataType::Int2:
      return 2;
    case ShaderDataType::Int3:
      return 3;
    case ShaderDataType::Int4:
      return 4;
    case ShaderDataType::Mat3:
      return 3;
    case ShaderDataType::Mat4:
      return 4;
    case ShaderDataType::Bool:
      return 1;
    }
    return 0;
  }
};

class BufferLayout {
private:
  std::vector<BufferElement> m_elements;
  uint32_t m_Stride = 0;

public:
  BufferLayout() = default;
  BufferLayout(const std::initializer_list<BufferElement> &elements)
      : m_elements(elements) {
    CalculateOffsetsAndStride();
  }
  const std::vector<BufferElement> &GetElements() const { return m_elements; }
  uint32_t GetStride() const { return m_Stride; }
  // Итераторы для for-each
  std::vector<BufferElement>::iterator begin() { return m_elements.begin(); }
  std::vector<BufferElement>::iterator end() { return m_elements.end(); }
  std::vector<BufferElement>::const_iterator begin() const {
    return m_elements.begin();
  }
  std::vector<BufferElement>::const_iterator end() const {
    return m_elements.end();
  }

private:
  void CalculateOffsetsAndStride() {
    size_t offset = 0;
    m_Stride = 0;
    for (auto &element : m_elements) {
      element.offset = offset;
      offset += element.size;
      m_Stride += element.size;
    }
  }
};

class VertexBuffer {
private:
  GLuint m_id;
public:
  virtual ~VertexBuffer() = default;

  virtual void Bind() const = 0;
  virtual void Unbind() const = 0;
  virtual void SetData(const void *data, uint32_t size) = 0;
  virtual void SetLayout(const BufferLayout &layout) = 0;
  virtual size_t GetSize() const = 0;
  virtual const BufferLayout &GetLayout() const = 0;
  static std::shared_ptr<VertexBuffer> Create(int32_t size);
  static std::shared_ptr<VertexBuffer> Create(float *vertices, int32_t size);
};

} // namespace engine

void Renderer::Init() {
  m_shader = Shader::Create("res/shaders/shader.sh");
  m_modelShader = Shader::Create("res/shaders/model_shader.sh");
  m_fireGeometryShader = Shader::Create("res/shaders/fire_geom.sh");
  m_instanceModelShader =
      Shader::Create("res/shaders/instance_model_shader.sh");
  m_terrainShader = Shader::Create("res/shaders/terrain.sh");

  m_vao = VertexArrayObject::Create();

  m_vbo = VertexBuffer::Create(vertices, sizeof(vertices));
  m_vbo->SetLayout({{ShaderDataType::Float3, "a_position"}});
  m_vao->AddVertexBuffer(m_vbo);

  m_ebo = IndexBuffer::Create(indices, sizeof(indices) / sizeof(unsigned int));
  m_vao->SetIndexBuffer(m_ebo);

  m_agentVAO = VertexArrayObject::Create();
  m_agentVBO = VertexBuffer::Create(50000 * sizeof(AgentRenderData));
  m_agentVBO->SetLayout({{ShaderDataType::Float3, "a_position"},
                         {ShaderDataType::Float, "a_intensity"},
                         {ShaderDataType::Float3, "a_bounds_min"},
                         {ShaderDataType::Float3, "a_bounds_max"}});
  m_agentVAO->AddVertexBuffer(m_agentVBO);
  m_instanceBuffer = VertexBuffer::Create(10000 * sizeof(glm::mat4));
  m_instanceBuffer->SetLayout(
      {{ShaderDataType::Float4, "a_instance_matrix0", 1},
       {ShaderDataType::Float4, "a_instance_matrix1", 1},
       {ShaderDataType::Float4, "a_instance_matrix2", 1},
       {ShaderDataType::Float4, "a_instance_matrix3", 1}});

  // Создаем отдельный VAO для инстансинга
  m_instanceVAO = VertexArrayObject::Create();

  m_universalParticleShader = Shader::Create("res/shaders/fire_geom.sh");
  
  m_particleVAO = VertexArrayObject::Create();
  m_particleVBO = VertexBuffer::Create(100000 * sizeof(UniversalParticleData));
  m_particleVBO->SetLayout({
      {ShaderDataType::Float3, "a_position"},
      {ShaderDataType::Float, "a_intensity"},
      {ShaderDataType::Float3, "a_bounds_min"},
      {ShaderDataType::Float3, "a_bounds_max"},
      {ShaderDataType::Float, "a_effect_type"}
  });
  m_particleVAO->AddVertexBuffer(m_particleVBO);
  SetFireParams({});
}

void Renderer::DrawCube(const glm::vec3 &position, const glm::vec3 &color,
                        const glm::vec3 &size) {
  glm::mat4 model = glm::mat4(1.0f);
  model = glm::translate(model, position);
  model = glm::scale(model, size);
  m_shader->Bind();

  m_shader->SetMat4("u_model", model);
  m_shader->SetVec3("u_color", color);
  m_shader->SetMat4("u_view", m_viewMat);
  m_shader->SetMat4("u_projection", m_projectionMat);
  m_vao->Bind();
  glDrawElements(GL_TRIANGLES, 36, GL_UNSIGNED_INT, 0);
  m_vao->Unbind();
  m_shader->Unbind();
}

void Renderer::DrawModel(Model &model, const glm::vec3 &position,
                         const glm::vec3 &scale, const glm::vec3 &camPos) {
  glm::mat4 modelmat = glm::mat4(1.0f);
  modelmat = glm::translate(modelmat, position);
  modelmat = glm::scale(modelmat, scale);
  m_modelShader->Bind();
  m_modelShader->SetMat4("u_model", modelmat);
  m_modelShader->SetMat4("u_view", m_viewMat);
  m_modelShader->SetMat4("u_projection", m_projectionMat);
  m_modelShader->SetVec3("viewPos", camPos);
  m_modelShader->SetVec3("light.position", glm::vec3(500.0f, 200.0f, 100.0f));
  m_modelShader->SetVec3("light.ambient", glm::vec3(0.2f));
  m_modelShader->SetVec3("light.diffuse", glm::vec3(0.8f));
  m_modelShader->SetVec3("light.specular", glm::vec3(1.0f));
  model.Draw(*m_modelShader);
}

void Renderer::DrawModelWithMaterial(
    Model &model, const glm::vec3 &position, const glm::vec3 &scale,
    const glm::vec3 &camPos, const std::shared_ptr<Material> &material) {

  // 1. Вычисляем матрицу модели
  glm::mat4 modelmat = glm::mat4(1.0f);
  modelmat = glm::translate(modelmat, position);
  modelmat = glm::scale(modelmat, scale);

  // 2. Биндим шейдер и устанавливаем uniform'ы
  m_modelShader->Bind();

  // ДОБАВЬ ПРОВЕРКУ:
  m_modelShader->SetMat4("u_model", modelmat);
  m_modelShader->SetMat4("u_view", m_viewMat);
  m_modelShader->SetMat4("u_projection", m_projectionMat);
  m_modelShader->SetVec3("viewPos", camPos);

  // 3. Настройка освещения (как у тебя было)
  m_modelShader->SetVec3("light.position", glm::vec3(500.0f, 200.0f, 100.0f));
  m_modelShader->SetVec3("light.ambient", glm::vec3(0.2f));
  m_modelShader->SetVec3("light.diffuse", glm::vec3(0.8f));
  m_modelShader->SetVec3("light.specular", glm::vec3(1.0f));

  // 4. Биндим материал (если он есть)
  if (material) {
    material->Bind(*m_modelShader);
  } else {
    m_modelShader->SetBool("material.use_diffuse_map", false);
    m_modelShader->SetBool("material.use_specular_map", false);
    m_modelShader->SetVec3("material.diffuse", glm::vec3(0.8f));
    m_modelShader->SetVec3("material.specular", glm::vec3(0.5f));
    m_modelShader->SetFloat("material.shininess", 32.0f);
  }

  model.DrawGeometryOnly(*m_modelShader);
}

void Renderer::BeginScene(const glm::mat4 &view, const glm::mat4 &projection) {
  glClearColor(0.1, 0.1, 0.1, 1.0f);

  glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
  glEnable(GL_DEPTH_TEST);
  glEnable(GL_CULL_FACE);
  glCullFace(GL_BACK);
  glFrontFace(GL_CCW);
  m_viewMat = view;
  m_projectionMat = projection;
}
void Renderer::UploadAgentsData(const std::vector<Agent> &agents) {
  m_particleData.clear();
    
    for (const auto &agent : agents) {
        if (!agent.isBurning()) continue;
        
        float intensity = agent.getOverallIntensity();
        
        // ОГОНЬ (всегда)
        if (m_fireParams.fire.enabled) {
            m_particleData.push_back({
                glm::vec3(agent.position.x, agent.position.y, agent.position.z),
                intensity,
                glm::vec3(agent.fire_collider.bounds_min.x, agent.fire_collider.bounds_min.y, agent.fire_collider.bounds_min.z),
                glm::vec3(agent.fire_collider.bounds_max.x, agent.fire_collider.bounds_max.y, agent.fire_collider.bounds_max.z),
                0.0f // fire
            });
        }
            if (m_fireParams.smoke.enabled && intensity > 0.3f) {
            m_particleData.push_back({
                glm::vec3(agent.position.x, agent.position.y, agent.position.z),
                intensity * 0.7f,
                glm::vec3(agent.fire_collider.bounds_min.x, agent.fire_collider.bounds_min.y, agent.fire_collider.bounds_min.z),
                glm::vec3(agent.fire_collider.bounds_max.x, agent.fire_collider.bounds_max.y, agent.fire_collider.bounds_max.z),
                1.0f // smoke
            });
        }
        
        // ИСКРЫ (при высокой интенсивности)
        if (m_fireParams.sparks.enabled && intensity > 0.7f) {
            m_particleData.push_back({
                glm::vec3(agent.position.x, agent.position.y, agent.position.z),
                intensity * 0.5f,
                glm::vec3(agent.fire_collider.bounds_min.x, agent.fire_collider.bounds_min.y, agent.fire_collider.bounds_min.z),
                glm::vec3(agent.fire_collider.bounds_max.x, agent.fire_collider.bounds_max.y, agent.fire_collider.bounds_max.z),
                2.0f // sparks
            });
        }
          if (m_fireParams.ash.enabled) {
            m_particleData.push_back({
                glm::vec3(agent.position.x, agent.position.y, agent.position.z),
                intensity * 0.3f,
                glm::vec3(agent.fire_collider.bounds_min.x, agent.fire_collider.bounds_min.y, agent.fire_collider.bounds_min.z),
                glm::vec3(agent.fire_collider.bounds_max.x, agent.fire_collider.bounds_max.y, agent.fire_collider.bounds_max.z),
                3.0f // ash
            });
        }
        
        // ТЕПЛО (при любой интенсивности)
        if (m_fireParams.heat.enabled) {
            m_particleData.push_back({
                glm::vec3(agent.position.x, agent.position.y, agent.position.z),
                intensity * 0.2f,
                glm::vec3(agent.fire_collider.bounds_min.x, agent.fire_collider.bounds_min.y, agent.fire_collider.bounds_min.z),
                glm::vec3(agent.fire_collider.bounds_max.x, agent.fire_collider.bounds_max.y, agent.fire_collider.bounds_max.z),
                4.0f // heat
            });
        }
    }
    
    // ОДИН раз загружаем все данные
    if (!m_particleData.empty()) {
        m_particleVBO->Bind();
        m_particleVBO->SetData(m_particleData.data(), m_particleData.size() * sizeof(UniversalParticleData));
    }
}

void Renderer::EndScene() {

  glUseProgram(0);
}

void Renderer::RenderWithGeometryShader(const glm::vec3 &wind_force) {
   if (m_particleData.empty()) return;
    
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE);  // ← ДОБАВЬ ЭТУ СТРОКУ!

    glDepthMask(GL_FALSE);

    m_universalParticleShader->Bind();
    
    // Общие uniform'ы
    m_universalParticleShader->SetMat4("u_view", m_viewMat);
    m_universalParticleShader->SetMat4("u_projection", m_projectionMat);
    m_universalParticleShader->SetFloat("u_time", glfwGetTime());
    m_universalParticleShader->SetVec3("u_wind_force", wind_force);
    
    // Камерные векторы
    glm::mat4 view = m_viewMat;
    glm::vec3 camera_right = glm::vec3(view[0][0], view[1][0], view[2][0]);
    glm::vec3 camera_up = glm::vec3(view[0][1], view[1][1], view[2][1]);
    m_universalParticleShader->SetVec3("u_camera_right", camera_right);
    m_universalParticleShader->SetVec3("u_camera_up", camera_up);
    
    // БИНДИМ ТЕКСТУРЫ (КОТОРЫЕ УЖЕ ЗАГРУЖЕНЫ В SetFireParams)
    m_fireTexture->Bind(0);
    m_smokeTexture->Bind(1);
    m_sparksTexture->Bind(2);
    m_ashTexture->Bind(3);
    m_heatTexture->Bind(4);
    
    m_universalParticleShader->SetInt("u_fire_texture", 0);
    m_universalParticleShader->SetInt("u_smoke_texture", 1);
    m_universalParticleShader->SetInt("u_sparks_texture", 2);
    m_universalParticleShader->SetInt("u_ash_texture", 3);
    m_universalParticleShader->SetInt("u_heat_texture", 4);
    
    // ОДИН вызов отрисовки для всех эффектов
    m_particleVAO->Bind();
    glDrawArrays(GL_POINTS, 0, m_particleData.size());
    m_particleVAO->Unbind();

    m_universalParticleShader->Unbind();

    glDepthMask(GL_TRUE);
    glDisable(GL_BLEND);
}

void Renderer::DrawTerrain(std::shared_ptr<VertexArrayObject> vao,
                           const glm::vec3 &terrainPos) {
  if (!vao || !m_terrainShader)
    return;

  auto ib = vao->GetIndexBuffer();
  uint32_t count = ib ? ib->GetCount() : 0;
  glm::mat4 model = glm::translate(glm::mat4(1.0f), terrainPos);
  m_terrainShader->Bind();
  m_terrainShader->SetMat4("u_View", m_viewMat);
  m_terrainShader->SetMat4("u_Projection", m_projectionMat);
  m_terrainShader->SetMat4("u_Model", model);

  m_terrainShader->SetVec3("u_LightPos", glm::vec3(100.0f, 100.0f, 100.0f));
  m_terrainShader->SetFloat("u_MaxHeight", 80.0f);

  vao->Bind();
  glDrawElements(GL_TRIANGLES, count, GL_UNSIGNED_INT, nullptr);
}

void Renderer::SetFireParams(const FireParams &params) {
    m_fireParams = params;

    // ЗАГРУЖАЕМ ВСЕ ТЕКСТУРЫ ТОЛЬКО ЗДЕСЬ
    try {
        m_fireTexture = Texture::Create(params.fireTexturePath);
    } catch (const std::exception &e) {
        std::cout << "Failed to load fire texture: " << e.what() << std::endl;
        m_fireTexture = Texture::Create("res/textures/fire_texture3.png");
    }
    
    try {
        m_smokeTexture = Texture::Create(params.smokeTexturePath);
    } catch (const std::exception &e) {
        std::cout << "Failed to load smoke texture: " << e.what() << std::endl;
        m_smokeTexture = Texture::Create("res/textures/smoke_texture.png");
    }
    
    try {
        m_sparksTexture = Texture::Create(params.sparksTexturePath);
    } catch (const std::exception &e) {
        std::cout << "Failed to load sparks texture: " << e.what() << std::endl;

    // Устанавливаем параметры в шейдер
    if (m_universalParticleShader) {

static GLenum ShaderDataTypeToOpenGLBaseType(ShaderDataType type) {
  switch (type) {
  case ShaderDataType::Float:
    return GL_FLOAT;
  case ShaderDataType::Float2:
    return GL_FLOAT;
  case ShaderDataType::Float3:
    return GL_FLOAT;
  case ShaderDataType::Float4:
    return GL_FLOAT;
  case ShaderDataType::Mat3:
    return GL_FLOAT;
  case ShaderDataType::Mat4:
    return GL_FLOAT;
  case ShaderDataType::Int:
    return GL_INT;
  case ShaderDataType::Int2:
    return GL_INT;
  case ShaderDataType::Int3:
    return GL_INT;
  case ShaderDataType::Int4:
    return GL_INT;
  case ShaderDataType::Bool:
    return GL_BOOL;
  }
  return 0;
}
OpenGLVertexArrayObject::OpenGLVertexArrayObject() {
  glGenVertexArrays(1, &m_id);
  glBindVertexArray(m_id);
}

OpenGLVertexArrayObject::~OpenGLVertexArrayObject() {
  glDeleteVertexArrays(1, &m_id);
}

void OpenGLVertexArrayObject::Bind() const { glBindVertexArray(m_id); }

void OpenGLVertexArrayObject::Unbind() const { glBindVertexArray(0); }

void OpenGLVertexArrayObject::AddVertexBuffer(
    const std::shared_ptr<VertexBuffer> &vertexBuffer) {
  glBindVertexArray(m_id);
  vertexBuffer->Bind();
  const auto &layout = vertexBuffer->GetLayout();

  for (const auto &elem : layout) {
    switch (elem.type) {
    case ShaderDataType::Float:
    case ShaderDataType::Float2:
    case ShaderDataType::Float3:
    case ShaderDataType::Float4: {
      glEnableVertexAttribArray(m_VertexBufferIndex);
      glVertexAttribPointer(m_VertexBufferIndex, elem.GetComponentCount(),
                            ShaderDataTypeToOpenGLBaseType(elem.type),
                            elem.normalized ? GL_TRUE : GL_FALSE,
                            layout.GetStride(),
                            reinterpret_cast<void *>(elem.offset));
      m_VertexBufferIndex++;
      break;
    }

    case ShaderDataType::Int:
    case ShaderDataType::Int2:
    case ShaderDataType::Int3:
    case ShaderDataType::Int4:
    case ShaderDataType::Bool: {
      glEnableVertexAttribArray(m_VertexBufferIndex);

      glVertexAttribIPointer(m_VertexBufferIndex, elem.GetComponentCount(),
                             ShaderDataTypeToOpenGLBaseType(elem.type),
                             layout.GetStride(),
                             reinterpret_cast<void *>(elem.offset));
      m_VertexBufferIndex++;
      break;
    }

    case ShaderDataType::Mat3:
    case ShaderDataType::Mat4: {
      uint8_t count = elem.GetComponentCount();
      for (uint8_t i = 0; i < count; ++i) {
        glEnableVertexAttribArray(m_VertexBufferIndex);
        glVertexAttribPointer(
            m_VertexBufferIndex, count,
            ShaderDataTypeToOpenGLBaseType(elem.type),
            elem.normalized ? GL_TRUE : GL_FALSE, layout.GetStride(),
            reinterpret_cast<void *>(elem.offset + sizeof(float) * count * i));
        glVertexAttribDivisor(m_VertexBufferIndex, 1);
        m_VertexBufferIndex++;
      }
    }
    default:
      std::cout << "Unknown ShaderDataType!" << std::endl;
    }
  }
  m_vertexBuffers.push_back(vertexBuffer);
}

void OpenGLVertexArrayObject::SetIndexBuffer(
    const std::shared_ptr<IndexBuffer> &indexBuffer) {
  glBindVertexArray(m_id);
  indexBuffer->Bind();
  m_indexBuffer = indexBuffer;

OpenGLTexture::OpenGLTexture(const std::string &path) {
  int height, width, channels;
  stbi_set_flip_vertically_on_load(1);
  stbi_uc *data = stbi_load(path.c_str(), &width, &height, &channels, 0);
  if (data) {

    m_width = width;
    m_height = height;

    GLenum internalFormat = 0, dataFormat = 0;

    if (channels == 4) {
      internalFormat = GL_RGBA8;
      dataFormat = GL_RGBA;
    } else if (channels == 3) {
      internalFormat = GL_RGB8;
      dataFormat = GL_RGB;
    } else {
      stbi_image_free(data);
      throw std::runtime_error("Unsupported image format: " + path);
    }
    glGenTextures(1, &m_rendererID);
    glBindTexture(GL_TEXTURE_2D, m_rendererID);
    glTexImage2D(GL_TEXTURE_2D, 0, internalFormat, m_width, m_height, 0, dataFormat, GL_UNSIGNED_BYTE, data);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);

    stbi_image_free(data);
  } else {
    throw std::runtime_error("Failed to load texture: " + path);
  }
}

OpenGLShader::OpenGLShader(const std::string &filename) {
  std::ifstream stream(filename);
  std::string line;
  enum ShaderType { NONE = -1, VERTEX = 0, GEOMETRY = 1, FRAGMENT = 2 }; 
  std::stringstream ss[3];
  ShaderType type = ShaderType::NONE;
  
  while (getline(stream, line)) {
    if (line.find("#shader") != std::string::npos) {
      if (line.find("vertex") != std::string::npos) {
        type = ShaderType::VERTEX;
      } else if (line.find("geometry") != std::string::npos) { // ДОБАВИЛ
        type = ShaderType::GEOMETRY;
      } else if (line.find("fragment") != std::string::npos) {
        type = ShaderType::FRAGMENT;
      }
    } else {
      if (type != ShaderType::NONE) {
        ss[(int)type] << line << '\n';
      }
    }
  }

  m_rendererID = glCreateProgram();
  
  GLuint vertexShader = CompileShader(GL_VERTEX_SHADER, ss[0].str());
  glAttachShader(m_rendererID, vertexShader);
  
  if (!ss[1].str().empty()) {
    GLuint geometryShader = CompileShader(GL_GEOMETRY_SHADER, ss[1].str());
    glAttachShader(m_rendererID, geometryShader);
  }
  
  GLuint fragmentShader = CompileShader(GL_FRAGMENT_SHADER, ss[2].str());
  glAttachShader(m_rendererID, fragmentShader);
  
  glLinkProgram(m_rendererID);
  
  GLint isLinked;
  glGetProgramiv(m_rendererID, GL_LINK_STATUS, &isLinked);
  if (!isLinked) {
    GLint maxLength;
    glGetProgramiv(m_rendererID, GL_INFO_LOG_LENGTH, &maxLength);
    std::string infoLog(maxLength, ' ');
    glGetProgramInfoLog(m_rendererID, maxLength, &maxLength, &infoLog[0]);
    std::cout << "Shader linking failed: " + infoLog << '\n';
  }
}

GLuint OpenGLShader::CompileShader(GLenum type, const std::string &source) {
  GLuint shader = glCreateShader(type);
  const char *src = source.c_str();
  glShaderSource(shader, 1, &src, nullptr);
  glCompileShader(shader);
  GLint isCompiled;
  glGetShaderiv(shader, GL_COMPILE_STATUS, &isCompiled);
  if (!isCompiled) {
    GLint maxLength;
    glGetShaderiv(shader, GL_INFO_LOG_LENGTH, &maxLength);
    std::string infoLog(maxLength, ' ');
    glGetShaderInfoLog(shader, maxLength, &maxLength, &infoLog[0]);
    glDeleteShader(shader);
    std::cout << "Shader compilation failed: " + infoLog << '\n';
  }
  return shader;
}

std::shared_ptr<Texture> TextureCache::GetOrLoad(const std::string &path) {
  if (path.empty())
    return nullptr;

  auto it = m_cache.find(path);
  if (it != m_cache.end()) {
    if (auto texture = it->second.lock()) {
      return texture;
    }
  }

  auto texture = Texture::Create(path);
  m_cache[path] = texture;
  return texture;
}

===== src/graphics/Material.cc =====
#include "Material.h"
#include "TextureCache.h"
namespace engine {
std::shared_ptr<Material> Material::Create(const std::string &diffusePath,
                                           const std::string &specularPath,
                                           const std::string &normalPath,
                                           const glm::vec3 &diffuseColor,
                                           const glm::vec3 &specularColor,
                                           float shininess) {
  auto material = std::make_shared<Material>();
  material->m_diffuseColor = diffuseColor;
  material->m_specularColor = specularColor;
  material->m_shininess = shininess;

  auto &textureCache = TextureCache::Instance();

  if (!diffusePath.empty()) {
    material->m_diffuseTexture = textureCache.GetOrLoad(diffusePath);
  }
  if (!specularPath.empty()) {
    material->m_specularTexture = textureCache.GetOrLoad(specularPath);
  }
  if (!normalPath.empty()) {
    material->m_normalTexture = textureCache.GetOrLoad(normalPath);
  }

  return material;
}

void Material::Bind(Shader &shader) const {
  shader.SetVec3("material.diffuse", m_diffuseColor);
  shader.SetVec3("material.specular", m_specularColor);
  shader.SetBool("material.use_diffuse_map", false);
  shader.SetBool("material.use_normal_map", false);
  shader.SetBool("material.use_specular_map", false);

  shader.SetFloat("material.shininess", m_shininess);
}

    void Mesh::Draw(Shader& shader) {
        // Биндим текстуры
        unsigned int diffuseNr = 1;
        unsigned int specularNr = 1;
        unsigned int normalNr = 1;
        
        for (unsigned int i = 0; i < m_material.textures.size(); i++) {
            const auto& texInfo = m_material.textures[i];
            texInfo.texture->Bind(i);
            
            std::string number;
            std::string name = "texture_";
            
            if (texInfo.type == "diffuse")
                number = std::to_string(diffuseNr++);
            else if (texInfo.type == "specular")
                number = std::to_string(specularNr++);
            else if (texInfo.type == "normal")
                number = std::to_string(normalNr++);
                
            shader.SetInt(("material." + name + number).c_str(), i);
        }
        
        glActiveTexture(GL_TEXTURE0);
        
        SetupMaterial(shader, m_material);
        m_vao->Bind();
        
        glDrawElements(GL_TRIANGLES, m_indexCount, GL_UNSIGNED_INT, 0);
        glBindVertexArray(0);
    }

    void Mesh::DrawInstanced(Shader& shader, const std::vector<glm::mat4>& instanceMatrices) {
        if (instanceMatrices.empty()) return;
    
        SetupMaterial(shader, m_material);
        m_vao->Bind();
        
        m_vao->AddVertexBuffer(m_instanceBuffer);
        
        m_instanceBuffer->SetData(instanceMatrices.data(), instanceMatrices.size() * sizeof(glm::mat4));
        glDrawElementsInstanced(GL_TRIANGLES, m_indexCount, GL_UNSIGNED_INT, 0, instanceMatrices.size());
        
        m_vao->Unbind();
    }

    void Mesh::DrawGeometryOnly(Shader& shader) {
        m_vao->Bind();
        glDrawElements(GL_TRIANGLES, m_indexCount, GL_UNSIGNED_INT, 0);
        m_vao->Unbind();
    }

    
    void Mesh::SetupMaterial(Shader& shader, const data::MaterialData& material) {
          glm::vec3 diffuse = (material.diffuseColor == glm::vec3(0.0f)) 
        ? glm::vec3(0.8f) : material.diffuseColor;
        shader.SetVec3("material.diffuse", material.diffuseColor);
        shader.SetVec3("material.specular", material.specularColor);
        shader.SetFloat("material.shininess", material.shininess);
    }
    
    void Mesh::SetupMesh(const data::MeshData& meshData) {
        m_vao = VertexArrayObject::Create();
        
        m_vbo = VertexBuffer::Create(
            meshData.vertices.size() * sizeof(data::Vertex)
        );
        m_vbo->SetData(meshData.vertices.data(), meshData.vertices.size() * sizeof(data::Vertex));

        m_vbo->SetLayout({
            {ShaderDataType::Float3, "a_position"},
            {ShaderDataType::Float3, "a_normal"}, 
            {ShaderDataType::Float2, "a_texCoords"},
            {ShaderDataType::Float3, "a_tangent"},
            {ShaderDataType::Float3, "a_bitangent"}
        });

        m_vao->AddVertexBuffer(m_vbo);
        
        m_ebo = IndexBuffer::Create(
            const_cast<uint32_t*>(meshData.indices.data()),
            meshData.indices.size()
        );

        m_vao->SetIndexBuffer(m_ebo);
        m_indexCount = meshData.indices.size();
    }

    {
        for (auto& mesh : m_meshes) {
            mesh.Draw(shader);
        }
    }
    void Model::DrawInstanced(Shader& shader, const std::vector<glm::mat4>& instanceMatrices) {
        if (instanceMatrices.empty()) return;
        
        for (auto& mesh : m_meshes) {
            mesh.DrawInstanced(shader, instanceMatrices);
        }
    }

    void Model::DrawGeometryOnly(Shader& shader) {
          for (auto& mesh : m_meshes) {
            mesh.DrawGeometryOnly(shader);
        }
    }

    void Model::SetupModel()
    {
        for (size_t i = 0; i < m_modelData.meshes.size(); i++) {
            const auto& meshData = m_modelData.meshes[i];
            
            const data::MaterialData* material = nullptr;
            if (meshData.materialIndex < m_modelData.materials.size()) {
                material = &m_modelData.materials[meshData.materialIndex];
            } else {
                static data::MaterialData defaultMaterial;
                material = &defaultMaterial;
            }
            
            data::MaterialData materialWithTextures = *material;
            for (auto& texInfo : materialWithTextures.textures) {
                bool found = false;
                for (const auto& cachedTex : m_texturesLoaded) {
                    if (cachedTex.path == texInfo.path) {
                        texInfo.texture = cachedTex.texture;
                        found = true;
                        break;
                    }
                }
                
                if (!found) {
                    texInfo.texture = Texture::Create(texInfo.path);
                    m_texturesLoaded.push_back(texInfo);
                }
            }
            
            m_meshes.emplace_back(meshData, materialWithTextures);
        }
    }
}

#include "SceneObjectStore.h"
#include <utility> // std::swap

namespace engine {

size_t SceneObjectStore::Add(SceneObject obj) {
  m_objs.push_back(std::move(obj));
  return m_objs.size() - 1;
}

void SceneObjectStore::RemoveByIndex(size_t idx) {
  if (idx >= m_objs.size()) { return;
}
  if (idx != m_objs.size() - 1) {
    std::swap(m_objs[idx], m_objs.back());
  }
  m_objs.pop_back();
}

SceneObject* SceneObjectStore::Get(size_t idx) {
  if (idx >= m_objs.size()) { return nullptr;
}
  return &m_objs[idx];
}

const SceneObject* SceneObjectStore::Get(size_t idx) const {
  if (idx >= m_objs.size()) { return nullptr;
}
  return &m_objs[idx];
}

int SceneObjectStore::FindClosestXZ(glm::vec3 world, float maxXZDist) const {
  int   best   = -1;
  float bestD2 = maxXZDist * maxXZDist;

  for (int i = 0; i < static_cast<int>(m_objs.size()); ++i) {
    const auto& o = m_objs[i];
    if (!o.m_is_visible) { continue;
}
    const float DX = o.m_position.x - world.x;
    const float DZ = o.m_position.z - world.z;
    const float D2 = (DX*DX) + (DZ*DZ);
    if (D2 <= bestD2) { bestD2 = D2; best = i; }
  }
  return best;
}

#pragma once
#include <glm/glm.hpp>
#include <glm/gtc/matrix_transform.hpp>
#define YAW -90.0f
#define PITCH -40.0f
#define SPEED 50.0f
#define SENSITIVITY 0.1f
#define ZOOM 45.0f
namespace engine {

enum Camera_Movement { FORWARD, BACKWARD, LEFT, RIGHT };

class Camera {
public:
  Camera(glm::vec3 position = glm::vec3(0.0f, 0.0f, -3.0f),
         glm::vec3 up = glm::vec3(0.0f, 1.0f, 0.0f), float yaw = YAW,
         float pitch = PITCH)
      : m_front(glm::vec3(0.0f, 0.0f, 1.0f)), m_movementSpeed(SPEED),
        m_mouseSensitivity(SENSITIVITY), m_zoom(ZOOM) {
    m_position = position;
    m_worldUp = up;
    m_yaw = yaw;
    m_pitch = pitch;
    updateCameraVectors();
  }

  glm::mat4 GetViewMatrix() {
    return glm::lookAt(m_position, m_position + m_front, m_up);
  }

  void ProcessKeyboard(Camera_Movement direction, float deltaTime) {
    float velocity = m_movementSpeed * deltaTime;
    if (direction == FORWARD)
      m_position += m_front * velocity;
    if (direction == BACKWARD)
      m_position -= m_front * velocity;
    if (direction == LEFT)
      m_position -= m_right * velocity;
    if (direction == RIGHT)
      m_position += m_right * velocity;
  }

  void ProcessMouseMovement(float xoffset, float yoffset,
                            bool constrainPitch = true) {
    xoffset *= m_mouseSensitivity;
    yoffset *= m_mouseSensitivity;

    m_yaw += xoffset;
    m_pitch -= yoffset;

    if (constrainPitch) {
      if (m_pitch > 89.0f)
        m_pitch = 89.0f;
      if (m_pitch < -89.0f)
        m_pitch = -89.0f;
    }

    updateCameraVectors();
  }

  void updateCameraVectors() {
    // calculate the new Front vector
    glm::vec3 front;
    front.x = cos(m_yaw * (M_PI / 180.0)) * cos(m_pitch * (M_PI / 180.0));
    front.y = sin(m_pitch * (M_PI / 180.0));
    front.z = sin(m_yaw * (M_PI / 180.0)) * cos(m_pitch * (M_PI / 180.0));
    if (front.length() > 0) {
      m_front = glm::normalize(front);
    }

    // also re-calculate the Right and Up vector
    m_right = glm::normalize(glm::cross(m_front, m_worldUp));
    m_up = glm::normalize(glm::cross(m_right, m_front));
  }

  void ProcessMouseScroll(float yoffset) {
    m_zoom += (float)yoffset;
    if (m_zoom < 1.0f)
      m_zoom = 1.0f;
    if (m_zoom > 45.0f)
      m_zoom = 45.0f;
  }

  float GetZoom() const { return m_zoom; }
  glm::vec3 GetPosition() const { return m_position; }
  void LookAt(const glm::vec3 &target) {
    m_front = glm::normalize(target - m_position);
    // Пересчитываем остальные векторы
    updateCameraVectors();
  }

  void SetPosition(const glm::vec3 &position) {
    m_position = position;
    updateCameraVectors();
  }

private:
  glm::vec3 m_position;
  glm::vec3 m_front;
  glm::vec3 m_up;
  glm::vec3 m_right;
  glm::vec3 m_worldUp;
  // euler Angles
  float m_yaw;
  float m_pitch;
  // camera options
  float m_movementSpeed;
  float m_mouseSensitivity;
  float m_zoom;
};
} // namespace engine

void Scene::Render() {
  const int FB_W = static_cast<int>(m_vpW * m_vpScaleX);
  const int FB_H = static_cast<int>(m_vpH * m_vpScaleX);
  glViewport(0, 0, FB_W, FB_H);

  const float ASPECT =
      FB_H > 0 ? static_cast<float>(FB_W) / static_cast<float>(FB_H) : 1.0F;
  auto projection = glm::perspective(m_camera.GetZoom(), ASPECT, 0.1F, 2000.0F);

  auto view = m_camera.GetViewMatrix();
  Renderer::Instance().BeginScene(view, projection);

  auto agents = LoadAgentsFromGPU();

  // Рендеринг объектов
  for (const auto &obj : m_sceneObjects.All()) {
    if (!obj.m_is_visible) {
      continue;
    }

    if (obj.m_agent_id < agents.size()) {
      const Agent &agent = agents[obj.m_agent_id];
      RenderAgent(agent, obj);
    }
  }

  Renderer::Instance().UploadAgentsData(agents);
  Renderer::Instance().DrawTerrain(m_terrain->GetVAO(),
                                   m_terrain->GetPosition());
  Renderer::Instance().RenderWithGeometryShader(
      m_windConfig.getCurrentWind(m_windTime));

  Renderer::Instance().EndScene();
}

std::vector<Agent> Scene::LoadAgentsFromGPU() {
  size_t currentCount = m_agentManager.getAgentCount();
  std::vector<Agent> agents(currentCount);

  cudaMemcpy(agents.data(), m_agentManager.getDeviceAgents(),
             currentCount * sizeof(Agent), cudaMemcpyDeviceToHost);

  cudaError_t err = cudaGetLastError();
  if (err != cudaSuccess) {
    std::println("CUDA memcpy error: {}", cudaGetErrorString(err));
  }

  return agents;
}

void Scene::RenderAgent(const Agent &agent, const SceneObject &scene_obj) {
  glm::vec3 position(agent.position.x, agent.position.y, agent.position.z);

  // Получаем префаб
  const auto &prefab = m_prefabSystem.GetPrefab(scene_obj.m_object_type);

  // Получаем модель
  const ObjectModels *om = m_objectModels.Find(scene_obj.m_object_type);
  if ((om == nullptr) || !om->m_normal) {
    std::cerr << "[RenderAgent] no model for '" << scene_obj.m_object_type
              << "'\n";
    return;
  }

  // Определяем состояние агента
  AgentType state = AgentType::NORMAL;
  if (agent.type == AgentType::BURNED) {
    state = AgentType::BURNED;
  } else if (agent.isBurning() || agent.type == AgentType::BURNING) {
    state = AgentType::BURNING;
  }

  auto material = prefab.GetMaterialForState(state);
  auto model = om->m_normal;

  if (model && material) {
    Renderer::Instance().DrawModelWithMaterial(
        *model, position, scene_obj.m_scale, m_camera.GetPosition(), material);
  } else {
    // Fallback
    Renderer::Instance().DrawModel(*model, position, scene_obj.m_scale,
                                   m_camera.GetPosition());
  }
}

void Scene::AddPrefab(const std::string &prefab_id, float x, float z) {

  EnsureModelsForPrefab(prefab_id, prefab);

  obj.m_object_type = prefab_id; // ключ в m_objectModels
  obj.m_position = glm::vec3(x, y, z);
  obj.m_scale = prefab.m_modelScale;
  obj.m_is_visible = true;
  m_sceneObjects.Add(std::move(obj));
}

void Scene::RemoveSceneObjectByIndex(size_t idx) {
  const SceneObject *victim = m_sceneObjects.Get(idx);
  if (!victim)
    return;

  const int removed_agent = victim->m_agent_id;

  int swapped_from = -1;
  int swapped_to = -1;
  m_agentManager.RemoveAgentAt(removed_agent, &swapped_from, &swapped_to);

  m_sceneObjects.RemoveByIndex(idx);

  if (swapped_from >= 0 && swapped_to >= 0 && swapped_from != swapped_to) {
    for (auto &so : m_sceneObjects.All()) {
      if (so.m_agent_id == swapped_from) {
        so.m_agent_id = swapped_to;
        break;
      }
    }
  }
}

bool Scene::RemoveNearestAt(const glm::vec3 &world, float maxXZDist) {
  const int idx = FindClosestObjectXZ(world, maxXZDist);
  if (idx < 0) {
    return false;
  }
  RemoveSceneObjectByIndex(static_cast<size_t>(idx));
  return true;
}

#include "Terrain.h"
#include <cmath>
#include <cstdint>
#include <memory>
#include <vector>

#include "graphics/IndexBuffer.h"
#include "graphics/VertexArrayObject.h"
#include "graphics/VertexBuffer.h"
#include <glm/gtc/noise.hpp>
#include <random>
namespace engine {
class Terrain::PImpl {
public:
  int m_width;
  int m_depth;
  float m_scale = 1.0f;
  float m_maxHeight = 100.0f;
  uint32_t m_seed = 0;
  float m_frequency = 0.02f;
  int m_octaves = 6;
  float m_persistence = 0.5f;
  float m_lacunarity = 2.0f;
  glm::vec3 m_position = glm::vec3(0.0f);
  std::vector<std::vector<float>> m_heightMap;
  std::shared_ptr<VertexArrayObject> m_vao;

  float FBM(glm::vec2 p, const TerrainGenerationSettings &s) const {
    std::mt19937 gen(s.seed);
    std::uniform_real_distribution<float> dist(0.0f, 1000.0f);
    glm::vec2 offset(dist(gen), dist(gen));

    float total = 0.0f;
    float amplitude = 1.0f;
    float freq = s.frequency;
    float maxValue = 0.0f;

    for (int i = 0; i < s.octaves; ++i) {
      glm::vec2 coord = (p + offset) * freq;
      total += glm::perlin(coord) * amplitude;
      maxValue += amplitude;
      amplitude *= s.persistence;
      freq *= s.lacunarity;
    }

    return total / maxValue;
  }

  void GenerateHeightmapRandom(const TerrainGenerationSettings &s) {
    m_width = s.width;
    m_depth = s.depth;
    m_scale = s.scale;
    m_maxHeight = s.maxHeight;
    m_seed = s.seed;
    m_frequency = s.frequency;
    m_octaves = s.octaves;
    m_persistence = s.persistence;
    m_lacunarity = s.lacunarity;
    m_heightMap.assign(m_width, std::vector<float>(m_depth, 0.0f));

    for (int z = 0; z < m_depth; ++z) {
      for (int x = 0; x < m_width; ++x) {
        float nx = static_cast<float>(x);
        float nz = static_cast<float>(z);

        float h = FBM({nx, nz}, s);

        h = (h + 1.0f) * 0.5f;

        m_heightMap[x][z] = h * s.maxHeight;
      }
    }
  }

  void GenerateMesh() {
    std::vector<float> vertices;
    std::vector<uint32_t> indices;

    for (int z = 0; z < m_depth; ++z) {
      for (int x = 0; x < m_width; ++x) {
        float worldX = x * m_scale;
        float worldZ = z * m_scale;
        float y = m_heightMap[x][z];

        glm::vec3 normal = {0, 1, 0};
        if (x > 0 && x < m_width - 1 && z > 0 && z < m_depth - 1) {
          float hL = m_heightMap[x - 1][z];
          float hR = m_heightMap[x + 1][z];
          float hD = m_heightMap[x][z - 1];
          float hU = m_heightMap[x][z + 1];
          normal = glm::normalize(glm::vec3(hL - hR, 2.0f * m_scale, hD - hU));
        }

        // UV
        float u = static_cast<float>(x) / (m_width - 1);
        float v = static_cast<float>(z) / (m_depth - 1);

        vertices.insert(vertices.end(), {worldX, y, worldZ, normal.x, normal.y,
                                         normal.z, u, v});
      }
    }

    for (int z = 0; z < m_depth - 1; ++z) {
      for (int x = 0; x < m_width - 1; ++x) {
        uint32_t i = x + z * m_width;
        indices.insert(indices.end(), {i, i + m_width, i + 1, i + 1,
                                       i + m_width, i + m_width + 1});
      }
    }

    auto vbo =
        VertexBuffer::Create(vertices.data(), vertices.size() * sizeof(float));
    vbo->SetLayout({{ShaderDataType::Float3, "a_Position"},
                    {ShaderDataType::Float3, "a_Normal"},
                    {ShaderDataType::Float2, "a_TexCoord"}});

    auto ibo = IndexBuffer::Create(indices.data(), indices.size());
    m_vao = VertexArrayObject::Create();
    m_vao->AddVertexBuffer(vbo);
    m_vao->SetIndexBuffer(ibo);
  }
  void Cleanup() { m_vao.reset(); }
};

void AgentManager::IgniteRandomPoint(int agent_index) {
  if (agent_index < 0 || agent_index >= m_count)
    return;
  Agent agent;
  cudaMemcpy(&agent, m_agents + agent_index, sizeof(Agent),
             cudaMemcpyDeviceToHost);

  if (agent.fire_collider.point_count > 0) {
    int random_point = rand() % agent.fire_collider.point_count;
    agent.fire_collider.intensities[random_point] = 1.0f;
    cudaMemcpy(m_agents + agent_index, &agent, sizeof(Agent),
               cudaMemcpyHostToDevice);
  }
}

void AgentManager::generateTreeHeatPoints(FireCollider &collider, float base_x,
                                          float base_z) {
  collider.point_count = 0;

  for (float height = 0.5f; height <= 8.0f; height += 0.8f) {
    collider.heat_points[collider.point_count].x = base_x;
    collider.heat_points[collider.point_count].y = height;
    collider.heat_points[collider.point_count].z = base_z;
    collider.fuels[collider.point_count] = 30.0f;
    collider.intensities[collider.point_count] = 0.0f;
    collider.point_count++;
  }

  float levels[] = {2.0f, 4.0f, 6.0f, 7.5f};
  float level_radii[] = {1.0f, 1.8f, 2.2f, 1.5f};
  int points_per_level[] = {8, 12, 16, 8};

  for (int level = 0; level < 4; level++) {
    float height = levels[level];
    float radius = level_radii[level];
    int points = points_per_level[level];

    for (int i = 0; i < points; i++) {
      float angle = (float)i / points * 360.0f;
      float rad = glm::radians(angle);

      collider.heat_points[collider.point_count].x = base_x + cos(rad) * radius;
      collider.heat_points[collider.point_count].y = height;
      collider.heat_points[collider.point_count].z = base_z + sin(rad) * radius;
      collider.fuels[collider.point_count] = 20.0f;
      collider.intensities[collider.point_count] = 0.0f;
      collider.point_count++;

      if (collider.point_count >= 64)
        break;
    }
    if (collider.point_count >= 64)
      break;
  }

  std::random_device rd;
  std::mt19937 gen(rd());
  std::uniform_real_distribution<float> dist_x(base_x - 2.0f, base_x + 2.0f);
  std::uniform_real_distribution<float> dist_z(base_z - 2.0f, base_z + 2.0f);
  std::uniform_real_distribution<float> dist_y(1.0f, 7.0f);

  while (collider.point_count < 50 && collider.point_count < 64) {
    collider.heat_points[collider.point_count].x = dist_x(gen);
    collider.heat_points[collider.point_count].y = dist_y(gen);
    collider.heat_points[collider.point_count].z = dist_z(gen);
    collider.fuels[collider.point_count] = 15.0f;
    collider.intensities[collider.point_count] = 0.0f;
    collider.point_count++;
  }
}

Agent AgentManager::CreateAgent(AgentType type, Float3 position,
                                const AgentComponents &components) {
  Agent agent;
  agent.type = type;
  agent.position = position;
  agent.object_id = components.object_id;
  agent.intensity = components.intencity;

  agent.fire_collider.bounds_min = components.collider_bounds_min;

  generateTreeHeatPoints(agent.fire_collider, position.x, position.z);

bool AgentManager::RemoveAgentAt(int idx, int* outFrom, int* outTo) {
  if (idx < 0 || (size_t)idx >= m_count) return false;
  const int last = (int)m_count - 1;
  if (outFrom) *outFrom = -1;
  if (outTo)   *outTo   = -1;

  if (idx != last) {
    // копируем последнего агента на место удаляемого
    cudaMemcpy(m_agents + idx, m_agents + last,
               sizeof(Agent), cudaMemcpyDeviceToDevice);
    if (outFrom) *outFrom = last;
    if (outTo)   *outTo   = idx;
  }
  // просто уменьшаем "логическое" количество
  --m_count;
  // (опционально) cudaGetLastError проверка
  return true;
}

#pragma once
#include <glm/glm.hpp>

#include "cuda_include.h"
enum AgentType {
    NORMAL = 0,
    BURNING = 1,   
    BURNED = 2
};

struct Float3 {
    float x, y, z;
    
    __host__ __device__ Float3() : x(0), y(0), z(0) {}
    __host__ __device__ Float3(float x, float y, float z) : x(x), y(y), z(z) {}
    __host__ __device__ static Float3 fromGlm(const glm::vec3& v) {
        return Float3(v.x, v.y, v.z);
    }
    
    __host__ __device__ glm::vec3 toGlm() const {
        return glm::vec3(x, y, z);
    }
    __host__ __device__ float& operator[](int i) { 
        return i == 0 ? x : (i == 1 ? y : z); 
    }
    __host__ __device__ float operator[](int i) const { 
        return i == 0 ? x : (i == 1 ? y : z); 
    }
};

struct FireCollider {
    Float3 bounds_min, bounds_max;
    Float3 heat_points[64];
    float fuels[64];
    float intensities[64];
    int point_count;
    
    __host__ __device__
    FireCollider() : point_count(0) {
        for (int i = 0; i < 3; i++) {
            bounds_min[i] = 0.0f;
            bounds_max[i] = 0.0f;
        }
    }
};

struct Agent {
    Float3 position;
    AgentType type;
    float intensity{};
    int object_id; 
    
    // Универсальный коллайдер для огня
    FireCollider fire_collider;
    
    __host__ __device__
    Agent() : position(), type(NORMAL), object_id(-1) {}
    
    __host__ __device__
    Agent(float x, float y, float z, AgentType type, int id) 
        : position(x, y, z), type(type), object_id(id) {}
    
    __host__ __device__
    bool isBurning() const { 
        for (int i = 0; i < fire_collider.point_count; i++) {
            if (fire_collider.intensities[i] > 0.1f) {
                return true;  
            }
        }
        return false;
    }
    
    __host__ __device__
    float getOverallIntensity() const {
        float max = 0.0f;
        for (int i = 0; i < fire_collider.point_count; i++) {
            if (fire_collider.intensities[i] > max) {
                max = fire_collider.intensities[i];
            }
        }
        return max;
    }
};

    for (auto &e : m_extensions)

  const bool previous = (m_previousKeyState.count(key) != 0U) ? m_previousKeyState.at(key) : false;

  switch (state) {
    case InputState::Held:     return currentIt;
    case InputState::Pressed:  return  currentIt && !previous;
    case InputState::Released: return !currentIt &&  previous;
  }
  return false;
}

===== src/math/ScreenRay.h =====
#pragma once
#include <cmath>
#include <glm/glm.hpp>
#include <glm/gtc/matrix_inverse.hpp>

namespace math {
inline void ScreenPointToRay(float px, float py, float viewportW,
                             float viewportH, const glm::mat4 &proj,
                             const glm::mat4 &view, glm::vec3 &outOrigin,
                             glm::vec3 &outDir) {
  float x = (2.0F * px / viewportW) - 1.0F;
  float y = 1.0F - (2.0F * py / viewportH);
  glm::vec4 ray_clip(x, y, -1.0F, 1.0F);
  glm::mat4 inv_proj = glm::inverse(proj);
  glm::vec4 ray_eye = inv_proj * ray_clip;
  ray_eye = glm::vec4(ray_eye.x, ray_eye.y, -1.0f, 0.0f);
  glm::mat4 inv_view = glm::inverse(view);
  glm::vec4 ray_world4 = inv_view * ray_eye;
  glm::vec3 ray_world = glm::normalize(glm::vec3(ray_world4));
  outOrigin = glm::vec3(inv_view[3]);
  outDir = ray_world;
}

inline bool IntersectRayPlaneY0(const glm::vec3 &ro, const glm::vec3 &rd,
                                glm::vec3 &hit) {
  if (std::fabs(rd.y) < 1e-6F) {
    return false;
}
  float t = -ro.y / rd.y;
  if (t < 0.0F) {
    return false;
}
  hit = ro + rd * t;
  return true;
}

inline bool IntersectRayPlaneAtDepth(const glm::vec3 &ro, const glm::vec3 &rd,
                                     const glm::vec3 &camDir, float depth,
                                     glm::vec3 &hit) {
  // Плоскость: нормаль = camDir, точка на плоскости = ro + depth * camDir
  glm::vec3 planePoint = ro + depth * camDir;
  float denom = glm::dot(rd, camDir);
  if (std::fabs(denom) < 1e-6f) {
    return false; // Луч параллелен плоскости
  }
  float t = glm::dot(planePoint - ro, camDir) / denom;
  if (t < 0.0f) {
    return false; // Пересечение позади камеры
  }
  hit = ro + rd * t;
  return true;
}
} // namespace math

bool SceneSerializer::Serialize(const std::string &filepath) {
  try {
    YAML::Emitter out;
    out << YAML::BeginMap;

    // Метаданные
    out << YAML::Key << "metadata" << YAML::Value;
    out << YAML::BeginMap;
    out << YAML::Key << "version" << YAML::Value << "1.0";
    out << YAML::Key << "date" << YAML::Value << getCurrentDateTime();
    out << YAML::EndMap;

    // Агенты
    out << YAML::Key << "agents" << YAML::Value;
    out << YAML::BeginSeq;
    auto agents = m_scene->LoadAgentsFromGPU();
    for (int i = 0; i < agents.size(); i++) {
      SerializeAgent(out, agents[i], i);
    }
    out << YAML::EndSeq;

    // SceneObject'ы
    out << YAML::Key << "scene_objects" << YAML::Value;
    out << YAML::BeginSeq;
    for (int i = 0; i < m_scene->GetSceneObjects().size(); i++) {
      SerializeSceneObject(out, m_scene->GetSceneObjects()[i], i);
    }
    out << YAML::EndSeq;
    out << YAML::Key << "terrain" << YAML::Value;
    out << YAML::BeginMap;

    // Получаем настройки из текущего террейна
    auto terrain = m_scene->GetTerrain();
    if (terrain) {
      out << YAML::Key << "width" << YAML::Value << terrain->GetWidth();
      out << YAML::Key << "depth" << YAML::Value << terrain->GetDepth();
      out << YAML::Key << "scale" << YAML::Value << terrain->GetScale();
      out << YAML::Key << "maxHeight" << YAML::Value << terrain->GetMaxHeight();
      out << YAML::Key << "seed" << YAML::Value << terrain->GetSeed();
      out << YAML::Key << "frequency" << YAML::Value << terrain->GetFrequency();
      out << YAML::Key << "octaves" << YAML::Value << terrain->GetOctaves();
      out << YAML::Key << "persistence" << YAML::Value
          << terrain->GetPersistence();
      out << YAML::Key << "lacunarity" << YAML::Value
          << terrain->GetLacunarity();
      out << YAML::Key << "position" << YAML::Value;
      out << YAML::Flow << YAML::BeginSeq << terrain->GetPosition().x
          << terrain->GetPosition().y << terrain->GetPosition().z
          << YAML::EndSeq;
    }
    out << YAML::EndMap;
    out << YAML::EndMap;

    std::ofstream file(filepath);
    file << out.c_str();
    std::cout << "Scene saved to: " << filepath << std::endl;
    return true;

  } catch (const std::exception &e) {
    std::cerr << "Failed to serialize scene: " << e.what() << std::endl;
    return false;
  }
}

    // Heat points (только используемые)
    out << YAML::Key << "heat_points" << YAML::Value;
    out << YAML::BeginSeq;
    for (int i = 0; i < collider.point_count; i++) {
      out << YAML::Flow << YAML::BeginSeq << collider.heat_points[i].x
          << collider.heat_points[i].y << collider.heat_points[i].z
          << YAML::EndSeq;
    }
    out << YAML::EndSeq;

    // Intensities
    out << YAML::Key << "intensities" << YAML::Value;
    out << YAML::Flow << YAML::BeginSeq;
    for (int i = 0; i < collider.point_count; i++) {

    }
    out << YAML::EndSeq;

    // Fuels
    out << YAML::Key << "fuels" << YAML::Value;

      out << collider.fuels[i];
    }
    out << YAML::EndSeq;

    out << YAML::EndMap;

    return out;
  }

void SceneSerializer::Deserialize(const std::string &filename) {
  try {
    YAML::Node scene_data = YAML::LoadFile(filename);

    // Очищаем сцену
    m_scene->Clear();

    // 1. Загружаем агентов
    const YAML::Node &agents_node = scene_data["agents"];
    std::vector<Agent> agents;

    for (const YAML::Node &agent_node : agents_node) {
      agents.push_back(DeserializeAgent(agent_node));
    }

    // Загружаем агентов в GPU
    m_scene->GetAgentManager().SetAgents(agents);

    const YAML::Node &objects_node = scene_data["scene_objects"];
    for (const YAML::Node &obj_node : objects_node) {
      SceneObject obj = DeserializeSceneObject(obj_node);
      m_scene->AddSceneObjectFromSerialized(obj);
    }

    const YAML::Node &terrain_node = scene_data["terrain"];
    if (terrain_node) {
      TerrainGenerationSettings s;
      s.width = terrain_node["width"].as<int>();
      s.depth = terrain_node["depth"].as<int>();
      s.scale = terrain_node["scale"].as<float>();
      s.maxHeight = terrain_node["maxHeight"].as<float>();
      s.seed = terrain_node["seed"].as<uint32_t>();
      s.frequency = terrain_node["frequency"].as<float>();
      s.octaves = terrain_node["octaves"].as<int>();
      s.persistence = terrain_node["persistence"].as<float>();
      s.lacunarity = terrain_node["lacunarity"].as<float>();
      const YAML::Node &pos_node = terrain_node["position"];
      if (pos_node) {
        glm::vec3 pos(pos_node[0].as<float>(), pos_node[1].as<float>(),
                      pos_node[2].as<float>());
        m_scene->GetTerrain()->SetPosition(pos);
      }
      m_scene->GenerateTerrain(s);
    }

    std::cout << "Scene loaded: " << agents.size() << " agents, "
              << objects_node.size() << " objects" << std::endl;

  } catch (const std::exception &e) {
    std::cerr << "Failed to deserialize scene: " << e.what() << std::endl;
  }
}
SceneObject
SceneSerializer::DeserializeSceneObject(const YAML::Node &obj_node) {

  obj.m_object_type = obj_node["object_type"].as<std::string>();

  // Позиция
  const YAML::Node &pos_node = obj_node["position"];
  obj.m_position = glm::vec3(pos_node[0].as<float>(), pos_node[1].as<float>(),
                             pos_node[2].as<float>());

  // Масштаб
  const YAML::Node &scale_node = obj_node["scale"];
  obj.m_scale = glm::vec3(scale_node[0].as<float>(), scale_node[1].as<float>(),
                          scale_node[2].as<float>());

  return obj;
}

FireCollider
SceneSerializer::DeserializeFireCollider(const YAML::Node &collider_node) {
  FireCollider collider;

  const YAML::Node &min_node = collider_node["bounds_min"];
  collider.bounds_min = Float3(min_node[0].as<float>(), min_node[1].as<float>(),
                               min_node[2].as<float>());

  const YAML::Node &max_node = collider_node["bounds_max"];
  collider.bounds_max = Float3(max_node[0].as<float>(), max_node[1].as<float>(),
                               max_node[2].as<float>());

  // Point count
  collider.point_count = collider_node["point_count"].as<int>();

  // Heat points
  const YAML::Node &points_node = collider_node["heat_points"];
  for (int i = 0; i < collider.point_count && i < points_node.size(); i++) {
    const YAML::Node &point_node = points_node[i];
    collider.heat_points[i] =
        Float3(point_node[0].as<float>(), point_node[1].as<float>(),
               point_node[2].as<float>());
  }

  // Intensities
  const YAML::Node &intensities_node = collider_node["intensities"];
  for (int i = 0; i < collider.point_count && i < intensities_node.size();
       i++) {
    collider.intensities[i] = intensities_node[i].as<float>();
  }

  // Fuels
  const YAML::Node &fuels_node = collider_node["fuels"];
  for (int i = 0; i < collider.point_count && i < fuels_node.size(); i++) {
    collider.fuels[i] = fuels_node[i].as<float>();
  }

  return collider;
}

Agent SceneSerializer::DeserializeAgent(const YAML::Node &agent_node) {
  Agent agent;

  agent.type = static_cast<AgentType>(agent_node["type"].as<int>());
  agent.object_id = agent_node["object_id"].as<int>();
  agent.intensity = agent_node["intensity"].as<float>();

  const YAML::Node &pos_node = agent_node["position"];
  agent.position = Float3(pos_node[0].as<float>(), pos_node[1].as<float>(),
                          pos_node[2].as<float>());

  if (agent_node["fire_collider"]) {
    agent.fire_collider = DeserializeFireCollider(agent_node["fire_collider"]);
  }

  return agent;
}

#include "ModelData.h"

    ModelData AssimpLoader::LoadModel(const std::string& path) {
        Assimp::Importer importer;
        const aiScene* scene = importer.ReadFile(path, ModelData::IMPORT_FLAGS);
        
        if (!scene || scene->mFlags & AI_SCENE_FLAGS_INCOMPLETE || !scene->mRootNode) {
            throw std::runtime_error("Assimp error: " + std::string(importer.GetErrorString()));
        }
        
        ModelData modelData;
        modelData.directory = path.substr(0, path.find_last_of('/'));
        
        for (unsigned int i = 0; i < scene->mNumMaterials; i++) {
            modelData.materials.push_back(
                ProcessMaterial(scene->mMaterials[i], modelData.directory)
            );
        }
        
        ProcessNode(scene->mRootNode, scene, modelData);
        
        return modelData;
    }

    void AssimpLoader::ProcessNode(aiNode *node, const aiScene *scene, ModelData &modelData)
    {
        for(unsigned int i = 0; i < node->mNumMeshes; i++)
        {
            aiMesh* mesh = scene->mMeshes[node->mMeshes[i]];
            modelData.meshes.push_back(ProcessMesh(mesh, scene));
        }
        for(unsigned int i = 0; i < node->mNumChildren; i++)
        {
            ProcessNode(node->mChildren[i], scene, modelData);
        }
    }

    MeshData AssimpLoader::ProcessMesh(aiMesh *mesh, const aiScene *scene)
    {
        MeshData meshData;
        meshData.materialIndex = mesh->mMaterialIndex;
        
        for(unsigned int i = 0; i < mesh->mNumVertices; i++) {
            Vertex vertex;

            // Позиции
            vertex.position = glm::vec3(
                mesh->mVertices[i].x,
                mesh->mVertices[i].y, 
                mesh->mVertices[i].z
            );

            // Нормали
            if(mesh->HasNormals()) {
                vertex.normal = glm::vec3(
                    mesh->mNormals[i].x,
                    mesh->mNormals[i].y,
                    mesh->mNormals[i].z
                );
            }

            // Текстурные координаты (берем первый набор)
            if(mesh->mTextureCoords[0]) {
                vertex.texCoords = glm::vec2(
                    mesh->mTextureCoords[0][i].x,
                    mesh->mTextureCoords[0][i].y
                );
            } else {
                vertex.texCoords = glm::vec2(0.0f);
            }

            // Тангенты и битангенты (если есть)
            if(mesh->HasTangentsAndBitangents()) {
                vertex.tangent = glm::vec3(
                    mesh->mTangents[i].x,
                    mesh->mTangents[i].y,
                    mesh->mTangents[i].z
                );
                vertex.bitangent = glm::vec3(
                    mesh->mBitangents[i].x,
                    mesh->mBitangents[i].y, 
                    mesh->mBitangents[i].z
                );
            }

            meshData.vertices.push_back(vertex);
        }

        // Обрабатываем индексы
        for(unsigned int i = 0; i < mesh->mNumFaces; i++) {
            aiFace face = mesh->mFaces[i];
            for(unsigned int j = 0; j < face.mNumIndices; j++) {
                meshData.indices.push_back(face.mIndices[j]);
            }
        }

        return meshData;
    }
    MaterialData AssimpLoader::ProcessMaterial(aiMaterial* material, const std::string& directory) {
        MaterialData materialData;

        aiString name;
        if (material->Get(AI_MATKEY_NAME, name) == AI_SUCCESS) {
            materialData.name = name.C_Str();
        }

        // Диффузный цвет
        aiColor3D diffuse(0.f, 0.f, 0.f);
        if (material->Get(AI_MATKEY_COLOR_DIFFUSE, diffuse) == AI_SUCCESS) {
            materialData.diffuseColor = glm::vec3(diffuse.r, diffuse.g, diffuse.b);
        }

        // Спекулярный цвет
        aiColor3D specular(0.f, 0.f, 0.f);
        if (material->Get(AI_MATKEY_COLOR_SPECULAR, specular) == AI_SUCCESS) {
            materialData.specularColor = glm::vec3(specular.r, specular.g, specular.b);
        }

        // Блеск
        float shininess = 32.0f;
        if (material->Get(AI_MATKEY_SHININESS, shininess) == AI_SUCCESS) {
            materialData.shininess = shininess;
        }

        auto diffuseMaps = LoadMaterialTextures(material, aiTextureType_DIFFUSE, "texture_diffuse", directory);
        materialData.textures.insert(materialData.textures.end(), diffuseMaps.begin(), diffuseMaps.end());

        auto specularMaps = LoadMaterialTextures(material, aiTextureType_SPECULAR, "texture_specular", directory);
        materialData.textures.insert(materialData.textures.end(), specularMaps.begin(), specularMaps.end());

        auto normalMaps = LoadMaterialTextures(material, aiTextureType_NORMALS, "texture_normal", directory);
        materialData.textures.insert(materialData.textures.end(), normalMaps.begin(), normalMaps.end());
       
        return materialData;
    }

    std::vector<TextureData> AssimpLoader::LoadMaterialTextures(
        aiMaterial* mat, aiTextureType type, const std::string& typeName, const std::string& directory) {
        
        std::vector<TextureData> textures;
        
        for (unsigned int i = 0; i < mat->GetTextureCount(type); i++) {
            aiString str;
            mat->GetTexture(type, i, &str);

            TextureData textureData;
            textureData.type = typeName;
            textureData.path = directory + "/" + str.C_Str();

            textures.push_back(textureData);
        }

        return textures;
    }
}

static void PushUnique(std::vector<std::string> &v, const std::string &id) {

void PrefabSystem::SavePrefabToFile(const std::string &prefab_name,
                                    const std::string &file_path) {
  if (!PrefabExists(prefab_name)) {
    throw std::runtime_error("Prefab not found: " + prefab_name);
  }

  const AgentPrefab &prefab = GetPrefab(prefab_name);

  YAML::Node root;
  root["name"] = prefab_name;
  SerializePrefab(prefab, root);

  fs::path path(file_path);
  fs::create_directories(path.parent_path());

  std::ofstream file(file_path);
  file << root;
  PushUnique(m_idsInLoadOrder, prefab_name);
  std::cout << "Saved prefab '" << prefab_name << "' to " << file_path << '\n';
}
void PrefabSystem::LoadPrefabFromFile(const std::string &file_path) {
  if (!fs::exists(file_path)) {
    throw std::runtime_error("File not found: " + file_path);
  }

  YAML::Node root = YAML::LoadFile(file_path);

  // name из YAML, иначе — из имени файла
  std::string name;
  if (auto n = root["name"]; n && n.IsScalar()) {
    name = n.as<std::string>();
  } else {
    name = fs::path(file_path).stem().string();
  }

  AgentPrefab prefab = DeserializePrefab(root);
  prefab.m_name = name; // <<< ВАЖНО: заполняем поле структуры

  // защита от дубликатов (как минимум лог и замена/пропуск)
  auto it = m_prefabs.find(name);
  if (it != m_prefabs.end()) {
    std::cerr << "[PrefabSystem] duplicate name '" << name << "' from "
              << file_path << " — replacing previous.\n";
  }

  m_prefabs[name] = std::move(prefab);
  PushUnique(m_idsInLoadOrder, name);
  std::cout << "Loaded prefab '" << name << "' from " << file_path << '\n';
}

Agent PrefabSystem::CreateAgentFromPrefab(const AgentPrefab &prefab,
                                          const glm::vec3 &position) {
  Agent agent;
  agent.position = Float3::fromGlm(position);
  agent.fire_collider.bounds_min =
      Float3::fromGlm(prefab.m_bounds_min + position);
  agent.fire_collider.bounds_max =
      Float3::fromGlm(prefab.m_bounds_max + position);

  agent.fire_collider.point_count =
      std::min(static_cast<int>(prefab.m_firePoints.size()), 64);

  for (int i = 0; i < agent.fire_collider.point_count; ++i) {
    Float3 local_pos = Float3::fromGlm(prefab.m_firePoints[i].m_position);
    agent.fire_collider.heat_points[i] =
        Float3(position.x + local_pos.x, position.y + local_pos.y,
               position.z + local_pos.z);
    agent.fire_collider.fuels[i] = prefab.m_firePoints[i].m_initialFuel;
    agent.fire_collider.intensities[i] = 0.0F;
  }

  agent.intensity = prefab.m_physics.m_intencity;
  agent.type = prefab.m_type;

  return agent;
}

void PrefabSystem::CreatePrefabFromAgent(const std::string &name,
                                         const Agent &agent,
                                         const std::string &model_path,
                                         const glm::vec3 &scale) {
  AgentPrefab prefab;
  prefab.m_name = name;
  prefab.m_type = agent.type;
  prefab.m_modelPath = model_path;
  prefab.m_modelScale = scale;
  prefab.m_bounds_min = agent.fire_collider.bounds_min.toGlm();
  prefab.m_bounds_max = agent.fire_collider.bounds_max.toGlm();

  for (int i = 0; i < agent.fire_collider.point_count; i++) {
    AgentPrefab::FirePoint fp;
    fp.m_position =
        glm::vec3(agent.fire_collider.heat_points[i].x - agent.position.x,
                  agent.fire_collider.heat_points[i].y - agent.position.y,
                  agent.fire_collider.heat_points[i].z - agent.position.z);
    fp.m_initialFuel = agent.fire_collider.fuels[i];
    prefab.m_firePoints.push_back(fp);
  }

  prefab.m_physics.m_intencity = agent.intensity;

  m_prefabs[name] = prefab;
  PushUnique(m_idsInLoadOrder, name);
}

void PrefabSystem::SerializePrefab(const AgentPrefab &prefab,
                                   YAML::Node &node) const {
  node["type"] = static_cast<int>(prefab.m_type);
  node["model_path"] = prefab.m_modelPath;
  node["model_scale"] = std::vector<float>{
      prefab.m_modelScale.x, prefab.m_modelScale.y, prefab.m_modelScale.z};
  node["bounds_min"] = std::vector<float>{
      prefab.m_bounds_min.x, prefab.m_bounds_min.y, prefab.m_bounds_min.z};
  node["bounds_max"] = std::vector<float>{
      prefab.m_bounds_max.x, prefab.m_bounds_max.y, prefab.m_bounds_max.z};

  node["physics"]["intensity"] = prefab.m_physics.m_intencity;

  for (const auto &fp : prefab.m_firePoints) {
    YAML::Node fp_node;
    fp_node["position"] =
        std::vector<float>{fp.m_position.x, fp.m_position.y, fp.m_position.z};
    fp_node["initial_fuel"] = fp.m_initialFuel;
    node["fire_points"].push_back(fp_node);
  }
  YAML::Node normal_node;
  YAML::Node burning_node;
  YAML::Node burned_node;

  SerializeMaterialInfo(prefab.normal_material, normal_node);
  SerializeMaterialInfo(prefab.burning_material, burning_node);
  SerializeMaterialInfo(prefab.burned_material, burned_node);

  // ПРИСВАИВАЕМ созданные ноды
  node["materials"]["normal"] = normal_node;
  node["materials"]["burning"] = burning_node;
  node["materials"]["burned"] = burned_node;
}

// устойчивый к неполным файлам
AgentPrefab PrefabSystem::DeserializePrefab(const YAML::Node &node) const {
  AgentPrefab prefab;

  // type
  if (auto t = node["type"]) {
    prefab.m_type = static_cast<AgentType>(t.as<int>());
  } else {
    prefab.m_type = AgentType::NORMAL;
  }

  // model_path
  if (auto mp = node["model_path"]) {
    prefab.m_modelPath = mp.as<std::string>();
  } else {
    prefab.m_modelPath.clear();
  }

  // model_scale (опционален)
  if (auto ms = node["model_scale"]) {
    auto v = ms.as<std::vector<float>>();
    if (v.size() >= 3)
      prefab.m_modelScale = glm::vec3(v[0], v[1], v[2]);
    else
      prefab.m_modelScale = glm::vec3(1.0f);
  } else {
    prefab.m_modelScale = glm::vec3(1.0f);
  }

  // bounds_min / bounds_max (обязательные для твоей логики — но читаем бережно)
  if (auto bmin = node["bounds_min"]) {
    auto v = bmin.as<std::vector<float>>();
    if (v.size() >= 3)
      prefab.m_bounds_min = glm::vec3(v[0], v[1], v[2]);
  }
  if (auto bmax = node["bounds_max"]) {
    auto v = bmax.as<std::vector<float>>();
    if (v.size() >= 3)
      prefab.m_bounds_max = glm::vec3(v[0], v[1], v[2]);
  }

  // physics.intensity (может отсутствовать)
  if (auto phys = node["physics"]) {
    if (auto inten = phys["intensity"]) {
      prefab.m_physics.m_intencity = inten.as<float>();
    } else {
      prefab.m_physics.m_intencity = 0.0f;
    }
  } else {
    prefab.m_physics.m_intencity = 0.0f;
  }

  // fire_points (опциональны)
  prefab.m_firePoints.clear();
  if (auto fps = node["fire_points"]) {
    for (const auto &fp_node : fps) {
      AgentPrefab::FirePoint fp{};
      if (auto pos = fp_node["position"]) {
        auto v = pos.as<std::vector<float>>();
        if (v.size() >= 3)
          fp.m_position = glm::vec3(v[0], v[1], v[2]);
      }
      if (auto fuel = fp_node["initial_fuel"]) {
        fp.m_initialFuel = fuel.as<float>();
      } else {
        fp.m_initialFuel = 100.0f;
      }
      prefab.m_firePoints.push_back(fp);
    }
  }
  // Materials - ЗАПОЛНЯЕМ MaterialInfo
  if (node["materials"]["normal"]) {
    prefab.normal_material = ParseMaterialInfo(node["materials"]["normal"]);
  }
  if (node["materials"]["burning"]) {
    prefab.burning_material = ParseMaterialInfo(node["materials"]["burning"]);
  }
  if (node["materials"]["burned"]) {
    prefab.burned_material = ParseMaterialInfo(node["materials"]["burned"]);
  }

  prefab.LoadMaterials();

  return prefab;
}

void PrefabSystem::Clear() {
  m_prefabs.clear();
  m_idsInLoadOrder.clear();
}

void PrefabSystem::LoadAllFromDirectory(const std::string &dir) {
  namespace fs = std::filesystem;
  Clear();

  fs::path root(dir);
  if (!fs::exists(root))
    return;

  for (auto &e : fs::directory_iterator(root)) {
    if (!e.is_regular_file())
      continue;
    const auto ext = e.path().extension().string();
    if (ext != ".yaml" && ext != ".yml")
      continue;

    try {
      LoadPrefabFromFile(e.path().string());
    } catch (const std::exception &ex) {
      std::cerr << "[PrefabSystem] skip " << e.path() << " : " << ex.what()
                << "\n";
      continue;
    }
  }
}

  MaterialInfo burned_material;

bool PlaceOnGroundTool::RaycastTerrain(const engine::Terrain& T, const glm::vec3& ro,
                                       const glm::vec3& rd, float anchorY,
                                       glm::vec3& outHit) {
  auto f = [&](float t) -> float {
    const glm::vec3 p = ro + rd * t;
    return p.y - (T.GetHeightAt(p.x, p.z) + anchorY);
  };

  const float T_MAX = 5000.F;
  const float STEP = 2.F;

  float t0 = 0.F;
  float v0 = f(t0);

  for (float t = STEP; t <= T_MAX; t += STEP) {
    float v1 = f(t);
    const bool SIGN_CHANGE = (v0 > 0.F && v1 <= 0.F) || (v0 < 0.F && v1 >= 0.F);
    if (SIGN_CHANGE) {
      float a = t - STEP;
      float b = t;
      for (int i = 0; i < 24; ++i) {
        float m = 0.5F * (a + b);
        float vm = f(m);
        if ((v0 > 0.F && vm > 0.F) || (v0 < 0.F && vm < 0.F)) {
          a = m;
          v0 = vm;
        } else {
          b = m;
        }
      }
      const float THIT = 0.5F * (a + b);
      outHit = ro + rd * THIT;
      return true;
    }
    t0 = t;
    v0 = v1;
  }
  return false;
}

  // 1) курсор окна
  const glm::dvec2 MOUSE_WIN = ReadMouseWindow(ctx.m_window);

  // 2) окно → панель (логические px)
  glm::vec2 vpLogical{};
  const bool INSIDE = WindowToViewportLogical(ctx, MOUSE_WIN, vpLogical);

  if (!INSIDE) {
    return;
}

  // 3) логические → framebuffer px
  const glm::vec2 PX = ToFramebufferPx(ctx, vpLogical);

  // 4) луч из пикселя FB
  glm::vec3 ro{};
  glm::vec3 rd{};
  BuildScreenRay(ctx, PX, ro, rd);

 // 5) сначала — попытка удаления (ПКМ или Ctrl+ЛКМ)
  if (HandleDeleteClick(ctx, ro, rd))
    return;

  // 6) далее — ИСХОДНАЯ логика: плейсмент только при ЛКМ
  if (!engine::InputSystem::Instance().GetMouseButton(engine::MouseButton::Left,
                                                      engine::InputState::Pressed))
    return;

  // 6a) террейн (если есть) — ставим точно по земле
  glm::vec3 hitTerrain{};
  bool hitT = false;
  float anchorY = 0.0f;
  if (ctx.m_scene->GetTerrain() != nullptr) {
    if (!ctx.State().m_selectedPrefab.empty()) {
      const auto& prefab = ctx.m_scene->GetPrefabSystem().GetPrefab(ctx.State().m_selectedPrefab);
      anchorY = -prefab.m_bounds_min.y * prefab.m_modelScale.y;
    }
    hitT = RaycastTerrain(*ctx.m_scene->GetTerrain(), ro, rd, anchorY, hitTerrain);
  }
  if (hitT) {
    if (ctx.State().m_placeKind == EditorState::PlaceKind::FIRE) {
      ctx.m_scene->AddFireAt(hitTerrain, ctx.State().m_fireIntensity);
    } else {
      ctx.m_scene->AddPrefab(ctx.State().m_selectedPrefab, hitTerrain.x, hitTerrain.z);
    }
    return;
  }

  // 6b) фолбек: Y=0 (как у вас было)
  glm::vec3 hitPlane{};
  if (!RaycastPlaneY0(ro, rd, hitPlane)) {
    return;
}

  if (ctx.State().m_placeKind == EditorState::PlaceKind::FIRE) {
    ctx.m_scene->AddFire(hitPlane.x, hitPlane.z, ctx.State().m_fireIntensity);
  } else {
    ctx.m_scene->AddPrefab(ctx.State().m_selectedPrefab, hitPlane.x, hitPlane.z);
  }
}


bool PlaceOnGroundTool::IsDeleteGesture() noexcept {
  using engine::InputSystem;
  using engine::MouseButton;
  using engine::InputState;
  using engine::KeyCode;

  const bool RMB_PRESSED =
      InputSystem::Instance().GetMouseButton(MouseButton::Right, InputState::Pressed);
  const bool SPACE_LMB_PRESSED =
      InputSystem::Instance().GetKey(KeyCode::Space) &&
      InputSystem::Instance().GetMouseButton(MouseButton::Left, InputState::Pressed);
  return RMB_PRESSED || SPACE_LMB_PRESSED;
}

// ── Обработчик удаления ────────────────────────────
bool PlaceOnGroundTool::HandleDeleteClick(EditorContext& ctx,
                                          const glm::vec3& ro,
                                          const glm::vec3& rd) {
  if (!IsDeleteGesture()) {
    return false;
}

  glm::vec3 hit{};
  bool got = false;

  // Предпочтительно — точка на террейне
  if (ctx.m_scene->GetTerrain() != nullptr) {
    got = RaycastTerrain(*ctx.m_scene->GetTerrain(), ro, rd, /*anchorY=*/0.0f, hit);
  }
  // Фолбек — плоскость Y=0
  if (!got) {
    got = RaycastPlaneY0(ro, rd, hit);
  }
  if (!got) return true; // жест был, но по лучу ничего не попали — считать обработанным

  // Удаляем ближайший объект (или огонь) рядом с hit.xz
  // ВАЖНО: метод сцены должен существовать (см. нашу предыдущую заготовку RemoveNearestAt)
  ctx.m_scene->RemoveNearestAt(hit, kPickRadiusXZ);
  return true;
}

// ui/Editor.h
#pragma once
#include <memory>

#include "ui/EditorState.h"
#include "ui/IModule.h"

namespace ui {

namespace ui {
  struct EditorContext;

class Editor {
public:
  Editor() = default;

  // Запрет копирования
  Editor(const Editor &) = delete;
  Editor &operator=(const Editor &) = delete;

  // Перемещение: переносим владение, но сбрасываем стек (переактивация позже)
  Editor(Editor &&other) noexcept
      : m_panels(std::move(other.m_panels)), m_tools(std::move(other.m_tools)),
        m_modules(std::move(other.m_modules)) {
    m_activeStack.clear();
  }

  Editor &operator=(Editor &&other) noexcept {
    if (this != &other) {
      ShutdownModules(); // на всякий случай корректно выгружаем свои
      m_panels = std::move(other.m_panels);
      m_tools = std::move(other.m_tools);
      m_modules = std::move(other.m_modules);
      m_activeStack.clear();
    }
    return *this;
  }

  ~Editor() {
    // Гарантируем Shutdown модулей перед разрушением редактора
    for (auto &m_module : std::ranges::reverse_view(m_modules)) {
      m_module->Shutdown();
    }
  }

  // --- Регистрация панелей/инструментов (как было) ---
  void RegisterPanel(std::unique_ptr<IPanel> panel) {
    m_panels.emplace_back(std::move(panel));
  }
  void RegisterTool(std::unique_ptr<ITool> tool) {
    m_tools.emplace_back(std::move(tool));
  }

  template <class T> T *GetTool() {
    static_assert(std::is_base_of_v<ITool, T>, "T must be ITool");
    for (auto &up : m_tools) {
      if (auto *tool_pointer = dynamic_cast<T *>(up.get())) {
        return tool_pointer;
      }
    }
    return nullptr;
  }
  template <class T> const T *GetTool() const {
    static_assert(std::is_base_of_v<ITool, T>, "T must be ITool");
    for (const auto &up : m_tools) {
      if (auto *tool_pointer = dynamic_cast<const T *>(up.get())) {
        return tool_pointer;
      }
    }
    return nullptr;
  }

  // --- Модули ---
  void RegisterModule(std::unique_ptr<IModule> module) {
    m_modules.emplace_back(std::move(module));
    m_modules.back()->Startup(*this);
  }

  // При желании — явная остановка модулей до деструктора
  void ShutdownModules() {
    for (auto &m_module : std::ranges::reverse_view(m_modules)) {
      m_module->Shutdown();
    }
  }

  // --- Активные инструменты (как уже сделали) ---
  void SetActiveTool(ToolType type, EditorContext &ctx) {
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnDeactivate(ctx);
      m_activeStack.clear();
    }
    if (auto *tool = FindTool(type)) {
      m_activeStack.push_back(tool);
      tool->OnActivate(ctx);
    }
  }
  void PushTool(ToolType type, EditorContext &ctx) {
    if (auto *tool = FindTool(type)) {
      if (!m_activeStack.empty()) {
        m_activeStack.back()->OnDeactivate(ctx);
      }
      m_activeStack.push_back(tool);
      tool->OnActivate(ctx);
    }
  }
  void PopTool(EditorContext &ctx) {
    if (m_activeStack.empty()) {
      return;
    }
    m_activeStack.back()->OnDeactivate(ctx);
    m_activeStack.pop_back();
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnActivate(ctx);
    }
  }

  void DrawUI(EditorContext &ctx) {
    ctx.m_editor = this;
    for (auto &panel : m_panels) {
      panel->Draw(ctx);
    }
  }
  void TickTools(EditorContext &ctx) {
    if (!m_activeStack.empty()) {
      m_activeStack.back()->OnUpdate(ctx);
    }
  }

  [[nodiscard]] ToolType ActiveToolType() const {
    return m_activeStack.empty() ? ToolType::OTHER_TOOL
                                 : m_activeStack.back()->Type();
  }
  EditorState m_state;

private:
  ITool *FindTool(ToolType type) {
    for (auto &tool : m_tools) {
      if (tool->Type() == type) {
        return tool.get();
      }
    }
    return nullptr;
  }

  std::vector<std::unique_ptr<IPanel>> m_panels;
  std::vector<std::unique_ptr<ITool>> m_tools;
  std::vector<ITool *> m_activeStack;
  std::vector<std::unique_ptr<IModule>> m_modules; // <-- NEW
};

} // namespace ui

===== src/ui/ui_utils/SpatialGrid.h =====
// ui/utils/SpatialGrid.h
#pragma once
#include <cmath>
#include <cstdint>
#include <unordered_map>
#include <vector>

#include <glm/geometric.hpp>  // glm::dot
#include <glm/vec2.hpp>

namespace ui::util {

struct SpatialGrid {
  float cell = 1.0f;  // шаг сетки, обычно >= minSpacing
  std::unordered_map<std::int64_t, std::vector<glm::vec2>> buckets;

  static constexpr std::int64_t Key(int gx, int gy) noexcept {
    return (static_cast<std::int64_t>(gx) << 32) ^ static_cast<std::int64_t>(gy & 0xFFFFFFFF);
  }

  static constexpr std::pair<int, int> ToCell(const glm::vec2& p, float cell) noexcept {
    return {static_cast<int>(std::floor(p.x / cell)), static_cast<int>(std::floor(p.y / cell))};
  }

  void SetCell(float c) noexcept { cell = (c > 0.f ? c : 1.f); }
  void Clear() noexcept { buckets.clear(); }
  void Reserve(std::size_t nCells) { buckets.reserve(nCells); }  // опционально

  // Возвращает true, если точка принята (нет ближних соседей ближе minSpacing)
  bool Accept(const glm::vec2& p, float minSpacing) {
    const auto [cx, cy] = ToCell(p, cell);
    const float r2 = minSpacing * minSpacing;

    for (int dy = -1; dy <= 1; ++dy) {
      for (int dx = -1; dx <= 1; ++dx) {
        const auto it = buckets.find(Key(cx + dx, cy + dy));
        if (it == buckets.end())
          continue;
        for (const glm::vec2& q : it->second) {
          const glm::vec2 d = p - q;
          if (glm::dot(d, d) < r2)
            return false;
        }
      }
    }
    buckets[Key(cx, cy)].emplace_back(p);
    return true;
  }
};

}  // namespace ui::util

using engine::InputState;

static float ComputeAnchorY(const ui::EditorContext& ctx) {
  float anchorY = 0.f;
  if ((ctx.m_scene != nullptr) && !ctx.State().m_selectedPrefab.empty()) {
    const auto& id = ctx.State().m_selectedPrefab;
    const auto& P = ctx.m_scene->GetPrefabSystem().GetPrefab(id);
    // поднимаем “центр” на высоту нижней грани bbox * scale
    anchorY = -P.m_bounds_min.y * P.m_modelScale.y;
  }
  return anchorY;
}

static bool RaycastTerrain(const engine::Terrain& T, const glm::vec3& ro, const glm::vec3& rd,
                           float anchorY, glm::vec3& outHit) {
  auto f = [&](float t) {
    glm::vec3 p = ro + rd * t;
    return p.y - (T.GetHeightAt(p.x, p.z) + anchorY);
  };
  const float T_MAX = 5000.F;
  const float STEP = 2.F;
  float t0 = 0.F;
  float v0 = f(t0);
  for (float t = STEP; t <= T_MAX; t += STEP) {
    float v1 = f(t);
    if ((v0 > 0.F && v1 <= 0.F) || (v0 < 0.F && v1 >= 0.F)) {
      float a = t - STEP;
      float b = t;
      for (int i = 0; i < 24; ++i) {
        float m = 0.5F * (a + b);
        float vm = f(m);
        if ((v0 > 0.f && vm > 0.f) || (v0 < 0.f && vm < 0.f)) {
          a = m;
          v0 = vm;
        } else {
          b = m;
        }
      }
      float thit = 0.5f * (a + b);
      outHit = ro + rd * thit;
      return true;
    }
    t0 = t;
    v0 = v1;
  }
  return false;
}

  m_preview.clear();
  m_dragging = false;
  EnsureSeed();

void FoliageBrushTool::EnsureSeed() {
  m_rng.seed(m_settings.seed);

bool FoliageBrushTool::ScreenToGround(EditorContext& ctx, float mx, float my, glm::vec3& outHit) {
  // как в PlaceOnGroundTool: окно → viewport → луч → пересечение Y=0
  double dx = 0.0;
  double dy = 0.0;
  dx = mx;
  dy = my;

  const glm::vec2 vp = ctx.WindowToViewportFB(static_cast<float>(dx), static_cast<float>(dy));
  if (vp.x < 0.f || vp.y < 0.f || vp.x >= (float)ctx.vpW || vp.y >= (float)ctx.vpH) {
    return false;
  }

  const float x_ndc = ((vp.x / static_cast<float>(ctx.vpW)) * 2.F) - 1.F;
  const float y_ndc = 1.f - (vp.y / (float)ctx.vpH) * 2.f;

  const glm::mat4& V = ctx.m_view;
  const glm::mat4& P = ctx.m_proj;
  const glm::mat4 invVP = glm::inverse(P * V);

  glm::vec4 p0 = invVP * glm::vec4(x_ndc, y_ndc, -1.f, 1.f);
  glm::vec4 p1 = invVP * glm::vec4(x_ndc, y_ndc, 1.f, 1.f);
  p0 /= p0.w;
  p1 /= p1.w;

  const glm::vec3 ro = glm::vec3(p0);
  const glm::vec3 rd = glm::normalize(glm::vec3(p1) - ro);

  if (std::abs(rd.y) < 1e-6f)
    return false;
  const float t = -ro.y / rd.y;
  if (t < 0.f)
    return false;

  outHit = ro + rd * t;
  return true;
}

void FoliageBrushTool::OnUpdate(EditorContext& ctx) {
  const bool uiCaptured = engine::InputSystem::Instance().UIMouseCaptured();

  // --- обрабатываем ввод только когда мышь не захвачена UI ---
  // одиночный клик — назначаем центр
  if (!uiCaptured) {
    if (engine::InputSystem::Instance().GetMouseButton(MouseButton::Left, InputState::Pressed)) {
      double mx;
      double my;
      glfwGetCursorPos(ctx.m_window, &mx, &my);

      // строим луч так же, как сейчас
      const glm::vec2 fb = ctx.WindowToViewportFB((float)mx, (float)my);
      glm::vec3 ro;
      glm::vec3 rd;
      math::ScreenPointToRay(fb.x, fb.y, (float)ctx.m_viewportW, (float)ctx.m_viewportH, ctx.m_proj,
                             ctx.m_view, ro, rd);

      glm::vec3 hit{};
      bool ok = false;
      const auto* T = ctx.m_scene ? ctx.m_scene->GetTerrain() : nullptr;
      const float anchorY = ComputeAnchorY(ctx);

      if (T)
        ok = RaycastTerrain(*T, ro, rd, anchorY, hit);
      if (!ok)
        ok = math::IntersectRayPlaneY0(ro, rd, hit);  // фолбек

      if (ok) {
        m_center = hit;
        m_center.y = T ? T->GetHeightAt(hit.x, hit.z) + anchorY : 0.f;
        m_hasCenter = true;
        // пересобираем предпросмотр как и раньше
        if (m_settings.shape == Shape::Circle) {
          m_anchor = m_center;
          m_current = m_center + glm::vec3(m_settings.radius, 0, 0);
          RebuildPreviewCircle();
        } else {
          m_anchor = m_center - glm::vec3(m_settings.width * 0.5f, 0, m_settings.height * 0.5f);
          m_current = m_center + glm::vec3(m_settings.width * 0.5f, 0, m_settings.height * 0.5f);
          RebuildPreviewRect();
        }
      }
    }

    // хоткеи
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::Enter, InputState::Pressed)) {
      ApplyPlacement(*ctx.m_scene);
    }
    if (engine::InputSystem::Instance().GetKey(engine::KeyCode::Escape, InputState::Pressed)) {
      m_preview.clear();
      m_hasCenter = false;
    }
  }

  // --- оверлей (контур + точки) рисуем всегда ---
  DrawOverlay(ctx);
}

// ——— генераторы предпросмотра ———

void FoliageBrushTool::RebuildPreviewCircle() {
  m_preview.clear();
  const glm::vec2 c = {m_anchor.x, m_anchor.z};
  const glm::vec2 q = {m_current.x, m_current.z};
  const float R = (m_dragging ? glm::length(q - c) : m_settings.radius);

  const float area = std::numbers::pi_v<float> * R * R;
  const float expectedN = area * m_settings.density;

  const float cell = std::max(m_settings.minSpacing, 0.5f);
  std::uniform_real_distribution<float> uni(-0.5f, 0.5f);

  Grid grid;
  grid.SetCell(cell);
  m_preview.reserve(static_cast<size_t>(std::max(0.f, expectedN)));

  for (float x = -R; x <= R; x += cell) {
    for (float y = -R; y <= R; y += cell) {
      if (x * x + y * y > R * R)
        continue;
      glm::vec2 p = {c.x + x, c.y + y};
      p += glm::vec2(uni(m_rng), uni(m_rng)) * (cell * m_settings.jitter);
      if (m_settings.minSpacing <= 0.0f || grid.Accept(p, m_settings.minSpacing)) {
        m_preview.emplace_back(p.x, 0.f, p.y);
      }
    }
  }

  if (expectedN > 1 && m_preview.size() > static_cast<size_t>(expectedN)) {
    m_preview.resize(static_cast<size_t>(expectedN));
  }

  // (опционально)
  constexpr size_t kMaxPreviewPoints = 120'000;
  if (m_preview.size() > kMaxPreviewPoints)
    m_preview.resize(kMaxPreviewPoints);
}

void FoliageBrushTool::RebuildPreviewRect() {
  m_preview.clear();
  const glm::vec2 a = {m_anchor.x, m_anchor.z};
  const glm::vec2 b = {m_current.x, m_current.z};
  const glm::vec2 minp = {std::min(a.x, b.x), std::min(a.y, b.y)};
  const glm::vec2 maxp = {std::max(a.x, b.x), std::max(a.y, b.y)};
  const glm::vec2 size = maxp - minp;

  const float area = size.x * size.y;
  const float expectedN = area * m_settings.density;

  const float cell = std::max(m_settings.minSpacing, 0.5f);
  std::uniform_real_distribution<float> uni(-0.5f, 0.5f);

  Grid grid;
  grid.SetCell(cell);
  m_preview.reserve(static_cast<size_t>(std::max(0.f, expectedN)));

  for (float x = minp.x; x <= maxp.x; x += cell) {
    for (float y = minp.y; y <= maxp.y; y += cell) {
      glm::vec2 p = {x, y};
      p += glm::vec2(uni(m_rng), uni(m_rng)) * (cell * m_settings.jitter);
      if (m_settings.minSpacing <= 0.0f || grid.Accept(p, m_settings.minSpacing)) {
        m_preview.emplace_back(p.x, 0.f, p.y);
      }
    }
  }

  if (expectedN > 1 && m_preview.size() > static_cast<size_t>(expectedN)) {
    m_preview.resize(static_cast<size_t>(expectedN));
  }

  // (опционально)
  constexpr size_t kMaxPreviewPoints = 120'000;
  if (m_preview.size() > kMaxPreviewPoints)
    m_preview.resize(kMaxPreviewPoints);
}

static bool IsFireName(std::string_view id) {
  // временная эвристика — до появления нормального category/flags в префабе
  return id.find("fire") != std::string_view::npos || id.find("flame") != std::string_view::npos;
}
// ——— размещение ———

void FoliageBrushTool::ApplyPlacement(engine::Scene& scene) {
  if (m_preview.empty())
    return;
  if (IsFireName(m_settings.prefab)) {
    m_preview.clear();
    return;  // тихо выходим (можно логнуть/мигнуть тултипом)
  }

  std::uniform_real_distribution<float> unif01(0.f, 1.f);
  const float smin = m_settings.scaleMin, smax = m_settings.scaleMax;
  const float yawMin = glm::radians(m_settings.yawMinDeg);
  const float yawMax = glm::radians(m_settings.yawMaxDeg);

  for (auto& p : m_preview) {
    // scale (для «разной высоты» используешь масштаб по Y, но я даю униформ по всем осям)
    const float t = unif01(m_rng);
    const float s = smin + (smax - smin) * t;

    // поворот по Y (если модели поддерживают, можно хранить как часть SceneObject при расширении)
    (void)yawMin;
    (void)yawMax;  // пока не сохраняем углы — при надобности расширь SceneObject

    // вызываем твой пайплайн:
    // сейчас Scene::AddTree(name=default_tree) создаёт Agent и SceneObject, скейл берётся из префаба
    // если хочешь именно свой s — либо расширь AddTree/SceneObject, либо применяй prefab.model_scale*s
    scene.AddTree(p.x, p.z, m_settings.prefab);
  }

  m_preview.clear();
}

// FoliageBrushTool.cc
bool FoliageBrushTool::ProjectToScreen(const EditorContext& ctx, const glm::vec3& p,
                                       glm::vec2& out_px) {
  const glm::vec4 clip = ctx.m_proj * ctx.m_view * glm::vec4(p, 1.0f);
  if (clip.w <= 0.0f)
    return false;
  const glm::vec3 ndc = glm::vec3(clip) / clip.w;  // [-1..1]
  if (ndc.x < -1.f || ndc.x > 1.f || ndc.y < -1.f || ndc.y > 1.f)
    return false;
  // в пиксели framebuffer
  const float x = (ndc.x * 0.5f + 0.5f) * (float)ctx.vpW;
  const float y = (1.f - (ndc.y * 0.5f + 0.5f)) * (float)ctx.vpH;
  out_px = {x, y};
  return true;
}

void FoliageBrushTool::DrawOverlay(EditorContext& ctx) {
  auto* dl = ImGui::GetBackgroundDrawList();  // рисуем под UI (можно Foreground)
  const ImU32 colOutline = IM_COL32(80, 200, 120, 200);
  const ImU32 colPoints = IM_COL32(80, 200, 120, 180);

  const auto* T = ctx.m_scene ? ctx.m_scene->GetTerrain() : nullptr;
  const float anchorY = ComputeAnchorY(ctx);
  auto lift = [&](glm::vec3 p) {
    if (T)
      p.y = T->GetHeightAt(p.x, p.z) + anchorY;
    return p;
  };
  // контур области
  std::vector<ImVec2> poly;
  poly.reserve(128);

  if (m_hasCenter) {
    if (m_settings.shape == Shape::Circle) {
      const float R =
          (m_dragging ? glm::length(glm::vec2(m_current.x - m_anchor.x, m_current.z - m_anchor.z))
                      : m_settings.radius);
      const int SEG = 64;
      for (int i = 0; i < SEG; i++) {
        float a = (float)i / SEG * 2.f * std::numbers::pi_v<float>;
        glm::vec3 pw = m_center + glm::vec3(std::cos(a) * R, 0.f, std::sin(a) * R);
        pw = lift(pw);
        glm::vec2 ps;
        if (ProjectToScreen(ctx, pw, ps))
          poly.emplace_back(ps.x, ps.y);
      }
    } else {  // Rect — тесселированная рамка по рельефу
      const float w = m_settings.width;
      const float h = m_settings.height;
      const glm::vec3 c = m_center;  // y не важен — lift() выставит

      // сколько сегментов на метр (подбери; 0.5…2 — ок)
      const float segPerMeter = 1.0f;
      const int segX = std::max(4, (int)std::ceil(w * segPerMeter));
      const int segZ = std::max(4, (int)std::ceil(h * segPerMeter));

      auto push_proj = [&](const glm::vec3& pw) {
        glm::vec2 s;
        if (ProjectToScreen(ctx, pw, s))
          poly.emplace_back(s.x, s.y);
      };

      poly.clear();
      poly.reserve(2 * (segX + segZ) * 2);

      // Левая кромка: (-w/2, -h/2) → (-w/2, +h/2)
      for (int i = 0; i <= segZ; ++i) {
        float t = (float)i / segZ;
        glm::vec3 p = c + glm::vec3(-w * 0.5f, 0.f, -h * 0.5f + t * h);
        push_proj(lift(p));
      }
      // Верхняя: (-w/2, +h/2) → (+w/2, +h/2)
      for (int i = 1; i <= segX; ++i) {
        float t = (float)i / segX;
        glm::vec3 p = c + glm::vec3(-w * 0.5f + t * w, 0.f, +h * 0.5f);
        push_proj(lift(p));
      }
      // Правая: (+w/2, +h/2) → (+w/2, -h/2)
      for (int i = 1; i <= segZ; ++i) {
        float t = (float)i / segZ;
        glm::vec3 p = c + glm::vec3(+w * 0.5f, 0.f, +h * 0.5f - t * h);
        push_proj(lift(p));
      }
      // Нижняя: (+w/2, -h/2) → (-w/2, -h/2)
      for (int i = 1; i <= segX; ++i) {
        float t = (float)i / segX;
        glm::vec3 p = c + glm::vec3(+w * 0.5f - t * w, 0.f, -h * 0.5f);
        push_proj(lift(p));
      }
    }
  }

if (poly.size() >= 2) {
  const bool isFullCircle = (m_settings.shape == Shape::Circle && poly.size() == 64);
  dl->AddPolyline(poly.data(),
                  (int)poly.size(),
                  colOutline,
                  isFullCircle ? ImDrawFlags_Closed : 0,
                  2.0f);
}

  // точки предпросмотра
  // при очень больших количествах точек можно прореживать stride'ом
  const size_t stride = (m_preview.size() > 20000 ? 3 : 1);
  for (size_t i = 0; i < m_preview.size(); i += stride) {
    const auto& p = m_preview[i];
    glm::vec3 pw = lift(p);
    glm::vec2 px;
    if (ProjectToScreen(ctx, pw, px)) {
      dl->AddCircleFilled(ImVec2(px.x, px.y), 2.0f, colPoints, 8);
    }
  }
}

// ui/tools/FoliageBrushTool.h
#pragma once
#include "ui/ITool.h"
#include <string>
#include <random>
#include <vector>
#include <glm/vec3.hpp>

namespace engine { class Scene; }
namespace ui {
class EditorContext;

class FoliageBrushTool final : public ITool {
public:
  enum class Shape { Circle, Rect };

  struct Settings {
    Shape shape = Shape::Circle;
    float radius = 10.f;        // для круга
    float width  = 20.f;        // для прямоугольника
    float height = 20.f;        // для прямоугольника
    float density = 0.1f;       // объектов на м² (например)
    float minSpacing = 1.0f;    // минимальная дистанция между точками
    float jitter = 0.5f;        // доля случайного смещения в ячейке
    unsigned seed = 12345;

    // визуальное разнообразие
    float scaleMin = 0.9f;
    float scaleMax = 1.3f;
    float yawMinDeg = 0.f;
    float yawMaxDeg = 360.f;

    // какой префаб/тип дерева
    std::string prefab = "default_tree";
  };

  explicit FoliageBrushTool(std::string_view /*unused*/) {}

  [[nodiscard]] ToolType Type() const noexcept override { return ToolType::OTHER_TOOL; }
  [[nodiscard]] const char* Name() const noexcept { return "Foliage Brush"; }
  [[nodiscard]] const char* Id()   const noexcept { return "Tool.FoliageBrush"; }

  void OnActivate(EditorContext& ctx) override;
  void OnDeactivate(EditorContext& /*ctx*/) override { m_preview.clear(); m_dragging = false; }
  void OnUpdate(EditorContext& ctx) override;

  // Контролы инструмента (без Begin/End)
  void DrawControls(EditorContext& ctx);

private:
  // выбор области
  bool m_dragging = false;
  glm::vec3 m_anchor{};     // первая точка (мир)
  glm::vec3 m_current{};    // текущая точка (мир)
  std::vector<glm::vec3> m_preview; // точки предпросмотра (мир)

  Settings m_settings{};
  std::mt19937 m_rng{m_settings.seed};

  // генераторы
  void RebuildPreviewCircle();
  void RebuildPreviewRect();

                                4.0f, "%.2f");
  ImGui::SeparatorText("Position");

  bool pos_changed = false;
  pos_changed |=
      ImGui::SliderFloat("X", &m_settings.position.x, -500.0f, 500.0f, "%.1f");
  pos_changed |=
      ImGui::SliderFloat("Y", &m_settings.position.y, -200.0f, 200.0f, "%.1f");
  pos_changed |=

void FireParamsPanel::LoadAndPreviewTexture(const std::string& path, int textureIndex) {
  try {
    if (std::filesystem::exists(path)) {
      m_texturePreviews[textureIndex] = engine::Texture::Create(path);
      m_textureLoaded[textureIndex] = true;
      strncpy(m_texturePaths[textureIndex], path.c_str(), sizeof(m_texturePaths[textureIndex])-1);
      std::cout << "Texture preview loaded: " << path << std::endl;
    } else {
      m_textureLoaded[textureIndex] = false;
      std::cout << "Texture file not found: " << path << std::endl;
    }
  }
  catch (const std::exception& e) {
    m_textureLoaded[textureIndex] = false;
    std::cerr << "Failed to load texture: " << e.what() << std::endl;
  }
}

void FireParamsPanel::DrawTexturePreview(int textureIndex, const char* name) {
  if (!m_textureLoaded[textureIndex] || !m_texturePreviews[textureIndex]) {
    ImGui::TextColored(ImVec4(1.0f, 0.4f, 0.4f, 1.0f), "%s texture not loaded", name);
    return;
  }
  
  auto texture_id = m_texturePreviews[textureIndex]->GetId();
  
  if (texture_id) {
    float preview_size = 100.0f;
    
    ImGui::BeginChild((std::string("TexturePreview") + name).c_str(), 
                     ImVec2(preview_size + 10, preview_size + 30), true);
    {
      ImGui::Image((ImTextureID)(intptr_t)texture_id, ImVec2(preview_size, preview_size));
      ImGui::Text("Size: %dx%d", m_texturePreviews[textureIndex]->GetWidth(), 
                  m_texturePreviews[textureIndex]->GetHeight());
    }
    ImGui::EndChild();
  }
}


void FireParamsPanel::DrawPresetsSection() {
  ImGui::SeparatorText("Presets");
  
  // Создание директории для пресетов, если её нет
  if (!std::filesystem::exists(m_presetsDirectory)) {
    std::filesystem::create_directory(m_presetsDirectory);
  }
  
  // Кнопка обновления списка пресетов
  if (ImGui::Button("Refresh Presets")) {
    RefreshPresetsList();
  }
  ImGui::SameLine();
  
  // Сохранение пресета
  ImGui::InputText("##NewPresetName", m_newPresetName, sizeof(m_newPresetName));
  ImGui::SameLine();
  if (ImGui::Button("Save Preset") && strlen(m_newPresetName) > 0) {
    SavePreset(m_newPresetName);
    m_newPresetName[0] = '\0';
    RefreshPresetsList();
  }
  
  // Список пресетов
  if (!m_presets.empty()) {
    ImGui::Text("Available Presets:");
    ImGui::BeginChild("PresetsList", ImVec2(0, 150), true);
    
    for (const auto& [name, preset] : m_presets) {
      ImGui::PushID(name.c_str());
      
      bool isCurrent = (m_currentPresetName == name);
      if (ImGui::Button("Load")) {
        LoadPreset(name);
      }
      ImGui::SameLine();
      
      if (ImGui::Button("Delete")) {
        DeletePreset(name);
        ImGui::PopID();
        break;
      }
      ImGui::SameLine();
      
      if (isCurrent) {
        ImGui::TextColored(ImVec4(0.0f, 1.0f, 0.0f, 1.0f), "%s", name.c_str());
      } else {
        ImGui::Text("%s", name.c_str());
      }
      
      ImGui::PopID();
    }
    
    ImGui::EndChild();
  } else {
    ImGui::TextDisabled("No presets found. Create one or refresh the list.");
  }
}

void FireParamsPanel::SavePreset(const std::string& name) {
  Preset preset;
  preset.name = name;
  preset.params = m_params;
  
  SavePresetToYAML(preset);
  m_currentPresetName = name;
  
  std::cout << "Preset saved: " << name << std::endl;
}

void FireParamsPanel::LoadPreset(const std::string& name) {
  std::string filename = m_presetsDirectory + "/" + name + ".yaml";
  Preset preset;
  
  if (LoadPresetFromYAML(filename, preset)) {
    m_params = preset.params;
    m_currentPresetName = name;
    
    // Обновляем пути текстур в UI
    strncpy(m_texturePaths[0], m_params.fireTexturePath.c_str(), sizeof(m_texturePaths[0])-1);
    strncpy(m_texturePaths[1], m_params.smokeTexturePath.c_str(), sizeof(m_texturePaths[1])-1);
    strncpy(m_texturePaths[2], m_params.sparksTexturePath.c_str(), sizeof(m_texturePaths[2])-1);
    strncpy(m_texturePaths[3], m_params.ashTexturePath.c_str(), sizeof(m_texturePaths[3])-1);
    strncpy(m_texturePaths[4], m_params.heatTexturePath.c_str(), sizeof(m_texturePaths[4])-1);
    
    // Перезагружаем preview текстур
    for (int i = 0; i < 5; i++) {
      LoadAndPreviewTexture(m_texturePaths[i], i);
    }
    
    ApplyParamsToShader();
    
    std::cout << "Preset loaded: " << name << std::endl;
  }
}

void FireParamsPanel::DeletePreset(const std::string& name) {
  std::string filename = m_presetsDirectory + "/" + name + ".yaml";
  if (std::filesystem::exists(filename)) {
    std::filesystem::remove(filename);
    m_presets.erase(name);
    if (m_currentPresetName == name) {
      m_currentPresetName.clear();
    }
    std::cout << "Preset deleted: " << name << std::endl;
  }
}

void FireParamsPanel::RefreshPresetsList() {
  m_presets.clear();
  
  if (!std::filesystem::exists(m_presetsDirectory)) {
    return;
  }
  
  for (const auto& entry : std::filesystem::directory_iterator(m_presetsDirectory)) {
    if (entry.path().extension() == ".yaml") {
      std::string filename = entry.path().stem().string();
      Preset preset;
      
      if (LoadPresetFromYAML(entry.path().string(), preset)) {
        m_presets[filename] = preset;
      }
    }
  }
  
  std::cout << "Refreshed presets list. Found: " << m_presets.size() << " presets" << std::endl;
}

void FireParamsPanel::SavePresetToYAML(const Preset& preset) {
  YAML::Emitter out;
  
  out << YAML::BeginMap;
  out << YAML::Key << "name" << YAML::Value << preset.name;
  
  // Global settings
  // Texture paths
  out << YAML::Key << "fireTexturePath" << YAML::Value << preset.params.fireTexturePath;
  out << YAML::Key << "smokeTexturePath" << YAML::Value << preset.params.smokeTexturePath;
  out << YAML::Key << "sparksTexturePath" << YAML::Value << preset.params.sparksTexturePath;
  out << YAML::Key << "ashTexturePath" << YAML::Value << preset.params.ashTexturePath;
  out << YAML::Key << "heatTexturePath" << YAML::Value << preset.params.heatTexturePath;
  
  // Effect parameters
  auto saveEffectParams = [&](const std::string& effectName, const FireParams::EffectParams& params) {
    out << YAML::Key << effectName + "_enabled" << YAML::Value << params.enabled;
    out << YAML::Key << effectName + "_particleSize" << YAML::Value << params.particleSize;
    out << YAML::Key << effectName + "_sizeVariation" << YAML::Value << params.sizeVariation;
    out << YAML::Key << effectName + "_animationSpeed" << YAML::Value << params.animationSpeed;
    out << YAML::Key << effectName + "_lifetime" << YAML::Value << params.lifetime;
    out << YAML::Key << effectName + "_spawnRate" << YAML::Value << params.spawnRate;
    out << YAML::Key << effectName + "_riseSpeed" << YAML::Value << params.riseSpeed;
    out << YAML::Key << effectName + "_turbulence" << YAML::Value << params.turbulence;
    out << YAML::Key << effectName + "_windInfluence" << YAML::Value << params.windInfluence;
    out << YAML::Key << effectName + "_spread" << YAML::Value << params.spread;
    out << YAML::Key << effectName + "_colorVariation" << YAML::Value << params.colorVariation;

    out << YAML::Key << effectName + "_redIntensity" << YAML::Value << params.redIntensity;
    out << YAML::Key << effectName + "_yellowReduction" << YAML::Value << params.yellowReduction;
    out << YAML::Key << effectName + "_alphaMultiplier" << YAML::Value << params.alphaMultiplier;
    out << YAML::Key << effectName + "_textureInfluence" << YAML::Value << params.textureInfluence;
    
    out << YAML::Key << effectName + "_baseColor" << YAML::Value << YAML::Flow << YAML::BeginSeq
        << params.baseColor[0] << params.baseColor[1] << params.baseColor[2] << YAML::EndSeq;
    out << YAML::Key << effectName + "_endColor" << YAML::Value << YAML::Flow << YAML::BeginSeq
        << params.endColor[0] << params.endColor[1] << params.endColor[2] << YAML::EndSeq;
  };
  
  saveEffectParams("fire", preset.params.fire);
  saveEffectParams("smoke", preset.params.smoke);
  saveEffectParams("sparks", preset.params.sparks);
  saveEffectParams("ash", preset.params.ash);
  saveEffectParams("heat", preset.params.heat);
  
  out << YAML::EndMap;
  
  std::string filename = m_presetsDirectory + "/" + preset.name + ".yaml";
  std::ofstream file(filename);
  file << out.c_str();
}

bool FireParamsPanel::LoadPresetFromYAML(const std::string& filename, Preset& preset) {
  try {
    YAML::Node config = YAML::LoadFile(filename);
    
    preset.name = config["name"].as<std::string>();
    
    // Global settings
    
    // Texture paths
    if (config["fireTexturePath"]) preset.params.fireTexturePath = config["fireTexturePath"].as<std::string>();
    if (config["smokeTexturePath"]) preset.params.smokeTexturePath = config["smokeTexturePath"].as<std::string>();
    if (config["sparksTexturePath"]) preset.params.sparksTexturePath = config["sparksTexturePath"].as<std::string>();
    if (config["ashTexturePath"]) preset.params.ashTexturePath = config["ashTexturePath"].as<std::string>();
    if (config["heatTexturePath"]) preset.params.heatTexturePath = config["heatTexturePath"].as<std::string>();
    
    // Effect parameters loader
    auto loadEffectParams = [&](const std::string& effectName, FireParams::EffectParams& params) {
      if (config[effectName + "_enabled"]) params.enabled = config[effectName + "_enabled"].as<bool>();
      if (config[effectName + "_particleSize"]) params.particleSize = config[effectName + "_particleSize"].as<float>();
      if (config[effectName + "_sizeVariation"]) params.sizeVariation = config[effectName + "_sizeVariation"].as<float>();
      if (config[effectName + "_animationSpeed"]) params.animationSpeed = config[effectName + "_animationSpeed"].as<float>();
      if (config[effectName + "_lifetime"]) params.lifetime = config[effectName + "_lifetime"].as<float>();
      if (config[effectName + "_spawnRate"]) params.spawnRate = config[effectName + "_spawnRate"].as<float>();
      if (config[effectName + "_riseSpeed"]) params.riseSpeed = config[effectName + "_riseSpeed"].as<float>();
      if (config[effectName + "_turbulence"]) params.turbulence = config[effectName + "_turbulence"].as<float>();
      if (config[effectName + "_windInfluence"]) params.windInfluence = config[effectName + "_windInfluence"].as<float>();
      if (config[effectName + "_spread"]) params.spread = config[effectName + "_spread"].as<float>();
      if (config[effectName + "_colorVariation"]) params.colorVariation = config[effectName + "_colorVariation"].as<float>();
           
      if (config[effectName + "_redIntensity"]) params.redIntensity = config[effectName + "_redIntensity"].as<float>();
      if (config[effectName + "_yellowReduction"]) params.yellowReduction = config[effectName + "_yellowReduction"].as<float>();
      if (config[effectName + "_alphaMultiplier"]) params.alphaMultiplier = config[effectName + "_alphaMultiplier"].as<float>();
      if (config[effectName + "_textureInfluence"]) params.textureInfluence = config[effectName + "_textureInfluence"].as<float>();
      if (config[effectName + "_baseColor"]) {
        auto baseColor = config[effectName + "_baseColor"].as<std::vector<float>>();
        for (int i = 0; i < 3 && i < baseColor.size(); i++) {
          params.baseColor[i] = baseColor[i];
        }
      }
      
      if (config[effectName + "_endColor"]) {
        auto endColor = config[effectName + "_endColor"].as<std::vector<float>>();
        for (int i = 0; i < 3 && i < endColor.size(); i++) {
          params.endColor[i] = endColor[i];
        }
      }
    };
    
    loadEffectParams("fire", preset.params.fire);
    loadEffectParams("smoke", preset.params.smoke);
    loadEffectParams("sparks", preset.params.sparks);
    loadEffectParams("ash", preset.params.ash);
    loadEffectParams("heat", preset.params.heat);
    
    return true;
  }
  catch (const std::exception& e) {
    std::cerr << "Error loading preset from " << filename << ": " << e.what() << std::endl;
    return false;
  }
}

void ToolbarPanel::Draw(EditorContext& ctx) {
  // 1) режимы (Prefab / Fire)
  DrawModeSwitch(ctx);

  ImGui::Separator();

  // 2) селектор инструментов (PlaceOnGround / FoliageBrush)
  DrawToolSelector(ctx);

  ImGui::Separator();

  // 3) панель активного инструмента (например, настройки кисти)
  DrawActiveToolUI(ctx);
}

void ToolbarPanel::DrawModeSwitch(EditorContext& ctx) {
  auto& st = ctx.State();

  bool isPrefab = (st.m_placeKind == EditorState::PlaceKind::PREFAB);
  if (ImGui::RadioButton("Prefab##mode", isPrefab)) {
    st.m_placeKind = EditorState::PlaceKind::PREFAB;
  }
  ImGui::SameLine();

  bool isFire = (st.m_placeKind == EditorState::PlaceKind::FIRE);
  if (ImGui::RadioButton("Fire##mode", isFire)) {
    st.m_placeKind = EditorState::PlaceKind::FIRE;
    if (ctx.m_editor != nullptr) { ctx.m_editor->SetActiveTool(ui::ToolType::PLACE_ON_GROUND, ctx);
}
  }

  // огонь — его параметры сразу тут
  if (st.m_placeKind == EditorState::PlaceKind::FIRE) {
    ImGui::SliderFloat("Fire intensity", &st.m_fireIntensity, 0.f, 10.f);
    if (ctx.m_scene) ctx.m_scene->SetFireIntensity(st.m_fireIntensity);
    // при необходимости: кнопки запуска/остановки/сброса огня
  }

  // префабы — выбор актуального префаба
  if (st.m_placeKind == EditorState::PlaceKind::PREFAB) {
    if (ctx.m_scene) {
      const auto& ids = ctx.m_scene->ListPrefabIds();
      if (!ids.empty()) {
        if (std::ranges::find(ids, st.m_selectedPrefab) == ids.end()) {
          st.m_selectedPrefab = ids.front();
        }
        if (ImGui::BeginCombo("Prefab##combo", st.m_selectedPrefab.c_str())) {
          for (const auto& id : ids) {
            bool sel = (id == st.m_selectedPrefab);
            if (ImGui::Selectable(id.c_str(), sel)) st.m_selectedPrefab = id;
            if (sel) ImGui::SetItemDefaultFocus();
          }
          ImGui::EndCombo();
        }
      } else {
        ImGui::TextDisabled("no prefabs loaded");
      }
    }
  }
}

void ToolbarPanel::DrawToolSelector(EditorContext& ctx) {
  // тут просто переключаем активный тул
  const auto active = ctx.GetActiveToolType();

  if (ImGui::Selectable("Place on ground", active == ToolType::PLACE_ON_GROUND)) {
    if (ctx.m_editor) ctx.m_editor->SetActiveTool(ToolType::PLACE_ON_GROUND, ctx);
  }

  if (ImGui::Selectable("Foliage Brush", active == ToolType::OTHER_TOOL)) { // пока OTHER_TOOL
    if (ctx.m_editor) ctx.m_editor->SetActiveTool(ToolType::OTHER_TOOL, ctx);
  }
}

void ToolbarPanel::DrawActiveToolUI(EditorContext& ctx) {
  const auto active = ctx.GetActiveToolType();
  constexpr ImGuiWindowFlags kWinFlags =
      ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoSavedSettings;

  switch (active) {
    case ToolType::PLACE_ON_GROUND: {
      bool open = true; // без крестика можно передать nullptr
      if (ImGui::Begin("Place on ground", &open, kWinFlags)) {
        if (auto* t = ctx.m_editor ? ctx.m_editor->GetTool<ui::PlaceOnGroundTool>() : nullptr) {
          // если нет контролов — можно текст “no settings”
          // t->DrawControls(ctx);
          ImGui::TextDisabled("No settings.");
        }
      }
      ImGui::End();
      break;
    }
    case ToolType::OTHER_TOOL: { // Foliage Brush
      bool open = true;
      if (ImGui::Begin("Foliage Brush", &open, kWinFlags)) {
        if (auto* t = ctx.m_editor ? ctx.m_editor->GetTool<FoliageBrushTool>() : nullptr) {
          t->DrawControls(ctx); // ВАЖНО: без Begin/End внутри
        }
      }
      ImGui::End();
      break;
    }
    default: break;
  }
}

inline bool DragVec3(const char *label, glm::vec3 &v, float speed = 0.1F,
                     float v_min = -FLT_MAX, float v_max = FLT_MAX,
                     const char *fmt = "%.3f") {
  float a[3] = {v.x, v.y, v.z};
  const bool CHANGED = ImGui::DragFloat3(label, a, speed, v_min, v_max, fmt);
  if (CHANGED) {
    v.x = a[0];
    v.y = a[1];
    v.z = a[2];
  }
  return CHANGED;
}

void PrefabBrowserPanel::DrawModelPreview(EditorContext &ctx) {
  AgentPrefab &P = *m_edit;

  if (ImGui::IsWindowHovered() && ImGui::IsMouseDown(ImGuiMouseButton_Left)) {
    ImVec2 delta = ImGui::GetMouseDragDelta(0, 0.0f);
    if (!m_dragging) {
      m_lastMouse = ImGui::GetMousePos();
      m_dragging = true;
    }
    ImVec2 d = ImVec2(ImGui::GetMousePos().x - m_lastMouse.x,
                      ImGui::GetMousePos().y - m_lastMouse.y);
    m_preview->Rotate(d.x * 0.2f, d.y * 0.2f);
    m_lastMouse = ImGui::GetMousePos();
  } else {
    m_dragging = false;
  }

  if (ImGui::IsWindowHovered()) {
    float wheel = ImGui::GetIO().MouseWheel;
    if (wheel != 0) {
      m_preview->Zoom(wheel * 2.0f);
      ImGui::GetIO().MouseWheel = 0;
    }
  }

  m_preview->Render(P);

  ImGui::Image((ImTextureID)(intptr_t)m_preview->GetTexture(), ImVec2(300, 300),
               ImVec2(0, 1), ImVec2(1, 0));
}

const std::string &
PrefabBrowserPanel::EnsureSelection(EditorContext &ctx,
                                    const std::vector<std::string> &ids) {
  if (m_selectedIndex < 0 || m_selectedIndex >= (int)ids.size()) {
    m_selectedIndex = 0;
    m_edit.reset();
  }
  const std::string &id = ids[m_selectedIndex];
  if (!m_edit.has_value()) {
    m_edit = ctx.m_scene->GetPrefabSystem().GetPrefab(id);
    // Синхронизируем текстовые буферы один раз при загрузке
    std::snprintf(m_modelBuf, sizeof(m_modelBuf), "%s",
                  m_edit->m_modelPath.c_str());
    std::snprintf(m_matNormalBuf, sizeof(m_matNormalBuf), "%s",
                  m_edit->normal_material.normal_path.c_str());
    std::snprintf(m_matBurningBuf, sizeof(m_matBurningBuf), "%s",
                  m_edit->normal_material.specular_path.c_str());
    std::snprintf(m_matBurnedBuf, sizeof(m_matBurnedBuf), "%s",
                  m_edit->normal_material.diffuse_path.c_str());
  }
  return id;
}

void PrefabBrowserPanel::DrawLeftColumn(EditorContext &ctx,
                                        const std::vector<std::string> &ids) {
  for (int i = 0; i < (int)ids.size(); ++i) {
    const bool sel = (i == m_selectedIndex);
    if (ImGui::Selectable(ids[i].c_str(), sel)) {
      m_selectedIndex = i;
      // Перезагружаем редактируемую копию и буферы
      m_edit = ctx.m_scene->GetPrefabSystem().GetPrefab(ids[i]);
      std::snprintf(m_modelBuf, sizeof(m_modelBuf), "%s",
                    m_edit->m_modelPath.c_str());
      std::snprintf(m_matNormalBuf, sizeof(m_matNormalBuf), "%s",
                    m_edit->normal_material.normal_path.c_str());
      std::snprintf(m_matBurningBuf, sizeof(m_matBurningBuf), "%s",
                    m_edit->normal_material.specular_path.c_str());
      std::snprintf(m_matBurnedBuf, sizeof(m_matBurnedBuf), "%s",
                    m_edit->normal_material.diffuse_path.c_str());
    }
    if (sel)
      ImGui::SetItemDefaultFocus();
  }
}

  changed |=
      ImGui::ColorEdit3("Burned diffuse", &P.burned_material.diffuse_color[0]);
  changed |= ImGui::ColorEdit3("Burned specular",
                               &P.burned_material.specular_color[0]);

  // Шайнинесс
  changed |= ImGui::DragFloat("Normal shininess", &P.normal_material.shininess,
                              1.0f, 0.0f, 256.0f);
  changed |= ImGui::DragFloat(
      "Burning shininess", &P.burning_material.shininess, 1.0f, 0.0f, 256.0f);
  changed |= ImGui::DragFloat("Burned shininess", &P.burned_material.shininess,
                              1.0f, 0.0f, 256.0f);

  if (changed) {
    P.normal_material.diffuse_path = m_matNormalBuf;
    P.normal_material.specular_path = m_matNormalBuf;
    P.normal_material.normal_path = m_matNormalBuf;

    P.burning_material.diffuse_path = m_matBurningBuf;
    P.burning_material.specular_path = m_matBurningBuf;
    P.burning_material.normal_path = m_matBurningBuf;

    P.burned_material.diffuse_path = m_matBurnedBuf;
    P.burned_material.specular_path = m_matBurnedBuf;
    P.burned_material.normal_path = m_matBurnedBuf;
  }
}

void PrefabBrowserPanel::DrawSaveButtons(EditorContext &ctx,
                                         const std::string &id,
                                         AgentPrefab &P) {
  static float tConfirm = -1000.f;

  if (ImGui::Button("Apply (runtime)")) {
    ctx.m_scene->GetPrefabSystem().SavePrefab(id, P);
    P.LoadMaterials();
    ctx.m_scene->ReloadModelForPrefab(id, P);
    tConfirm = (float)ImGui::GetTime();
  }
  if ((float)ImGui::GetTime() - tConfirm < 1.0f) {
    ImGui::SameLine();
    ImGui::TextColored(ImVec4(0, 1, 0, 1), "Saved");
  }

  ImGui::SameLine();

  if (ImGui::Button("Save to file")) {
    if (m_savePath.empty())
      m_savePath = "res/prefabs/" + id + ".yaml";
    std::snprintf(m_saveBuf, sizeof(m_saveBuf), "%s", m_savePath.c_str());
    m_savePopupOpen = true;
    ImGui::OpenPopup("Save Prefab");
    P.LoadMaterials();
    ctx.m_scene->ReloadModelForPrefab(id, P);
  }
}

void PrefabBrowserPanel::DrawSavePopup(EditorContext &ctx,
                                       const std::string &id, AgentPrefab &P) {
  if (!ImGui::BeginPopupModal("Save Prefab", &m_savePopupOpen,
                              ImGuiWindowFlags_AlwaysAutoResize))
    return;

  ImGui::InputText("Path", m_saveBuf, IM_ARRAYSIZE(m_saveBuf));

  if (ImGui::Button("Save")) {
    m_savePath = m_saveBuf;

    auto &sys = ctx.m_scene->GetPrefabSystem();

    std::string newId;
    {
      std::filesystem::path out(m_savePath);
      newId = out.stem().string();
      if (newId.empty())
        newId = id; // fallback
    }
    sys.SavePrefab(newId, P);

    sys.SavePrefabToFile(newId, m_savePath);

    m_selectedIndex = 0;
    const auto &ids = ctx.m_scene->ListPrefabIds();
    for (int i = 0; i < (int)ids.size(); ++i) {
      if (ids[i] == newId) {
        m_selectedIndex = i;
        break;
      }
    }
    m_edit.reset();
    ImGui::CloseCurrentPopup();
    m_savePopupOpen = false;
  }

  ImGui::SameLine();
  if (ImGui::Button("Cancel")) {
    ImGui::CloseCurrentPopup();
    m_savePopupOpen = false;
  }
  ImGui::EndPopup();
}

#include "PrefabPreviewRenderer.h"
#include "graphics/Renderer.h"
#include <glm/gtc/matrix_transform.hpp>
#include <iostream>
namespace ui {

PrefabPreviewRenderer::PrefabPreviewRenderer(int w, int h)
    : m_width(w), m_height(h) {
  SetupFBO();
  UpdateCamera();
}

PrefabPreviewRenderer::~PrefabPreviewRenderer() {
  glDeleteFramebuffers(1, &m_fbo);
  glDeleteTextures(1, &m_texture);
  glDeleteRenderbuffers(1, &m_rbo);
}

void PrefabPreviewRenderer::SetupFBO() {
  glGenFramebuffers(1, &m_fbo);
  glBindFramebuffer(GL_FRAMEBUFFER, m_fbo);

  glGenTextures(1, &m_texture);
  glBindTexture(GL_TEXTURE_2D, m_texture);
  glTexImage2D(GL_TEXTURE_2D, 0, GL_RGB, m_width, m_height, 0, GL_RGB,
               GL_UNSIGNED_BYTE, nullptr);
  glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
  glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
  glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D,
                         m_texture, 0);

  glGenRenderbuffers(1, &m_rbo);
  glBindRenderbuffer(GL_RENDERBUFFER, m_rbo);
  glRenderbufferStorage(GL_RENDERBUFFER, GL_DEPTH24_STENCIL8, m_width,
                        m_height);
  glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_DEPTH_STENCIL_ATTACHMENT,
                            GL_RENDERBUFFER, m_rbo);

  if (glCheckFramebufferStatus(GL_FRAMEBUFFER) != GL_FRAMEBUFFER_COMPLETE)
    std::cerr << "FBO incomplete!\n";

  glBindFramebuffer(GL_FRAMEBUFFER, 0);
}

void PrefabPreviewRenderer::Resize(int w, int h) {
  if (w == m_width && h == m_height)
    return;
  m_width = w;
  m_height = h;
  glDeleteTextures(1, &m_texture);
  glDeleteRenderbuffers(1, &m_rbo);
  SetupFBO();
}

void PrefabPreviewRenderer::UpdateCamera() {
  glm::vec3 front(cos(glm::radians(m_yaw)) * cos(glm::radians(m_pitch)),
                  sin(glm::radians(m_pitch)),
                  sin(glm::radians(m_yaw)) * cos(glm::radians(m_pitch)));
  front = glm::normalize(front);
  glm::vec3 pos = m_target - front * m_distance;

  m_view = glm::lookAt(pos, m_target, glm::vec3(0, 1, 0));
  m_proj = glm::perspective(glm::radians(45.0f), (float)m_width / m_height,
                            0.1f, 100.0f);
}

void PrefabPreviewRenderer::Rotate(float dyaw, float dpitch) {
  m_yaw += dyaw;
  m_pitch = glm::clamp(m_pitch + dpitch, -89.0f, 89.0f);
  UpdateCamera();
}

void PrefabPreviewRenderer::Zoom(float delta) {
  m_distance = glm::clamp(m_distance - delta, m_minDist, m_maxDist);
  UpdateCamera();
}

void PrefabPreviewRenderer::ResetView() {
  m_yaw = -90.0f;
  m_pitch = 0.0f;
  m_distance = 10.0f;
  UpdateCamera();
}

void PrefabPreviewRenderer::Render(const engine::data::AgentPrefab &P) {
  if (!m_model || m_model->GetPath() != P.m_modelPath) {
    m_model = std::make_shared<engine::Model>(P.m_modelPath);
    m_material = engine::Material::Create();
  }

  glBindFramebuffer(GL_FRAMEBUFFER, m_fbo);
  glViewport(0, 0, m_width, m_height);
  glClearColor(0.2f, 0.2f, 0.2f, 1.0f);
  glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
  glEnable(GL_DEPTH_TEST);

  engine::Renderer::Instance().BeginScene(m_view, m_proj);
  engine::Renderer::Instance().DrawModelWithMaterial(
      *m_model, glm::vec3(0, 0, -10), P.m_modelScale, m_target, m_material);

  // Bounds wireframe
  glPolygonMode(GL_FRONT_AND_BACK, GL_LINE);
  glm::vec3 center = (P.m_bounds_min + P.m_bounds_max) * 0.5f;
  center.z -= 10;
  glm::vec3 size = P.m_bounds_max - P.m_bounds_min;
  engine::Renderer::Instance().DrawCube(center, glm::vec3(1, 0, 0), size);
  glPolygonMode(GL_FRONT_AND_BACK, GL_FILL);

  engine::Renderer::Instance().EndScene();
  glBindFramebuffer(GL_FRAMEBUFFER, 0);
}

// src/editor/EditorExtension.cpp
#include "app/EditorExtension.h"
#include <GLFW/glfw3.h>
#include "core/Application.h"

#include "core/scene/Scene.h"
#include "imgui.h"

EditorExtension::~EditorExtension() noexcept = default;

void EditorExtension::OnInit(engine::Application& app) {
  m_imgui = std::make_unique<ui::ImGuiBackend>();
  m_imgui->Init(static_cast<GLFWwindow*>(app.GetNativeWindow()), "#version 330");

  m_editor = std::make_unique<ui::Editor>();
  m_editor->RegisterModule(std::make_unique<ui::BasicToolsModule>());
    ui::EditorContext ec{};
  ec.m_app    = &app;
  ec.m_window = app.GetNativeWindow();
  ec.m_scene  = &app.GetScene();
  m_editor->SetActiveTool(ui::ToolType::PLACE_ON_GROUND, ec);
}

void EditorExtension::OnBeginUI(engine::Application& /*app*/, ui::EditorContext& ctx) {
  // 1) начинаем кадр ImGui
  m_imgui->BeginFrame();
  m_imgui->DrawDockspace();

  // 2) Получаем актуальный viewport из ImGui
  ImGuiViewport* vp = ImGui::GetMainViewport();
  ctx.vpX = static_cast<int>(vp->WorkPos.x);
  ctx.vpY = static_cast<int>(vp->WorkPos.y);
  ctx.vpW = static_cast<int>(vp->WorkSize.x);
  ctx.vpH = static_cast<int>(vp->WorkSize.y);

  // 3) HiDPI scale
  int fbw = 0, fbh = 0, ww = 0, wh = 0;
  glfwGetFramebufferSize(ctx.m_window, &fbw, &fbh);
  glfwGetWindowSize(ctx.m_window, &ww, &wh);
  ctx.m_vpScaleX = (ww > 0) ? static_cast<float>(fbw) / static_cast<float>(ww) : 1.f;
  ctx.m_vpScaleY = (wh > 0) ? static_cast<float>(fbh) / static_cast<float>(wh) : 1.f;

  // 4) Размеры вьюпорта для проекций (в framebuffer-пикселях)
  ctx.m_viewportW = static_cast<int>(ctx.vpW * ctx.m_vpScaleX);
  ctx.m_viewportH = static_cast<int>(ctx.vpH * ctx.m_vpScaleY);

  // 5) Обновляем матрицы из сцены с ПРАВИЛЬНЫМИ размерами
  ctx.m_view = ctx.m_scene->GetViewMatrix();
  ctx.m_proj = ctx.m_scene->GetProjMatrix(ctx.m_viewportW, ctx.m_viewportH);

  // 6) Сообщаем сцене актуальные параметры viewport
  ctx.m_scene->SetViewportRect(ctx.vpX, ctx.vpY, ctx.vpW, ctx.vpH);
  ctx.m_scene->SetViewportScale(ctx.m_vpScaleX, ctx.m_vpScaleY);

  // 7) Рендер UI редактора
  m_editor->DrawUI(ctx);
  
  // 8) Обновляем состояние захвата ввода
  const ImGuiIO& io = ImGui::GetIO();
  engine::InputSystem::Instance().SetUICapture(io.WantCaptureMouse, io.WantCaptureKeyboard);
  
  // 9) Обновляем активные инструменты
  m_editor->TickTools(ctx);
}


void EditorExtension::OnEndUI(engine::Application&, ui::EditorContext&) {
  // 3) завершаем кадр ImGui (Render + RenderDrawData + платформенные окна)
  m_imgui->EndFrame();
}

void EditorExtension::OnShutdown(engine::Application&) {
  m_editor.reset();  // сначала редактор
  if (m_imgui) {
    m_imgui->Shutdown();
  }
  m_imgui.reset();
}
